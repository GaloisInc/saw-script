   28 -- Support logic
   29 
   30 -- | Context type for the pass.
   31 --
   32 --   We keep a scoped map that remembers which names have been locally
   33 --   defined. If we see the builtin names we're looking for in this
   34 --   map, they don't actually refer to the builtins and shouldn't be
   35 --   rewritten.
   36 data Ctx = Ctx {
   37     ctxShadowed :: IORef (ScopedMap Name Bool)
   38 }
   39 
   40 -- | Create a new context.
   41 newctx :: IO Ctx
   42 newctx = do
   43     shadowed <- IORef.newIORef $ ScopedMap.empty
   44     pure $ Ctx { ctxShadowed = shadowed }
   45 
   46 -- | Enter a new scope.
   47 pushScope :: Ctx -> IO ()
   48 pushScope ctx = do
   49     shadowed <- IORef.readIORef (ctxShadowed ctx)
   50     let shadowed' = ScopedMap.push shadowed
   51     IORef.writeIORef (ctxShadowed ctx) shadowed'
   52 
   53 -- | Leave a scope.
   54 popScope :: Ctx -> IO ()
   55 popScope ctx = do
   56     shadowed <- IORef.readIORef (ctxShadowed ctx)
   57     let shadowed' = ScopedMap.pop shadowed
   58     IORef.writeIORef (ctxShadowed ctx) shadowed'
   59 
   60 -- | Mark a variable as locally defined.
   61 shadow :: Ctx -> Name -> IO ()
   62 shadow ctx x = do
   63     shadowed <- IORef.readIORef (ctxShadowed ctx)
   64     let shadowed' = ScopedMap.insert x True shadowed
   65     IORef.writeIORef (ctxShadowed ctx) shadowed'
   66 
   67 -- | Check if a variable is locally (re)defined.
   68 isShadowed :: Ctx -> Name -> IO Bool
   69 isShadowed ctx x = do
   70     shadowed <- IORef.readIORef (ctxShadowed ctx)
   71     pure $ case ScopedMap.lookup x shadowed of
   72         Nothing -> False
   73         Just t -> t
   74

   79 -- | Process an expression.
   80 --
   81 --   If we see @subshell@ or @proof_subshell@, convert to the
   82 --   special-case version.
   83 --
   84 subshell'expr :: Ctx -> Expr -> IO Expr
   85 subshell'expr ctx e0 = case e0 of
   86     Bool{} -> pure e0
   87     String{} -> pure e0
   88     Int{} -> pure e0
   89     Code{} -> pure e0
   90     CType{} -> pure e0
   91     Array pos es -> do
   92         es' <- mapM (subshell'expr ctx) es
   93         pure $ Array pos es'
   94     Block pos (stmts, e1) -> do
   95         pushScope ctx
   96         stmts' <- mapM (subshell'stmt ctx) stmts
   97         e1' <- subshell'expr ctx e1
   98         popScope ctx
   99         pure $ Block pos (stmts', e1')
  100     Tuple pos es -> do
  101         es' <- mapM (subshell'expr ctx) es
  102         pure $ Tuple pos es'
  103     Record pos etbl -> do
  104         etbl' <- traverse (subshell'expr ctx) etbl
  105         pure $ Record pos etbl'
  106     Index pos e1 e2 -> do
  107         e1' <- subshell'expr ctx e1
  108         e2' <- subshell'expr ctx e2
  109         pure $ Index pos e1' e2'
  110     Lookup pos e1 field -> do
  111         e1' <- subshell'expr ctx e1
  112         pure $ Lookup pos e1' field
  113     TLookup pos e1 field -> do
  114         e1' <- subshell'expr ctx e1
  115         pure $ TLookup pos e1' field
  116     Var pos x -> do
  117         unshadowed <- not <$> isShadowed ctx x
  118         pure $ case x of
  119             "subshell" | unshadowed -> Subshell pos emptyAnnotations
  120             "proof_subshell" | unshadowed -> ProofSubshell pos emptyAnnotations
  121             _ -> Var pos x
  122     Subshell{} ->
  123         panic "subshell'expr" ["Preexisting Subshell expression"]
  124     ProofSubshell{} ->
  125         panic "subshell'expr" ["Preexisting ProofSubshell expression"]
  126     Lambda pos mname paramPos params namedParams body -> do
  127         pushScope ctx
  128         mapM_ (subshell'pat ctx) params
  129         namedParams' <- do
  130             let once (namepos, (patpos, e, pat)) = do
  131                   -- A default argument value is an unlikely place to
  132                   -- find a subshell, but you never know...
  133                   e' <- subshell'expr ctx e
  134                   subshell'pat ctx pat
  135                   pure $ (namepos, (patpos, e', pat))
  136             traverse once namedParams
  137         body' <- subshell'expr ctx body
  138         popScope ctx
  139         pure $ Lambda pos mname paramPos params namedParams' body'
  140     Application pos e1 args -> do
  141         e1' <- subshell'expr ctx e1
  142         args' <- do
  143             let once (mname, e2) = do
  144                   e2' <- subshell'expr ctx e2
  145                   pure (mname, e2')
  146             mapM once args
  147         pure $ Application pos e1' args'
  148     Let pos dg e1 -> do
  149         dg' <- subshell'declgroup ctx dg
  150         pushScope ctx
  151         e1' <- subshell'expr ctx e1
  152         popScope ctx
  153         pure $ Let pos dg' e1'
  154     TSig pos e1 ty -> do
  155         e1' <- subshell'expr ctx e1
  156         pure $ TSig pos e1' ty
  157     IfThenElse pos e1 e2 e3 -> do
  158         e1' <- subshell'expr ctx e1
  159         e2' <- subshell'expr ctx e2
  160         e3' <- subshell'expr ctx e3
  161         pure $ IfThenElse pos e1' e2' e3'
  162 
  163 -- | Process a pattern.
  164 subshell'pat :: Ctx -> Pattern -> IO ()
  165 subshell'pat ctx pat0 = case pat0 of
  166     PImplicit{} -> pure ()
  167     PWild{} -> pure ()
  168     PVar _allpos _xpos x _mty -> shadow ctx x
  169     PTuple _pos pats -> mapM_ (subshell'pat ctx) pats
  170 
  171 -- | Process a declaration group.
  172 subshell'declgroup :: Ctx -> DeclGroup -> IO DeclGroup
  173 subshell'declgroup ctx dg = case dg of
  174     Recursive ds -> do
  175         -- Mutually recursive decls: all the bound names go in scope
  176         -- at once, then after that we inspect all the bodies.
  177         mapM_ (subshell'pat ctx) $ map dPat ds
  178         let once d = do
  179               pushScope ctx
  180               def' <- subshell'expr ctx $ dDef d
  181               popScope ctx
  182               pure $ d { dDef = def' }
  183         ds' <- mapM once ds
  184         pure $ Recursive ds'
  185     NonRecursive d -> do
  186         subshell'pat ctx $ dPat d
  187         pushScope ctx
  188         def' <- subshell'expr ctx $ dDef d
  189         popScope ctx
  190         let d' = d { dDef = def' }
  191         pure $ NonRecursive d'
  192 
  193 -- | Process a statement.
  194 subshell'stmt :: Ctx -> Stmt -> IO Stmt
  195 subshell'stmt ctx s0 = case s0 of
  196     StmtBind spos pat e -> do
  197         subshell'pat ctx pat
  198         e' <- subshell'expr ctx e
  199         pure $ StmtBind spos pat e'
  200     StmtLet pos rb dg -> do
  201         dg' <- subshell'declgroup ctx dg
  202         pure $ StmtLet pos rb dg'
  203     StmtCode{} -> pure s0
  204     StmtImport{} -> pure s0
  205     StmtInclude{} -> pure s0
  206     StmtTypedef{} -> pure s0
  207     StmtPushdir{} -> pure s0
  208     StmtPopdir{} -> pure s0
  209

  211 ------------------------------------------------------------
  212 -- External interface
  213 
  214 processExpr :: Expr -> IO Expr
  215 processExpr e = do
  216     ctx <- newctx
  217     e' <- subshell'expr ctx e
  218     checkFail
  219     pure e'
  220 
  221 -- | Run the pass, and return either errors or an updated tree.
  222 processStmts :: [Stmt] -> IO [Stmt]
  223 processStmts stmts = do
  224     ctx <- newctx
  225     stmts' <- mapM (subshell'stmt ctx) stmts
  226     checkFail
  227     pure stmts'
