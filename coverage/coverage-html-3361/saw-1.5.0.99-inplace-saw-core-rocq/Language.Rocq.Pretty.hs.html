   27 -- | Replace all occurrences of the double quote character @"@ with
   28 --   the string @""@, i.e., two copies of it, as this is how Rocq
   29 --   escapes double quote characters.
   30 escapeStringLit :: String -> String
   31 escapeStringLit = concatMap (\c -> if c == '"' then "\"\"" else [c])

   43 -- | Like hsep, but behaves usefully for lists that might be empty:
   44 --   returns the empty doc like hsep for empty lists, but for
   45 --   non-empty lists prepends horizontal space. The result can then
   46 --   be inserted like this: x <> result <+> y, and will produce
   47 --   either "x y" or "x result y" but not "x  y", which is what
   48 --   happens if you do x <+> hsep ... <+> y.
   49 --
   50 --   This baloney arises because d1 <+> emptyDoc <+> d2 prints two
   51 --   spaces instead of one, which I'd describe as a bug. If you
   52 --   _really_ want to accumulate multiple spaces by concatenating
   53 --   empty docs, which seems against the concept of prettyprinting
   54 --   layout anyway, there can be another way to do that, like the
   55 --   latex "phantom" object.
   56 hsep' :: [PP.Doc ann] -> PP.Doc ann
   57 hsep' docs = case docs of
   58     [] -> PP.emptyDoc
   59     _ : _ -> PP.emptyDoc <+> PP.hsep docs
   60 
   61 -- | Common code to print a name with a type
   62 prettyNameType :: Ident -> Type -> PP.Doc ann
   63 prettyNameType x ty = prettyIdent x <+> ":" <+> prettyTerm PrecNone ty
   64 
   65 -- | Insert parens based on the first argument
   66 parensIf :: Bool -> PP.Doc ann -> PP.Doc ann
   67 parensIf p d = if p then PP.parens d else d

   73 -- | Type to hold the current expression precedence while printing
   74 data Prec
   75   = PrecNone
   76   | PrecLambda
   77   | PrecApp
   78   | PrecAtom
   79   deriving (Eq, Ord)
   80 
   81 -- | Print an `Ident`
   82 prettyIdent :: Ident -> PP.Doc ann
   83 prettyIdent (Ident s) = text s
   84 
   85 -- | Print a `sort`
   86 prettySort :: Sort -> PP.Doc ann
   87 prettySort s = case s of
   88     Prop -> "Prop"
   89     Set -> "Set"
   90     Type -> "Type"
   91 
   92 -- | Print an ordinary (lambda/let) binder
   93 prettyBinder :: Binder -> PP.Doc ann
   94 prettyBinder b = case b of
   95     Binder Explicit x Nothing   -> prettyIdent x
   96     Binder Explicit x (Just ty) -> PP.parens $ prettyNameType x ty
   97     Binder Implicit x Nothing    -> PP.braces $ prettyIdent x
   98     Binder Implicit x (Just ty)  -> PP.braces $ prettyNameType x ty
   99 
  100 -- | Print a pi (forall) binder
  101 --   (we don't seem to have a representation for @exists@)
  102 prettyPiBinder :: PiBinder -> PP.Doc ann
  103 prettyPiBinder b = case b of
  104     PiBinder Explicit Nothing ty ->
  105         prettyTerm PrecApp ty <+> "->"
  106     PiBinder Explicit (Just x) ty ->
  107         "forall" <+> PP.parens (prettyNameType x ty) <> ","
  108     PiBinder Implicit Nothing ty ->
  109         PP.braces (prettyTerm PrecApp ty) <+> "->"
  110     PiBinder Implicit (Just x) ty ->
  111         "forall" <+> PP.braces (prettyNameType x ty) <> ","
  112 
  113 -- | Print a list of binders
  114 prettyBinders :: [Binder] -> [PP.Doc ann]
  115 prettyBinders bs = map prettyBinder bs
  116 
  117 -- | Print a list of pi-binders
  118 prettyPiBinders :: [PiBinder] -> PP.Doc ann
  119 prettyPiBinders bs = PP.hsep $ map prettyPiBinder bs

  121 -- | Common code for printing things shaped like function headers
  122 --
  123 prettyFnHeader ::
  124     PP.Doc ann -> [PP.Doc ann] -> Maybe (PP.Doc ann) -> PP.Doc ann ->
  125     PP.Doc ann
  126 prettyFnHeader intro bindings mbRet sep =
  127     -- Long form
  128     --    fun x y z ...
  129     --         a b c ...
  130     --         : ty
  131     --       =>
  132     -- Short form:
  133     --    fun x : ty =>
  134     --
  135     -- (the above where intro is "fun", sep is "=>")
  136     --
  137     let firstpieces = intro : bindings
  138         shortHeader = case mbRet of
  139             Nothing -> PP.group $ PP.hsep firstpieces <+> sep
  140             Just ret -> PP.group $ PP.hsep firstpieces <+> ":" <+> ret <+> sep
  141         longBindings = case mbRet of
  142             Nothing -> PP.fillSep firstpieces
  143             Just ret -> PP.fillSep firstpieces <> PP.line <> ":" <+> ret
  144         longHeader = PP.nest 5 longBindings <> PP.line <> PP.indent 3 sep
  145     in
  146     PP.flatAlt longHeader shortHeader
  147 
  148 -- | Common code for printing things shaped like functions
  149 --   (@header@ will normally be the result of `prettyFnHeader`)
  150 prettyFunction :: PP.Doc ann -> PP.Doc ann -> PP.Doc ann
  151 prettyFunction header body =
  152     -- Long form with long argument list and long body:
  153     --    fun x y z ...
  154     --         a b c ...
  155     --       =>
  156     --       ...
  157     --       x + 1
  158     -- Long form with short argument list and long body:
  159     --    fun x =>
  160     --       ...
  161     --       x + 1
  162     -- Long form with long argument list and short body:
  163     --    fun x y z ...
  164     --         a b c ...
  165     --       => x + 1
  166     -- Short form:
  167     --    fun x => x + 1
  168     --
  169     let longbody = PP.line <> PP.indent 3 body <> PP.line
  170         shortbody = PP.group body
  171         body' = PP.flatAlt longbody shortbody
  172         long = header <> body'
  173         short = PP.group $ header <+> body
  174     in
  175     PP.flatAlt long short
  176 
  177 -- | Print a term
  178 prettyTerm :: Prec -> Term -> PP.Doc ann
  179 prettyTerm p e0 =
  180   case e0 of
  181     Lambda binders e1 ->
  182       let binders' = prettyBinders binders
  183           e1' = prettyTerm PrecLambda e1
  184           header = prettyFnHeader "fun" binders' Nothing "=>"
  185       in
  186       parensIf (p > PrecLambda) $ prettyFunction header e1'
  187     Fix ident binders returnType body ->
  188       let ident' = prettyIdent ident
  189           binders' = prettyBinders binders
  190           returnType' = Just $ prettyTerm PrecNone returnType
  191           body' = prettyTerm PrecLambda body
  192           intro = "fix" <+> ident'
  193           header = prettyFnHeader intro binders' returnType' ":="
  194       in
  195       parensIf (p > PrecLambda) $ prettyFunction header body'
  196     Pi bs t ->
  197       let bs' = prettyPiBinders bs
  198           t' = prettyTerm PrecLambda t
  199           long = bs' <> PP.line <> t'
  200           short = PP.group $ bs' <+> t'
  201       in
  202       parensIf (p > PrecLambda) $ PP.flatAlt long short
  203     Let x bs mty t body ->
  204       let x' = prettyIdent x
  205           bs' = prettyBinders bs
  206           mty' = prettyTerm PrecNone <$> mty
  207           t' = prettyTerm PrecNone t
  208           body' = prettyTerm PrecLambda body
  209           intro = "let" <+> x'
  210           header = prettyFnHeader intro bs' mty' ":="
  211           longest = PP.vsep [header, PP.indent 3 t', "in", body']
  212           second = PP.vsep [PP.group (header <+> t' <+> "in"), body']
  213           shortest = PP.group (header <+> t' <+> "in" <+> body')
  214           shorter = PP.flatAlt second shortest
  215       in
  216       parensIf (p > PrecLambda) $ PP.flatAlt longest shorter
  217     If c t f ->
  218       let c' = prettyTerm PrecNone c
  219           t' = prettyTerm PrecNone t
  220           f' = prettyTerm PrecLambda f
  221           first = "if" <+> c' <+> "then"
  222           long = PP.vsep [first, PP.indent 3 t', "else", PP.indent 3 f']
  223           short = PP.group (first <+> t' <+> "else" <+> f')
  224       in
  225       parensIf (p > PrecLambda) $ PP.flatAlt long short
  226     App f [] ->
  227       prettyTerm p f
  228     App f args ->
  229       let f' = prettyTerm PrecApp f
  230           args' = map (prettyTerm PrecAtom) args
  231       in
  232       parensIf (p > PrecApp) $ PP.nest 5 $ PP.fillSep (f' : args')
  233     Sort s ->
  234       prettySort s
  235     Var x ->
  236       prettyIdent x
  237     ExplVar x ->
  238       let x' = prettyIdent x in
  239       parensIf (p > PrecApp) $ "@" <> x'
  240     Ascription tm tp ->
  241       let tm' = prettyTerm PrecApp tm
  242           tp' = prettyTerm PrecApp tp
  243           long = PP.nest 5 $ PP.fillSep [tm' <> ":", tp']
  244           short = PP.group $ tm' <> ":" <+> tp'
  245       in
  246       parensIf (p > PrecLambda) $ PP.flatAlt long short
  247     NatLit i ->
  248       if i > 1000 then
  249         -- Explicitly convert from Z if an integer is too big
  250         parensIf (p > PrecLambda) ("Z.to_nat" <+> integer i <> "%Z")
  251       else
  252         integer i
  253     ZLit i ->
  254       -- we use hex unless our integer is a positive or negative digit
  255       -- XXX: this cannot possibly work as intended for negative values
  256       if abs i > 9 then
  257           let ui = toInteger (fromInteger i :: Word64)
  258               ui' = showHex ui []
  259           in
  260           text ("0x" ++ ui' ++ "%Z")
  261       else if i < 0 then
  262           text ("(" ++ show i ++ ")%Z")
  263       else
  264           text (show i ++ "%Z")
  265     List ts ->
  266       let ts' = PP.punctuate ";" $ map (prettyTerm PrecNone) ts
  267           long = PP.brackets (PP.line <> PP.indent 3 (PP.vsep ts') <> PP.line)
  268           short = PP.group $ PP.brackets $ PP.hsep ts'
  269       in
  270       PP.flatAlt long short
  271     StringLit s ->
  272       PP.dquotes (text $ escapeStringLit s)
  273     Scope term scope ->
  274       let term' = prettyTerm PrecAtom term
  275           scope' = text scope
  276       in
  277       term' <> "%" <> scope'
  278     Ltac s ->
  279       "ltac:" <> PP.parens (text s)

  281 -- | Print an optional type annotation
  282 prettyMaybeTy :: Maybe Type -> PP.Doc ann
  283 prettyMaybeTy Nothing = PP.emptyDoc
  284 prettyMaybeTy (Just ty) = ":" <+> prettyTerm PrecNone ty
  285 
  286 -- | Print a single constructor
  287 prettyConstructor :: Constructor -> PP.Doc ann
  288 prettyConstructor (Constructor {..}) =
  289   let name' = prettyIdent constructorName
  290       ty' = prettyTerm PrecNone constructorType
  291   in
  292   PP.nest 2 $ "|" <+> name' <+> ":" <+> ty'
  293 
  294 -- | Print an inductive type declaration
  295 prettyInductive :: Inductive -> PP.Doc ann
  296 prettyInductive (Inductive {..}) =
  297   let name' = prettyIdent inductiveName
  298       params' = hsep' $ map prettyBinder inductiveParameters
  299       indices' = hsep' $ map prettyPiBinder inductiveIndices
  300       sort' = prettySort inductiveSort
  301       ctors' = map prettyConstructor inductiveConstructors
  302       header = "Inductive" <+> name' <> params' <+> ":" <> indices' <+> sort' <+> ":="
  303   in
  304   PP.vsep ([PP.nest 2 header] ++ ctors' ++ ["."]) <> PP.hardline
  305 
  306 -- | Common code for the simple declarations
  307 prettyBasicDecl :: PP.Doc ann -> Ident -> Type -> PP.Doc ann
  308 prettyBasicDecl what nm ty =
  309   let nm' = prettyIdent nm
  310       ty' = prettyTerm PrecNone ty
  311   in
  312   PP.nest 2 (what <+> nm' <+> ":" <+> ty' <+> ".") <> PP.hardline
  313 
  314 -- | Print a top-level declaration
  315 prettyDecl :: Decl -> PP.Doc ann
  316 prettyDecl decl = case decl of
  317   Axiom nm ty -> prettyBasicDecl "Axiom" nm ty
  318   Parameter nm ty -> prettyBasicDecl "Parameter" nm ty
  319   Variable nm ty -> prettyBasicDecl "Variable" nm ty
  320   Comment s ->
  321     "(*" <+> text s <+> "*)" <> PP.hardline
  322   Definition nm bs mty body ->
  323     let nm' = prettyIdent nm
  324         bs' = hsep' $ map prettyBinder bs
  325         mty' = prettyMaybeTy mty
  326         body' = prettyTerm PrecNone body
  327     in
  328     PP.nest 2 (
  329       PP.vsep [
  330           "Definition" <+> nm' <> bs' <+> mty' <+> ":=",
  331           body' <> "."
  332       ]
  333     ) <> PP.hardline
  334   InductiveDecl ind ->
  335     prettyInductive ind
  336   Section nm ds ->
  337     let nm' = prettyIdent nm
  338         ds' = map (PP.indent 2 . prettyDecl) ds
  339         header = "Section" <+> nm' <+> "."
  340         footer = "End" <+> nm' <+> "."
  341     in
  342     -- XXX vsep issues soft newlines and there should be a hard newline
  343     -- after the head and after the foot. (Note that every other Decl
  344     -- always ends in a hard newline, so ds' is ok.)
  345     -- (XXX: Does `PP.vsep` on top of `PP.hardline` generate two lines?)
  346     PP.vsep $ [header] ++ ds' ++ [footer]
  347   Snippet s ->
  348     text s
