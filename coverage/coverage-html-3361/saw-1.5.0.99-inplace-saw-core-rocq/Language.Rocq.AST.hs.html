    3 {- |
    4 Module      : Language.Rocq.AST
    5 Description : AST definition for Rocq exporter

    9 This module declares a (simplified) Rocq/Gallina AST for the back end
   10 of the Rocq exporter.
   11 -}
   12 
   13 module Language.Rocq.AST (
   14     Ident(..),
   15     Sort(..),
   16     BinderImplicity(..),
   17     Binder(..),
   18     PiBinder(..),
   19     Term(..),
   20     Type,
   21     Constructor(..),
   22     Inductive(..),
   23     Decl(..)
   24   ) where
   25 
   26 import Data.String (IsString(..))
   27 import Data.Text (Text)
   28 import qualified Data.Text as Text
   29 
   30 -- | An 'Ident' is a Rocq qualified identifier represented as a
   31 --   string, with the invariant that it is lexically valid.
   32 --
   33 --   A valid Rocq identifier is a sequence of letters, digits,
   34 --   underscores and primes that starts with a letter or underscore.
   35 --
   36 --   A /qualified/ identifier is a sequence of one or more identifiers
   37 --   separated by periods.
   38 --
   39 --   We don't enforce the distinction between qualified and unqualified
   40 --   identifiers in this representation.
   41 --
   42 newtype Ident = Ident Text
   43   deriving (Eq, Ord)
   44 
   45 instance Show Ident where
   46   show (Ident s) = show s
   47 
   48 instance IsString Ident where
   49   fromString s = Ident $ Text.pack s
   50 
   51 -- | Type to hold universes
   52 --
   53 data Sort
   54   = Prop
   55   | Set
   56   | Type
   57   deriving (Eq, Show)
   58 
   59 -- | Is this a maximally-inserted implicit ("{}") or explicit binder?
   60 data BinderImplicity
   61   = Implicit
   62   | Explicit
   63     deriving (Eq, Show)
   64 
   65 -- | Bound variable in a `Lambda` or `Let`, optionally with type
   66 data Binder
   67   = Binder BinderImplicity Ident (Maybe Type)
   68     deriving (Show)
   69 
   70 -- | Bound variable in a `Pi` (forall), optionally with a name
   71 --
   72 --   XXX: we can't represent "exists". Currently we never generate it,
   73 --   but that could change.
   74 data PiBinder
   75   = PiBinder BinderImplicity (Maybe Ident) Type
   76     deriving (Show)
   77 
   78 -- | Type to hold Gallina expressions
   79 --
   80 data Term
   81 
   82     -- | fun x => e
   83   = Lambda [Binder] Term
   84 
   85     -- | Fix f x : ty := e
   86   | Fix Ident [Binder] Type Term
   87 
   88     -- | forall x, e
   89   | Pi [PiBinder] Term
   90 
   91     -- | let f arg [: ty] := e in e
   92   | Let Ident [Binder] (Maybe Type) Term Term
   93 
   94     -- | if e then e else e
   95   | If Term Term Term
   96 
   97     -- | f args
   98   | App Term [Term]
   99 
  100     -- | Prop or Type
  101   | Sort Sort
  102 
  103    -- | x
  104   | Var Ident
  105   
  106     -- | @x, that is, Var with @ in order to explicitly apply implicit
  107     --   arguments.
  108   | ExplVar Ident
  109 
  110     -- | Type annotation e: ty
  111   | Ascription Term Type
  112 
  113     -- | integer constant in nat
  114   | NatLit Integer
  115   
  116     -- | integer constant in Z
  117   | ZLit Integer
  118 
  119     -- | [e1; e2; ...]
  120   | List [Term]
  121 
  122     -- | "foo"
  123   | StringLit Text
  124 
  125     -- | e%scope
  126   | Scope Term Text
  127 
  128     -- | Expression-level ltac invocation ltac:(text)
  129   | Ltac Text
  130 
  131   deriving (Show)
  132 
  133 -- | Type synonym useful for indicating when a term is used as a type.
  134 type Type = Term
  135 
  136 -- | Single constructor declaration in an inductive type declaration.
  137 --
  138 --   Because saw-core does not give very helpful access to the parameters and
  139 --   indices, we just follow that style and define the constructor by its fully
  140 --   applied return type.
  141 --
  142 --   NOTE: constructor names must be unqualified.
  143 --
  144 data Constructor = Constructor
  145   { constructorName    :: Ident
  146   , constructorType    :: Type
  147   }
  148   deriving (Show)
  149 
  150 -- | Inductive type declaration
  151 --
  152 data Inductive = Inductive
  153   { inductiveName         :: Ident
  154   , inductiveParameters   :: [Binder]
  155   , inductiveIndices      :: [PiBinder]
  156   , inductiveSort         :: Sort
  157   , inductiveConstructors :: [Constructor]
  158   }
  159   deriving (Show)
  160 
  161 -- | Arbitrary top-level declaration.
  162 --
  163 --   Does not support modules or functors.
  164 --
  165 --   `Snippet` inserts raw text, presumably for things this AST can't
  166 --   represent. XXX: probably shouldn't need that
  167 --
  168 data Decl
  169   = Axiom Ident Type
  170   | Comment Text
  171   | Definition Ident [Binder] (Maybe Type) Term
  172   | Parameter Ident Type
  173   | Variable Ident Type
  174   | InductiveDecl Inductive
  175   | Section Ident [Decl]
  176   | Snippet Text
  177   deriving (Show)
