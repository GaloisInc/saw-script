    8 This module declares a (simplified) Rocq/Gallina AST for the back end
    9 of the Rocq exporter.
   10 -}
   11 
   12 module Language.Rocq.AST (
   13     Ident(..),
   14     Sort(..),
   15     BinderImplicity(..),
   16     Binder(..),
   17     PiBinder(..),
   18     Term(..),
   19     Type,
   20     Constructor(..),
   21     Inductive(..),
   22     Decl(..)
   23   ) where
   24 
   25 import Data.String (IsString(..))
   26 
   27 -- | An 'Ident' is a Rocq qualified identifier represented as a
   28 --   string, with the invariant that it is lexically valid.
   29 --
   30 --   A valid Rocq identifier is a sequence of letters, digits,
   31 --   underscores and primes that starts with a letter or underscore.
   32 --
   33 --   A /qualified/ identifier is a sequence of one or more identifiers
   34 --   separated by periods.
   35 --
   36 --   We don't enforce the distinction between qualified and unqualified
   37 --   identifiers in this representation.
   38 --
   39 newtype Ident = Ident String
   40   deriving (Eq, Ord)
   41 
   42 instance Show Ident where
   43   show (Ident s) = show s
   44 
   45 instance IsString Ident where
   46   fromString s = Ident s
   47 
   48 -- | Type to hold universes
   49 --
   50 data Sort
   51   = Prop
   52   | Set
   53   | Type
   54   deriving (Show)
   55 
   56 -- | Is this a maximally-inserted implicit ("{}") or explicit binder?
   57 data BinderImplicity
   58   = Implicit
   59   | Explicit
   60     deriving (Show)
   61 
   62 -- | Bound variable in a `Lambda` or `Let`, optionally with type
   63 data Binder
   64   = Binder BinderImplicity Ident (Maybe Type)
   65     deriving (Show)
   66 
   67 -- | Bound variable in a `Pi` (forall), optionally with a name
   68 data PiBinder
   69   = PiBinder BinderImplicity (Maybe Ident) Type
   70     deriving (Show)
   71 
   72 -- | Type to hold Gallina expressions
   73 --
   74 data Term
   75 
   76     -- | fun x => e
   77   = Lambda [Binder] Term
   78 
   79     -- | Fix f x : ty := e
   80   | Fix Ident [Binder] Type Term
   81 
   82     -- | forall x, e
   83   | Pi [PiBinder] Term
   84 
   85     -- | let f arg [: ty] := e in e
   86   | Let Ident [Binder] (Maybe Type) Term Term
   87 
   88     -- | if e then e else e
   89   | If Term Term Term
   90 
   91     -- | f args
   92   | App Term [Term]
   93 
   94     -- | Prop or Type
   95   | Sort Sort
   96 
   97    -- | x
   98   | Var Ident
   99   
  100     -- | @x, that is, Var with @ in order to explicitly apply implicit
  101     --   arguments.
  102   | ExplVar Ident
  103 
  104     -- | Type annotation e: ty
  105   | Ascription Term Type
  106 
  107     -- | integer constant in nat
  108   | NatLit Integer
  109   
  110     -- | integer constant in Z
  111   | ZLit Integer
  112 
  113     -- | [e1; e2; ...]
  114   | List [Term]
  115 
  116     -- | "foo"
  117   | StringLit String
  118 
  119     -- | e%scope
  120   | Scope Term String
  121 
  122     -- | Expression-level ltac invocation ltac:(text)
  123   | Ltac String
  124 
  125   deriving (Show)
  126 
  127 -- | Type synonym useful for indicating when a term is used as a type.
  128 type Type = Term
  129 
  130 -- | Single constructor declaration in an inductive type declaration.
  131 --
  132 --   Because saw-core does not give very helpful access to the parameters and
  133 --   indices, we just follow that style and define the constructor by its fully
  134 --   applied return type.
  135 --
  136 --   NOTE: constructor names must be unqualified.
  137 --
  138 data Constructor = Constructor
  139   { constructorName    :: Ident
  140   , constructorType    :: Type
  141   }
  142   deriving (Show)
  143 
  144 -- | Inductive type declaration
  145 --
  146 data Inductive = Inductive
  147   { inductiveName         :: Ident
  148   , inductiveParameters   :: [Binder]
  149   , inductiveIndices      :: [PiBinder]
  150   , inductiveSort         :: Sort
  151   , inductiveConstructors :: [Constructor]
  152   }
  153   deriving (Show)
  154 
  155 -- | Arbitrary top-level declaration.
  156 --
  157 --   Does not support modules or functors.
  158 --
  159 --   `Snippet` inserts raw text, presumably for things this AST can't
  160 --   represent. XXX: probably shouldn't need that
  161 --
  162 data Decl
  163   = Axiom Ident Type
  164   | Comment String
  165   | Definition Ident [Binder] (Maybe Type) Term
  166   | Parameter Ident Type
  167   | Variable Ident Type
  168   | InductiveDecl Inductive
  169   | Section Ident [Decl]
  170   | Snippet String
  171   deriving (Show)
