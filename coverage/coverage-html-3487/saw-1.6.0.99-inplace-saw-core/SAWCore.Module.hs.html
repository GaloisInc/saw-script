   89 -- | A Definition contains a name, the type of the definition, and an
   90 -- optional body (axioms and primitives do not have a body).

   92   Def
   93   { defName :: Name
   94   , defQualifier :: DefQualifier
   95   , defType :: Term
   96   , defBody :: Maybe Term
   97   }
   98   deriving (Eq, Show, Generic)

  103 -- Constructors ----------------------------------------------------------------

  105 -- | A specification of the type of an argument for a constructor of datatype
  106 -- @d@, that has a specified list @ixs@ of indices, inside a context @ctx@ of
  107 -- parameters and earlier arguments
  108 data CtorArg
  109   -- | A fixed, constant type
  110   = ConstArg Term
  111   -- | The construct @'RecursiveArg [(z1,tp1),..,(zn,tpn)] [e1,..,ek]'@
  112   -- specifies a recursive argument type of the form
  113   --
  114   -- > (z1::tp1) -> .. -> (zn::tpn) -> d p1 .. pm e1 .. ek
  115   --
  116   -- where @d@ is the datatype, the @zi::tpi@ are the elements of the Pi
  117   -- context (the first argument to 'RecursiveArgType'), the @pi@ are the
  118   -- parameters of @d@ (not given here), and the @ei@ are the type indices of
  119   -- @d@.
  120   | RecursiveArg [(VarName, Term)] [Term]
  121 
  122 -- | A structure that defines the parameters, arguments, and return type indices
  123 -- of a constructor.
  124 data CtorArgStruct =
  125   CtorArgStruct
  126   { ctorParams :: [(VarName, Term)],
  127     -- ^ earlier ctorParams are in scope
  128     ctorArgs :: [(VarName, CtorArg)],
  129     -- ^ ctorParams and earlier ctorArgs are in scope
  130     ctorIndices :: [Term]
  131     -- ^ ctorParams and ctorArgs are in scope
  132   }
  133 
  134 -- | A specification of a constructor
  135 data Ctor =
  136   Ctor
  137   { ctorName :: !Name
  138     -- ^ The name of this constructor
  139   , ctorArgStruct :: CtorArgStruct
  140     -- ^ Arguments to the constructor
  141   , ctorDataType :: !Name
  142     -- ^ The datatype this constructor belongs to
  143   , ctorNumber :: !Int
  144     -- ^ The 0-based index of this constructor among all the
  145     -- constructors of the datatype.
  146   , ctorType :: Term
  147     -- ^ Cached type of the constructor, which should always be equal to
  148     --
  149     -- > (p1::P1) -> .. -> (pn::Pn) -> (x1::arg1) -> .. -> (xm::argm) ->
  150     -- >  d p1 .. pn ix1 .. ixk
  151     --
  152     -- where the @pi@ are the 'ctorParams', the @argi@ are the types specified
  153     -- by the 'ctorArgs', and the @ixi@ are the 'ctorDataTypeIndices'. Note that
  154     -- this type should always be top-level, i.e., have no free variables.
  155   }
  156 
  157 -- | Return the number of parameters of a constructor
  158 ctorNumParams :: Ctor -> Int
  159 ctorNumParams (Ctor { ctorArgStruct = CtorArgStruct {..}}) =
  160   length ctorParams
  161 
  162 -- | Return the number of non-parameter arguments of a constructor
  163 ctorNumArgs :: Ctor -> Int
  164 ctorNumArgs (Ctor { ctorArgStruct = CtorArgStruct {..}}) =
  165   length ctorArgs

  170 -- | An inductively-defined datatype
  171 data DataType =
  172   DataType
  173   { dtName :: Name
  174     -- ^ The name of this datatype
  175   , dtParams :: [(VarName, Term)]
  176     -- ^ The context of parameters of this datatype
  177   , dtIndices :: [(VarName, Term)]
  178     -- ^ The context of indices of this datatype
  179   , dtSort :: Sort
  180     -- ^ The universe of this datatype
  181   , dtCtors :: [Ctor]
  182     -- ^ The list of constructors of this datatype
  183   , dtType :: Term
  184     -- ^ The cached type of this datatype, which should always be equal to
  185     --
  186     -- > (p1::P1) -> .. -> (pn::Pn) -> (i1::I1) -> .. -> (im::Im) -> 'dtSort'
  187     --
  188     -- where the @pi@ are the 'dtParams' and the @ii@ are the 'dtIndices'. Note
  189     -- that this type should always be top-level, i.e., have no free variables.
  190   , dtMotiveName :: VarName
  191     -- ^ Variable name to use for the motive parameter of the recursor.
  192   , dtArgName :: VarName
  193     -- ^ Variable name to use for the data argument of the recursor.
  194   }
  195 
  196 -- | Return the number of parameters of a datatype
  197 dtNumParams :: DataType -> Int
  198 dtNumParams dt = length $ dtParams dt
  199 
  200 -- | Return the number of indices of a datatype
  201 dtNumIndices :: DataType -> Int
  202 dtNumIndices dt = length $ dtIndices dt

  207 -- | Declarations that can occur in a module
  208 data ModuleDecl = TypeDecl DataType
  209                 | DefDecl Def
  210                     -- | Code to inject directly in-stream when
  211                     --   doing translations.
  212                 | InjectCodeDecl
  213                     Text {- ^ Code namespace -}
  214                     Text {- ^ Code to inject -}

  216 -- | The different sorts of things that a 'Text' name can be resolved to
  217 data ResolvedName
  218   = ResolvedCtor Ctor
  219   | ResolvedDataType DataType
  220   | ResolvedDef Def
  221 
  222 -- | Get the 'Name' of a 'ResolvedName'.
  223 resolvedNameName :: ResolvedName -> Name
  224 resolvedNameName r =
  225   case r of
  226     ResolvedCtor ctor -> ctorName ctor
  227     ResolvedDataType dt -> dtName dt
  228     ResolvedDef def -> defName def
  229 
  230 -- | Get the 'QualName' for a 'ResolvedName'.
  231 resolvedQualName :: ResolvedName -> QualName
  232 resolvedQualName r = nameQualName (resolvedNameName r)
  233 
  234 -- | Get the type of a 'ResolvedName' as a 'Term'.
  235 resolvedNameType :: ResolvedName -> Term
  236 resolvedNameType r =
  237   case r of
  238     ResolvedCtor ctor -> ctorType ctor
  239     ResolvedDataType dt -> dtType dt
  240     ResolvedDef def -> defType def
  241 
  242 -- | Get the 'VarIndex' for a 'ResolvedName'.
  243 resolvedNameVarIndex :: ResolvedName -> VarIndex
  244 resolvedNameVarIndex r = nameIndex (resolvedNameName r)
  245 
  246 -- | Modules define namespaces of datatypes, constructors, and definitions,
  247 -- i.e., mappings from 'Text' names to these objects. A module is allowed to
  248 -- map a 'Text' name to an object defined in a different module. Modules also
  249 -- keep a record of the top-level declarations and the imports that were used to
  250 -- build them.
  251 data Module = Module {
  252           moduleName    :: !ModuleName
  253         , moduleResolveMap :: !(Map Text ResolvedName)
  254         , moduleRDecls   :: [ModuleDecl] -- ^ All declarations in reverse order they were added.
  255         }
  256 
  257 emptyModule :: ModuleName -> Module
  258 emptyModule nm =
  259   Module { moduleName = nm
  260          , moduleResolveMap = Map.empty
  261          , moduleRDecls = []
  262          }
  263 
  264 
  265 -- | Resolve a 'Text' name in the namespace defined by a 'Module', to either a
  266 -- 'Ctor', 'DataType', or 'Def'
  267 resolveName :: Module -> Text -> Maybe ResolvedName
  268 resolveName m str = Map.lookup str (moduleResolveMap m)
  269 
  270 asResolvedCtor :: ResolvedName -> Maybe Ctor
  271 asResolvedCtor = \case { ResolvedCtor ctor -> Just ctor; _ -> Nothing }
  272 
  273 asResolvedDataType :: ResolvedName -> Maybe DataType
  274 asResolvedDataType = \case { ResolvedDataType d -> Just d; _ -> Nothing }
  275 
  276 asResolvedDef :: ResolvedName -> Maybe Def
  277 asResolvedDef = \case { ResolvedDef d -> Just d; _ -> Nothing }
  278

  280 -- | Insert a 'ResolvedName' into a 'Module', adding a mapping from the 'Text'
  281 -- name of that resolved name to it. Signal an error in the case of a name
  282 -- clash, i.e., an existing binding for the same 'Text' name.
  283 insResolvedName :: Module -> ResolvedName -> Module
  284 insResolvedName m nm =
  285   let str = toShortName $ resolvedQualName nm in
  286   if Map.member str (moduleResolveMap m) then
  287     panic "insResolvedName" [
  288         "inserting duplicate name " <> str <> " into module " <>
  289             moduleNameText (moduleName m)
  290     ]
  291   else
  292     m { moduleResolveMap = Map.insert str nm (moduleResolveMap m) }
  293 
  294 -- | @insImport i m@ returns the module obtained by importing @i@ into @m@,
  295 -- using a predicate to specify which names are imported from @i@ into @m@. In
  296 -- the case of name clashes, an error is signaled.
  297 insImport :: (ResolvedName -> Bool) -> Module -> Module -> Module
  298 insImport name_p i m =
  299   (foldl' insResolvedName m $ Map.elems $
  300    Map.filter name_p (moduleResolveMap i))
  301 
  302 -- | Add a declaration for a datatype along with its constructors.
  303 -- Return 'Left' if there is a name clash with the data type name or
  304 -- any constructor name.
  305 insTypeDeclInMap :: DataType -> ModuleMap -> Either Name ModuleMap
  306 insTypeDeclInMap dt mm0 =
  307   case lookupVarIndexInMap (nameIndex (dtName dt)) mm0 of
  308     Just _ -> Left (dtName dt)
  309     Nothing ->
  310       do let mm1 =
  311                case qualNameModule (nameQualName (dtName dt)) of
  312                  Just mname ->
  313                    insDeclInMap mname (TypeDecl dt) mm0
  314                  Nothing ->
  315                    mm0
  316          let rnames = ResolvedDataType dt : map ResolvedCtor (dtCtors dt)
  317          foldM (flip insResolvedNameInMap) mm1 rnames

  319 -- | Get the resolved names that are local to a module
  320 localResolvedNames :: Module -> [ResolvedName]
  321 localResolvedNames m =
  322   filter isLocal (Map.elems (moduleResolveMap m))
  323   where
  324     isLocal :: ResolvedName -> Bool
  325     isLocal r =
  326       case qualNameModule (resolvedQualName r) of
  327         Just mname -> mname == moduleName m
  328         Nothing -> False
  329 
  330 -- | Get all definitions defined in a module
  331 moduleDefs :: Module -> [Def]
  332 moduleDefs =
  333   foldr' (\case { ResolvedDef d -> (d :); _ -> id } ) [] .
  334   localResolvedNames
  335 
  336 -- | Get all declarations that are local to a module, i.e., that defined names
  337 -- that were not imported from some other module
  338 moduleDecls :: Module -> [ModuleDecl]
  339 moduleDecls = reverse . moduleRDecls

  341 -- | The type of mappings from module names to modules
  342 data ModuleMap =
  343   ModuleMap
  344   { mmNameEnv :: !(Map ModuleName DisplayNameEnv)
  345   , mmIndexMap :: !(IntMap ResolvedName) -- keyed by VarIndex
  346   , mmRDecls :: !(Map ModuleName [ModuleDecl])
  347   }
  348 
  349 emptyModuleMap :: ModuleMap
  350 emptyModuleMap =
  351   ModuleMap
  352   { mmNameEnv = Map.empty
  353   , mmIndexMap = IntMap.empty
  354   , mmRDecls = Map.empty
  355   }
  356 
  357 -- | Test whether a 'ModuleName' is in a 'ModuleMap'.
  358 moduleIsLoaded :: ModuleName -> ModuleMap -> Bool
  359 moduleIsLoaded mn mm = Map.member mn (mmRDecls mm)
  360 
  361 loadModule :: Module -> ModuleMap -> ModuleMap
  362 loadModule m mm =
  363   ModuleMap
  364   { mmNameEnv = Map.insert (moduleName m) env (mmNameEnv mm)
  365   , mmIndexMap = IntMap.union indexMap (mmIndexMap mm)
  366   , mmRDecls = Map.insert (moduleName m) (moduleRDecls m) (mmRDecls mm)
  367   }
  368   where
  369     env =
  370       foldr' (\(x, r) -> extendDisplayNameEnv (resolvedNameVarIndex r) [x]) emptyDisplayNameEnv $
  371       Map.assocs (moduleResolveMap m)
  372     indexMap = IntMap.fromList [ (resolvedNameVarIndex r, r) | r <- Map.elems (moduleResolveMap m) ]
  373 
  374 findModule :: ModuleName -> ModuleMap -> Maybe Module
  375 findModule mnm mm =
  376   do decls <- Map.lookup mnm (mmRDecls mm)
  377      env <- Map.lookup mnm (mmNameEnv mm)
  378      let rmap =
  379            Map.mapMaybe (\vi -> IntMap.lookup vi (mmIndexMap mm)) $
  380            Map.fromList [ (x, i) | (x, s) <- Map.assocs (displayIndexes env), i <- IntSet.elems s ]
  381      Just $ Module { moduleName = mnm, moduleResolveMap = rmap, moduleRDecls = decls }
  382 
  383 lookupVarIndexInMap :: VarIndex -> ModuleMap -> Maybe ResolvedName
  384 lookupVarIndexInMap vi mm = IntMap.lookup vi (mmIndexMap mm)
  385 
  386 requireNameInMap :: Name -> ModuleMap -> ResolvedName
  387 requireNameInMap nm mm =
  388   case lookupVarIndexInMap (nameIndex nm) mm of
  389     Just r -> r
  390     Nothing -> panic "requireNameInMap" ["Constant not found: " <> ppNameFullyQualified nm]

  392 -- | Get all definitions defined in any module in an entire module map. Note
  393 -- that the returned list might have redundancies if a definition is visible /
  394 -- imported in multiple modules in the module map.
  395 allModuleDefs :: ModuleMap -> [Def]
  396 allModuleDefs mm = mapMaybe asResolvedDef (IntMap.elems (mmIndexMap mm))
  397 
  398 -- | Get all local declarations from all modules in an entire module map
  399 allModuleDecls :: ModuleMap -> [ModuleDecl]
  400 allModuleDecls mm = concat (Map.elems (mmRDecls mm))
  401 
  402 -- | Get all local declarations from all modules in an entire module map that
  403 -- are marked as primitives
  404 allModulePrimitives :: ModuleMap -> [Def]
  405 allModulePrimitives modmap =
  406     [ def
  407     | DefDecl def <- allModuleDecls modmap
  408     , defQualifier def == PrimQualifier
  409     ]
  410 
  411 -- | Get all local declarations from all modules in an entire module map that
  412 -- are marked as axioms
  413 allModuleAxioms :: ModuleMap -> [Def]
  414 allModuleAxioms modmap =
  415     [ def
  416     | DefDecl def <- allModuleDecls modmap
  417     , defQualifier def == AxiomQualifier
  418     ]
  419 
  420 -- | Get all local declarations from all modules in an entire module map that
  421 -- are marked as neither primitives nor axioms
  422 allModuleActualDefs :: ModuleMap -> [Def]
  423 allModuleActualDefs modmap =
  424     [ def
  425     | DefDecl def <- allModuleDecls modmap
  426     , defQualifier def == NoQualifier
  427     ]
  428 
  429 -- | Get all datatypes in all modules in a module map
  430 allModuleDataTypes :: ModuleMap -> [DataType]
  431 allModuleDataTypes mm = mapMaybe asResolvedDataType (IntMap.elems (mmIndexMap mm))
  432 
  433 -- | Get all constructors in all modules in a module map
  434 allModuleCtors :: ModuleMap -> [Ctor]
  435 allModuleCtors mm = mapMaybe asResolvedCtor (IntMap.elems (mmIndexMap mm))

  437 insertResolvedName :: ResolvedName -> ModuleMap -> ModuleMap
  438 insertResolvedName r mm =
  439   mm { mmIndexMap = IntMap.insert vi r (mmIndexMap mm) }
  440   where
  441     vi = resolvedNameVarIndex r
  442 
  443 -- | Insert a 'ResolvedName' into a 'ModuleMap', adding a mapping from
  444 -- the 'Name' of that resolved name to it.
  445 -- Return 'Left' in the case of a name clash, i.e., an existing
  446 -- binding for the same 'Name'.
  447 insResolvedNameInMap :: ResolvedName -> ModuleMap -> Either Name ModuleMap
  448 insResolvedNameInMap r mm =
  449   let mm' = insertResolvedName r mm in
  450   case lookupVarIndexInMap (resolvedNameVarIndex r) mm of
  451     Just _ -> Left (resolvedNameName r)
  452     Nothing ->
  453       case qualNameModule (resolvedQualName r) of
  454         Just mname ->
  455           Right $ mm' { mmNameEnv = Map.insert mname env' (mmNameEnv mm) }
  456           where
  457             vi = resolvedNameVarIndex r
  458             base = toShortName (resolvedQualName r)
  459             env = fromMaybe emptyDisplayNameEnv $ Map.lookup mname (mmNameEnv mm)
  460             env' = extendDisplayNameEnv vi [base] env
  461         Nothing ->
  462           Right mm'
  463 
  464 insDeclInMap :: ModuleName -> ModuleDecl -> ModuleMap -> ModuleMap
  465 insDeclInMap mname decl mm =
  466   mm { mmRDecls = Map.alter (Just . (decl :) . fromMaybe []) mname (mmRDecls mm) }
  467 
  468 -- | Insert a definition into a 'ModuleMap'.
  469 insDefInMap :: Def -> ModuleMap -> Either Name ModuleMap
  470 insDefInMap d mm =
  471   insResolvedNameInMap (ResolvedDef d) $
  472   case qualNameModule (nameQualName (defName d)) of
  473     Just mname -> insDeclInMap mname (DefDecl d) mm
  474     Nothing -> mm
  475 
  476 -- | Insert an injectCode declaration into a 'ModuleMap'.
  477 insInjectCodeInMap :: ModuleName -> Text -> Text -> ModuleMap -> ModuleMap
  478 insInjectCodeInMap mname ns txt = insDeclInMap mname (InjectCodeDecl ns txt)

  480 -- | @insImportInMap p m1 m2@ adds all the names from module @m1@ into
  481 -- module @m2@ in the name table of the 'ModuleMap'. The predicate @p@
  482 -- determines which names to import.
  483 insImportInMap :: (Text -> Bool) -> ModuleName -> ModuleName -> ModuleMap -> ModuleMap
  484 insImportInMap p name1 name2 mm =
  485   mm { mmNameEnv = Map.insert name2 env2' (mmNameEnv mm) }
  486   where
  487     env1 = fromMaybe emptyDisplayNameEnv $ Map.lookup name1 (mmNameEnv mm)
  488     env2 = fromMaybe emptyDisplayNameEnv $ Map.lookup name2 (mmNameEnv mm)
  489     env2' = mergeDisplayNameEnv (filterDisplayNameEnv p env1) env2
