   21 -- Prettyprinting infrastructure.
   22 --
   23 -- This module is meant to be used like this:
   24 --
   25 -- @
   26 --    import qualified SAWSupport.Pretty as PPS
   27 -- @
   28 --
   29 -- or if you like, it's reasonable to import the print functions unqualified:
   30 --
   31 -- @
   32 --    import SAWSupport.Pretty (ppNat, ppTypeConstraint)
   33 --    import qualified SAWSupport.Pretty as PPS
   34 -- @
   35 --
   36 -- This module is a haphazard collection of prettyprinting
   37 -- infrastructure that used to be scattered in random places around
   38 -- the tree. Most of it came from (a) saw-core's Term prettyprinter,
   39 -- (b) saw-script's Value prettyprinter, or (c) saw-script's AST
   40 -- prettyprinter, none of which shared any logic other than the
   41 -- upstream prettyprinter package.
   42 --
   43 -- The naming convention for printing in SAW is that @prettyFoo@
   44 -- generates a `Doc` for a @Foo@ and @ppFoo@ generates `Text` for a
   45 -- @Foo@. Generally the latter calls the former and then renders the
   46 -- `Doc`; for simple cases it's reasonable to do things the other way
   47 -- around.
   48 --
   49 -- This matches the upstream @prettyprinter@ package, where @pretty@
   50 -- generates `Doc` and there is no @pp@. It is the other way around
   51 -- from Cryptol, and seemingly a lot of other Haskell code. Beware.
   52 --
   53 -- XXX: We should crack down on callers using `show` instead of
   54 -- `render` (or `renderText`) to produce output text.
   55 --
   56 -- It would also be nice in the long run to be able to do
   57 --
   58 -- @
   59 --    import qualified Prettyprinter as PP
   60 --    import qualified SAWSupport.Pretty as PP
   61 -- @
   62 --
   63 -- without this causing problems.
   64 --
   65 -- It may also be that this file should be split into two or more
   66 -- modules, one meant to be used qualified and one meant for selective
   67 -- import, or something. It isn't at all clear yet what a good
   68 -- interface is.
   69 --
   70 -- FUTURE: there's also been some talk of adopting Cryptol's technique
   71 -- of wrapping the upstream prettyprinter and exposing a monad, or
   72 -- some other similar way of allowing options to be applied to the
   73 -- document after it's assembled instead of needing to be threaded
   74 -- through everywhere.
   75 --
   76 -- In the meantime, the default options (@PPS.defaultOpts@) should be
   77 -- used only to initialize the system. The idea of those options is
   78 -- that the user can adjust them; therefore, the user's settings
   79 -- should be used when something needs to be printed. We have done a
   80 -- fair amount of work plumbing the current options through so that
   81 -- they're available nearly everywhere in the system.
   82 --

  168 -- | Global options for pretty-printing
  169 data Opts = Opts {
  170     -- | Passed to the @useAscii@ setting of Cryptol's prettyprinter.
  171     --   Default is false.
  172     ppUseAscii :: Bool,
  173 
  174     -- | The base to print integers in; default is 10.
  175     ppBase :: Int,
  176 
  177     -- | Whether to print in color; default is false.
  178     ppColor :: Bool,
  179 
  180     -- | Maximum depth to recurse into SAWCore terms. If not set, no
  181     --   limit. Default is unset.
  182     ppMaxDepth :: Maybe Int,
  183 
  184     -- | The numeric identifiers, as seen in the @memoFresh@ field of
  185     --   @MemoVar@ in the SAWCore prettyprinter, of SAWCore variables
  186     --   that shouldn't be inlined.
  187     ppNoInlineMemoFresh :: [Int],
  188 
  189     -- | The way to display SAWCore memoization variables.
  190     --   Default is Incremental.
  191     ppMemoStyle :: MemoStyle,
  192 
  193     -- | Minimum sharing level required to memoize SAWCore subterms.
  194     --   Default is 2 (i.e., any sharing).
  195     ppMinSharing :: Int
  196  }

  200 -- | How should memoization variables be displayed?
  201 --
  202 -- Note: actual text stylization is the province of @ppMemoVar@; this just
  203 -- describes the semantic information @ppMemoVar@ should be prepared to display.
  204 data MemoStyle
  205   = Incremental
  206   -- ^ 'Incremental' says to display a term's memoization variable with the
  207   -- value of a counter that increments after a term is memoized. The first
  208   -- term to be memoized will be displayed with @1@, the second with @2@, etc.
  209   | Hash Int
  210   -- ^ @Hash i@ says to display a term's memoization variable with the first @i@
  211   -- digits of the term's hash.
  212   | HashIncremental Int
  213   -- ^ @HashIncremental i@ says to display a term's memoization variable with
  214   -- /both/ the first @i@ digits of the term's hash /and/ the value of the
  215   -- counter described in 'Incremental'.
  216   | LabelIncremental Bool
  217   -- ^ Same as 'Incremental', but use term labels as the base name. If the
  218   -- flag is false, then only labels that are valid identifiers are printed.
  219   -- If it is set, all labels are printed, with invalid identifiers escaped
  220   -- as a string.
  221 
  222 -- | Default options for pretty-printing.
  223 --
  224 --   If the default 'ppMemoStyle' changes, be sure to update the help
  225 --   text in the interpreter functions that control the memoization
  226 --   style to reflect this change to users.
  227 defaultOpts :: Opts
  228 defaultOpts = Opts {
  229     ppUseAscii = False,
  230     ppBase = 10,
  231     ppColor = False,
  232     ppMaxDepth = Nothing,
  233     ppNoInlineMemoFresh = mempty,
  234     ppMemoStyle = Incremental,
  235     ppMinSharing = 2
  236  }

  238 -- | Update an `Opts` with a new (smaller) `ppMaxDepth`.
  239 limitMaxDepth :: Int -> Opts -> Opts
  240 limitMaxDepth limit opts =
  241   let adjust depth = case depth of
  242         Nothing -> Just limit
  243         Just d -> Just $ if d > limit then limit else d
  244   in
  245   opts { ppMaxDepth = adjust $ ppMaxDepth opts }

  252 -- | Print a string in double quotes. Does not escape any of the
  253 --   characters.
  254 --
  255 --   This is here as much to make sure that people don't find and use
  256 --   `ppStringLiteral` for miscellaneous quoting as it is because it's
  257 --   all that useful in its own right.
  258 --
  259 prettyStringDQ :: Text -> PP.Doc ann
  260 prettyStringDQ s = "\"" <> PP.pretty s <> "\""
  261 
  262 -- | Print a string literal. Escape characters as needed so it comes
  263 --   out in a form that can be read back in.
  264 --
  265 --   Output Haskell-shaped escape sequences because that is (currently)
  266 --   what we read back in both SAWCore and SAWScript. (Whether that is
  267 --   the way it should be is debatable.)
  268 --
  269 --   Numeric escapes in Haskell string constants have unbounded
  270 --   length, it seems, and the only way to allow a digit to follow one
  271 --   is to insert the empty escape sequence @\&@ after it.
  272 --
  273 --   FUTURE: complexify the code so it only generates @\&@ if the next
  274 --   character is a hex digit.
  275 --
  276 ppStringLiteral :: Text -> Text
  277 ppStringLiteral s = "\"" <> Text.concatMap escapeChar s <> "\""
  278   where
  279     -- Note: there's a stdlib function `Data.Char.showLitChar` that
  280     -- does this... mostly... but it has two fatal shortcomings for
  281     -- use here: first, it doesn't escape quote '"', and second, it
  282     -- doesn't terminate numeric escapes so a string with a digit
  283     -- following something that appears as one comes out corrupted.
  284     --
  285     -- There is a `GHC.Show.showLitString` that attends to these
  286     -- concerns, but isn't standard.
  287     --
  288     -- So do it by hand.
  289     escapeChar c = case c of
  290         -- Catch all the known cases first
  291         '"' -> "\\\""
  292         '\\' -> "\\\\"
  293         '\a' -> "\\a"
  294         '\b' -> "\\b"
  295         '\t' -> "\\t"
  296         '\n' -> "\\n"
  297         '\v' -> "\\v"
  298         '\f' -> "\\f"
  299         '\r' -> "\\r"
  300         _ ->
  301           if Char.isPrint c then
  302               Text.singleton c
  303           else
  304              let c' = showHex (Char.ord c) "" in
  305              "\\x" <> Text.pack c' <> "\\&"

  307 -- | Pretty-print an `Integer` in the correct base
  308 prettyInteger :: Opts -> Integer -> Doc
  309 prettyInteger (Opts{..}) i
  310   | ppBase > 36 = pretty i
  311   | otherwise = prefix <> pretty value
  312   where
  313     prefix = case ppBase of
  314       2  -> "0b"
  315       8  -> "0o"
  316       10 -> mempty
  317       16 -> "0x"
  318       _  -> "0" <> pretty '<' <> pretty ppBase <> pretty '>'
  319 
  320     value  = showIntAtBase (toInteger ppBase) (digits !!) i ""
  321     digits = "0123456789abcdefghijklmnopqrstuvwxyz"
  322 
  323 -- | Pretty-print a `Natural` in the correct base
  324 prettyNatural :: Opts -> Natural -> Doc
  325 prettyNatural opts i = prettyInteger opts (fromIntegral i)
  326 
  327 -- | Pretty-print a `Word64` in the correct base
  328 prettyWord64 :: Opts -> Word64 -> Doc
  329 prettyWord64 opts i = prettyInteger opts (fromIntegral i)

  331 -- | Pretty-print an `Int` in the correct base
  332 prettyInt :: Opts -> Int -> Doc
  333 prettyInt opts i = prettyInteger opts (fromIntegral i)

  335 -- | Pretty-print a type constraint (also known as an ascription) @x : tp@
  336 --   This is the formatting used by SAWCore.
  337 prettyTypeConstraint :: Doc -> Doc -> Doc
  338 prettyTypeConstraint x tp =
  339   PP.hang 2 $ PP.group $ PP.vsep [PP.annotate LocalVarStyle x, ":" <+> tp]
  340 
  341 -- | Pretty-print a type signature.
  342 --   This is the formatting used by SAWScript.
  343 --   XXX: should probably unify with prettyTypeConstraint
  344 prettyTypeSig :: PP.Doc ann -> PP.Doc ann -> PP.Doc ann
  345 prettyTypeSig n t =
  346     -- Allow it to split at the colon
  347     let line1 = n <+> PP.pretty ':'
  348         line2 = t
  349         long = line1 <> PP.line <> PP.indent 3 (PP.align line2)
  350         short = line1 <+> line2
  351     in
  352     PP.flatAlt long short

  354 -- | Concatenate n copies of a doc.
  355 replicate :: Integer -> PP.Doc ann -> PP.Doc ann
  356 replicate n d
  357   | n < 1 = PP.emptyDoc
  358   | True  = d <> replicate (n-1) d
  359 
  360 -- | Add a comma to d1 and append d2.
  361 --   (Functionally internal to commaSepAll.)
  362 commaSepDoc :: PP.Doc ann -> PP.Doc ann -> PP.Doc ann
  363 commaSepDoc d1 d2 = d1 <> PP.comma <+> d2
  364 
  365 -- | Concatenate ds, appending a comma to each.
  366 commaSepAll :: [PP.Doc ann] -> PP.Doc ann
  367 commaSepAll ds = case ds of
  368   [] -> PP.emptyDoc
  369   _  -> foldl1 commaSepDoc ds
  370 
  371 -- | Wrap d in single quotes that match.
  372 --   There's a PP.squotes that uses right-quote on both sides.
  373 --   (There's also a PP.dquotes.)
  374 squotesMatching :: PP.Doc ann -> PP.Doc ann
  375 squotesMatching d =
  376   PP.enclose "`" "'" d
  377 
  378 -- | Generalized layout for let-bindings. The first element in each thruple 
  379 --   is binding name (lhs), the second is the body (rhs). If the flag is set
  380 --   then the entry is printed as an equality (i.e @lhs = rhs@), if it is unset then
  381 --   it is printed as a type constraint (i.e. @lhs : rhs@).
  382 prettyLetBlock :: [(PP.Doc ann, PP.Doc ann, Bool)] -> PP.Doc ann -> PP.Doc ann
  383 prettyLetBlock defs body =
  384   let lets = PP.align $
  385         (PP.concatWith (\x y -> x <> PP.hardline <> y))
  386           (map ppEqn defs)
  387   in PP.vcat
  388         [ "let" PP.<+> PP.align (PP.lbrace PP.<+> lets <> PP.line <> PP.rbrace)
  389         , " in" PP.<+> PP.hang 0 body
  390         ]
  391   where
  392     ppEqn (var,d,b) =
  393       var PP.<+> (if b then "=" else ":") PP.<+> d <> ";"

  398 -- | Render a `Doc` to a `String`.
  399 render :: Opts -> Doc -> String
  400 render opts doc =
  401   TextL.unpack (PP.renderLazy (style (PP.layoutPretty layoutOpts doc)))
  402   where
  403     -- ribbon width 64, with effectively unlimited right margin
  404     layoutOpts = PP.LayoutOptions (PP.AvailablePerLine 8000 0.008)
  405     style = if ppColor opts then PP.reAnnotateS colorStyle else PP.unAnnotateS
  406 
  407 -- | Render a `Doc` to a `Text`.
  408 renderText :: Opts -> Doc -> Text
  409 renderText opts doc =
  410   PP.renderStrict (style (PP.layoutPretty layoutOpts doc))
  411   where
  412     -- ribbon width 64, with effectively unlimited right margin
  413     layoutOpts = PP.LayoutOptions (PP.AvailablePerLine 8000 0.008)
  414     style = if ppColor opts then PP.reAnnotateS colorStyle else PP.unAnnotateS
  415 
  416 -- | Render a `Doc` and print directly to standard output. This may
  417 --   allocate less for larger documents, or at least that's the only
  418 --   obvious reason for this function to exist. Should not be used
  419 --   anywhere other than UI code, because it will do the wrong thing
  420 --   in the remote API server.
  421 renderStdout :: Opts -> Doc -> IO ()
  422 renderStdout opts doc =
  423   PP.renderIO stdout (style (PP.layoutPretty layoutOpts doc))
  424   where
  425     -- ribbon width 64, with effectively unlimited right margin
  426     layoutOpts = PP.LayoutOptions (PP.AvailablePerLine 8000 0.008)
  427     style = if ppColor opts then PP.reAnnotateS colorStyle else PP.unAnnotateS
