   42 -- | Check that we didn't get an unexpected status.
   43 fpCheckStatus :: (BigFloat, Status) -> BigFloat
   44 fpCheckStatus (r,s) =
   45   case s of
   46     MemError  -> panic "fpCheckStatus" [ "libBF: Memory error" ]
   47     _         -> r
   48 
   49 -- | Make LibBF options for the given precision and rounding mode.
   50 fpOpts :: Natural -> Natural -> RoundMode -> BFOpts
   51 fpOpts e p r =
   52   case ok of
   53     Just opts -> opts
   54     Nothing   -> panic "fpOpts" [ "Invalid Float size"
   55                                 , "exponent: " <> Text.pack (show e)
   56                                 , "precision: " <> Text.pack (show p)
   57                                 ]
   58   where
   59     ok :: Maybe BFOpts
   60     ok = do eb <- rng expBits expBitsMin expBitsMax e
   61             pb <- rng precBits precBitsMin precBitsMax p
   62             pure (eb <> pb <> allowSubnormal <> rnd r)
   63 
   64     rng ::
   65       forall a.
   66       Integral a =>
   67       (a -> BFOpts) ->
   68       Int ->
   69       Int ->
   70       Natural ->
   71       Maybe BFOpts
   72     rng f a b x =
   73       if fromIntegral @Int @Natural a <= x && x <= fromIntegral @Int @Natural b
   74         then Just (f (fromIntegral @Natural @a x))
   75         else Nothing
   76 
   77 -- | Mapping from the rounding modes defined in the @Float.cry@ module to the
   78 -- rounding modes of LibBF.
   79 fpRound :: Integer -> Maybe RoundMode
   80 fpRound n =
   81   case n of
   82     0 -> Just NearEven
   83     1 -> Just NearAway
   84     2 -> Just ToPosInf
   85     3 -> Just ToNegInf
   86     4 -> Just ToZero
   87     _ -> Nothing
   88 
   89 -- | Make a floating point number from an integer, using the given rounding
   90 -- mode.
   91 floatFromInteger :: BFOpts -> Integer -> BigFloat
   92 floatFromInteger opts i = fpCheckStatus (bfRoundFloat opts (bfFromInteger i))
   93 
   94 -- | Make a floating point number from a rational, using the given rounding
   95 -- mode.
   96 floatFromRational :: Natural -> Natural -> RoundMode -> Rational -> BigFloat
   97 floatFromRational e p r rat =
   98   fpCheckStatus $
   99   if den == 1
  100     then bfRoundFloat opts num
  101     else bfDiv opts num (bfFromInteger den)
  102   where
  103     opts  = fpOpts e p r
  104 
  105     num   = bfFromInteger (numerator rat)
  106     den   = denominator rat
  107 
  108 -- | Convert a floating-point number to a 'Rational' if possible. Return 'Left'
  109 -- if the input is infinite or NaN.
  110 floatToRational :: BigFloat -> Either NotRationalError Rational
  111 floatToRational bf =
  112   case bfToRep bf of
  113     BFNaN -> Left IsNaN
  114     BFRep s num ->
  115       case num of
  116         Inf  -> Left IsInf
  117         Zero -> Right 0
  118         Num i ev -> Right $
  119           case s of
  120             Pos -> ab
  121             Neg -> negate ab
  122           where ab = fromInteger i * (2 ^^ ev)
  123 
  124 -- | Convert a floating point number to an integer, if possible. Return 'Left'
  125 -- if the input is infinite or NaN.
  126 floatToInteger :: RoundMode -> BigFloat -> Either NotRationalError Integer
  127 floatToInteger r fp =
  128   do rat <- floatToRational fp
  129      pure $
  130        case r of
  131          NearEven -> round rat
  132          NearAway -> roundAway rat
  133          ToPosInf -> ceiling rat
  134          ToNegInf -> floor rat
  135          ToZero   -> truncate rat
  136          _        -> panic "floatToInteger"
  137                            ["Unexpected rounding mode", Text.pack (show r)]
  138   where
  139     -- | Evaluate a rational to an integer with rounding away from zero.
  140     roundAway :: Rational -> Integer
  141     roundAway r' = truncate (r' + signum r' * 0.5)
  142 
  143 -- | Why a floating-point value cannot be represented as a 'Rational'.
  144 data NotRationalError = IsNaN | IsInf
  145 
  146 ppNotRationalError :: NotRationalError -> String
  147 ppNotRationalError IsNaN = "NaN value cannot be represented as a Rational"
  148 ppNotRationalError IsInf = "Infinite value cannot be represented as a Rational"
  149 
  150 floatFromBits ::
  151   Natural {- ^ Exponent width -} ->
  152   Natural {- ^ Precision width -} ->
  153   Integer {- ^ Raw bits -} ->
  154   BigFloat
  155 floatFromBits e p bv = bfFromBits (fpOpts e p NearEven) bv
  156 
  157 -- | Turn a float into raw bits.
  158 -- @NaN@ is represented as a positive "quiet" @NaN@
  159 -- (most significant bit in the significand is set, the rest of it is 0)
  160 floatToBits :: Natural -> Natural -> BigFloat -> Integer
  161 floatToBits e p bf = bfToBits (fpOpts e p NearEven) bf
  162 
  163 -- | Convert a floating point number to an unsigned bitvector. If the value of
  164 -- the float does not lie within the range of possible unsigned bitvector
  165 -- values, then this will return 'Left'.
  166 floatToBV :: Natural -> RoundMode -> BigFloat -> Either NotBitvectorError Integer
  167 floatToBV w r bf =
  168   case floatToInteger r bf of
  169     Left e -> Left $ IsSpecialValue e
  170     Right i ->
  171       if 0 <= i && i <= maxUnsigned
  172         then Right i
  173         else Left IsOutsideRange
  174   where
  175     maxUnsigned :: Integer
  176     maxUnsigned = bit (fromIntegral @Natural @Int w) - 1
  177 
  178 -- | Convert a floating point number to a signed bitvector. If the value of the
  179 -- float does not lie within the range of possible signed bitvector values,
  180 -- then this will return 'Left'.
  181 floatToSBV :: Natural -> RoundMode -> BigFloat -> Either NotBitvectorError Integer
  182 floatToSBV w r bf =
  183   case floatToInteger r bf of
  184     Left e -> Left $ IsSpecialValue e
  185     Right i ->
  186       if minSigned <= i && i <= maxSigned
  187         then Right i
  188         else Left IsOutsideRange
  189   where
  190     signedUpperBound :: Integer
  191     signedUpperBound = bit (fromIntegral @Natural @Int w - 1)
  192 
  193     minSigned, maxSigned :: Integer
  194     minSigned = negate signedUpperBound
  195     maxSigned = signedUpperBound - 1
  196 
  197 -- | Why a floating-point value cannot be represented as a bitvector.
  198 data NotBitvectorError
  199   = IsSpecialValue NotRationalError
  200     -- ^ The value is infinite or NaN.
  201   | IsOutsideRange
  202     -- ^ The value lies outside the range of possible bitvector values.
  203 
  204 ppNotBitvectorError :: NotBitvectorError -> String
  205 ppNotBitvectorError (IsSpecialValue e) = ppNotRationalError e
  206 ppNotBitvectorError IsOutsideRange = "Float lies outside range of possible bitvector values"
