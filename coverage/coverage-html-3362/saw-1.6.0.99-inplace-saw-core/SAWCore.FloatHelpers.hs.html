   40 -- | Check that we didn't get an unexpected status.
   41 fpCheckStatus :: (BigFloat, Status) -> BigFloat
   42 fpCheckStatus (r,s) =
   43   case s of
   44     MemError  -> panic "fpCheckStatus" [ "libBF: Memory error" ]
   45     _         -> r
   46 
   47 -- | Make LibBF options for the given precision and rounding mode.
   48 fpOpts :: Natural -> Natural -> RoundMode -> BFOpts
   49 fpOpts e p r =
   50   case ok of
   51     Just opts -> opts
   52     Nothing   -> panic "fpOpts" [ "Invalid Float size"
   53                                 , "exponent: " <> Text.pack (show e)
   54                                 , "precision: " <> Text.pack (show p)
   55                                 ]
   56   where
   57     ok :: Maybe BFOpts
   58     ok = do eb <- rng expBits expBitsMin expBitsMax e
   59             pb <- rng precBits precBitsMin precBitsMax p
   60             pure (eb <> pb <> allowSubnormal <> rnd r)
   61 
   62     rng ::
   63       forall a.
   64       Integral a =>
   65       (a -> BFOpts) ->
   66       Int ->
   67       Int ->
   68       Natural ->
   69       Maybe BFOpts
   70     rng f a b x =
   71       if fromIntegral @Int @Natural a <= x && x <= fromIntegral @Int @Natural b
   72         then Just (f (fromIntegral @Natural @a x))
   73         else Nothing
   74 
   75 -- | Mapping from the rounding modes defined in the @Float.cry@ module to the
   76 -- rounding modes of LibBF.
   77 fpRound :: Integer -> Maybe RoundMode
   78 fpRound n =
   79   case n of
   80     0 -> Just NearEven
   81     1 -> Just NearAway
   82     2 -> Just ToPosInf
   83     3 -> Just ToNegInf
   84     4 -> Just ToZero
   85     _ -> Nothing
   86 
   87 -- | Make a floating point number from an integer, using the given rounding
   88 -- mode.
   89 floatFromInteger :: BFOpts -> Integer -> BigFloat
   90 floatFromInteger opts i = fpCheckStatus (bfRoundFloat opts (bfFromInteger i))
   91 
   92 -- | Make a floating point number from a rational, using the given rounding
   93 -- mode.
   94 floatFromRational :: Natural -> Natural -> RoundMode -> Rational -> BigFloat
   95 floatFromRational e p r rat =
   96   fpCheckStatus $
   97   if den == 1
   98     then bfRoundFloat opts num
   99     else bfDiv opts num (bfFromInteger den)
  100   where
  101     opts  = fpOpts e p r
  102 
  103     num   = bfFromInteger (numerator rat)
  104     den   = denominator rat
  105 
  106 -- | Convert a floating-point number to a 'Rational' if possible. Return 'Left'
  107 -- if the input is infinite or NaN.
  108 floatToRational :: BigFloat -> Either NotRationalError Rational
  109 floatToRational bf =
  110   case bfToRep bf of
  111     BFNaN -> Left IsNaN
  112     BFRep s num ->
  113       case num of
  114         Inf  -> Left IsInf
  115         Zero -> Right 0
  116         Num i ev -> Right $
  117           case s of
  118             Pos -> ab
  119             Neg -> negate ab
  120           where ab = fromInteger i * (2 ^^ ev)
  121 
  122 -- | Convert a floating point number to an integer, if possible. Return 'Left'
  123 -- if the input is infinite or NaN.
  124 floatToInteger :: RoundMode -> BigFloat -> Either NotRationalError Integer
  125 floatToInteger r fp =
  126   do rat <- floatToRational fp
  127      pure $
  128        case r of
  129          NearEven -> round rat
  130          NearAway -> roundAway rat
  131          ToPosInf -> ceiling rat
  132          ToNegInf -> floor rat
  133          ToZero   -> truncate rat
  134          _        -> panic "floatToInteger"
  135                            ["Unexpected rounding mode", Text.pack (show r)]
  136   where
  137     -- | Evaluate a rational to an integer with rounding away from zero.
  138     roundAway :: Rational -> Integer
  139     roundAway r' = truncate (r' + signum r' * 0.5)
  140 
  141 -- | Why a floating-point value cannot be represented as a 'Rational'.
  142 data NotRationalError = IsNaN | IsInf
  143 
  144 ppNotRationalError :: NotRationalError -> String
  145 ppNotRationalError IsNaN = "NaN value cannot be represented as a Rational"
  146 ppNotRationalError IsInf = "Infinite value cannot be represented as a Rational"
  147 
  148 floatFromBits ::
  149   Natural {- ^ Exponent width -} ->
  150   Natural {- ^ Precision width -} ->
  151   Integer {- ^ Raw bits -} ->
  152   BigFloat
  153 floatFromBits e p bv = bfFromBits (fpOpts e p NearEven) bv
  154 
  155 -- | Turn a float into raw bits.
  156 -- @NaN@ is represented as a positive "quiet" @NaN@
  157 -- (most significant bit in the significand is set, the rest of it is 0)
  158 floatToBits :: Natural -> Natural -> BigFloat -> Integer
  159 floatToBits e p bf = bfToBits (fpOpts e p NearEven) bf
  160 
  161 -- | Convert a floating point number to an unsigned bitvector. See the
  162 -- documentation for @fpToBV@ in Cryptol's @Float.cry@ for an explanation of
  163 -- how exceptional values are treated.
  164 floatToBV :: Natural -> RoundMode -> BigFloat -> Integer
  165 floatToBV w r bf
  166   | bfIsNaN bf
  167   = 0
  168   | bfIsInf bf
  169   = if bfIsNeg bf then 0 else maxUnsigned
  170   | otherwise
  171   = case floatToInteger r bf of
  172       -- 'floatToInteger' can only return 'Left' if the argument is NaN or
  173       -- infinite, which we have already checked above.
  174       Left err ->
  175         panic
  176           "fpToBV"
  177           [ "Unexpected error when converting " <> Text.pack (show bf) <>
  178             " to integer"
  179           , Text.pack (ppNotRationalError err)
  180           ]
  181       Right i
  182         | i < 0 -> 0
  183         | maxUnsigned < i -> maxUnsigned
  184         | otherwise -> i
  185   where
  186     maxUnsigned :: Integer
  187     maxUnsigned = bit (fromIntegral @Natural @Int w) - 1
  188 
  189 -- | Convert a floating point number to a signed bitvector. See the
  190 -- documentation for @fpToSBV@ in Cryptol's @Float.cry@ for an explanation of
  191 -- how exceptional values are treated.
  192 floatToSBV :: Natural -> RoundMode -> BigFloat -> Integer
  193 floatToSBV w r bf
  194   | bfIsNaN bf
  195   = 0
  196   | bfIsInf bf
  197   = if bfIsNeg bf then minSigned else maxSigned
  198   | otherwise
  199   = case floatToInteger r bf of
  200       -- 'floatToInteger' can only return 'Left' if the argument is NaN or
  201       -- infinite, which we have already checked above.
  202       Left err ->
  203         panic
  204           "fpToSBV"
  205           [ "Unexpected error when converting " <> Text.pack (show bf) <>
  206             " to integer"
  207           , Text.pack (ppNotRationalError err)
  208           ]
  209       Right i
  210         | i < minSigned -> minSigned
  211         | maxSigned < i -> maxSigned
  212         | otherwise -> i
  213   where
  214     signedUpperBound :: Integer
  215     signedUpperBound = bit (fromIntegral @Natural @Int w - 1)
  216 
  217     minSigned, maxSigned :: Integer
  218     minSigned = negate signedUpperBound
  219     maxSigned = signedUpperBound - 1
