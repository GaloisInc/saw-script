   78 -- | ImportVisibility is an enumeration that indicates how we handle
   79 -- the visibility of the symbols in an imported module.
   80 --
   81 -- `OnlyPublic` makes only the public/exported symbols of a module
   82 -- visible from SAW. `PublicAndPrivate` instead makes all symbols
   83 -- visible, as if one is working inside it. The latter is often useful
   84 -- (or necessary) for verification and code generation.
   85 --
   86 -- `PublicAndPrivate` is akin to setting the module focus in the
   87 -- Cryptol REPL; however, it uses a simpler internal mechanism and is
   88 -- only settable at import time.
   89 --
   90 -- (See 'CryptolEnv.importCryptolModule'.)
   91 --
   92 -- NOTE: this notion of public vs. private symbols is specific to
   93 -- SAWScript and distinct from Cryptol's notion of private
   94 -- definitions.
   95 --
   96 -- FUTURE: this should probably be replaced with a way to manipulate
   97 -- the module focus like the Cryptol REPL uses. What you really want
   98 -- is not to expose module innards that weren't meant to be exposed,
   99 -- but to go inside to prove things in the module's internal context.
  100 --
  101 data ImportVisibility
  102   = OnlyPublic       -- ^ behaves like a normal Cryptol @import@
  103   | PublicAndPrivate -- ^ allows viewing of both @private@ sections
  104                      --   and (arbitrarily nested) submodules.
  105   deriving (Eq, Show)

  108 -- | The global environment for capturing the Cryptol state, both
  109 --   Cryptol's own state and the state associated with
  110 --   importing/translating into SAWCore. This is intended to be a
  111 --   write-once record of any work done during translation. Rather than
  112 --   using the fields of this datatype directly, it is accessed as
  113 --   'SharedContext' data via the setters and getters below.
  114 data GlobalCryptolEnv = GlobalCryptolEnv
  115   { geModuleEnv   :: ME.ModuleEnv
  116   -- | Invariant: This is a subset of 'geAllVars', which is
  117   --   enforced by 'addExtraVars'

  119   , geExtraTySyns :: Map C.Name C.TySyn
  120   , geAllVars     :: Map C.Name C.Schema
  121   , geAllTerms    :: Map C.Name Term
  122   , geRefPrims    :: Map C.PrimIdent C.Expr
  123   , gePrims       :: Map C.PrimIdent Term
  124   , gePrimTypes   :: Map C.PrimIdent Term
  125   , geFFITypes    :: Map NameInfo C.FFI
  126   }

  138 -- | Restore a `GlobalCryptolEnv` from a checkpoint. The first argument
  139 --   @chkEnv@ is the `GlobalCryptolEnv` saved by / copied into the
  140 --   checkpoint; the second argument @newEnv@ is the current one
  141 --   we wish to overwrite by rolling back to the checkpoint.
  142 --   The 'ME.meNameSeeds' and 'ME.meSupply' from the
  143 --   module environment are not rolled back, to avoid re-using old
  144 --   names.
  145 --   NOTE: This effectively
  146 --   invalidates any translated 'Term's or Cryptol expressions created
  147 --   after the checkpoint. Attempting to use them in the restored
  148 --   environment will have unpredictable results, and likely will
  149 --   result in a panic. Similarly, 'CryptolEnv's captured after the
  150 --   checkpoint are no longer safe to use in the resulting environment.
  151 
  152 --   We also ought to invalidate terms constructed since the checkpoint
  153 --   was taken, like SAWCore does. See #2859.
  154 
  155 --   We could, for example, have 'CryptolScope' track which
  156 --   identifiers it references, and check that they are in a valid
  157 --   range with respect to the corresponding global environment
  158 --   before combining them into a 'CryptolEnv'.
  159   restoreMetadata chk now = return $
  160     let newMEnv = geModuleEnv now
  161         chkMEnv = geModuleEnv chk
  162     in chk { geModuleEnv = chkMEnv 
  163                { ME.meNameSeeds = ME.meNameSeeds newMEnv
  164                , ME.meSupply = ME.meSupply newMEnv 
  165                }
  166            }

  168 -- | A scope frame that captures which Cryptol names are accessible.
  169 --  `fNamingEnv` is the local naming environment, which can be extended
  170 --  ad-hoc with additional declarations. `fImports` is a list of all
  171 --  the modules that have been imported, and a visibility setting for
  172 --  each. This does not include, for example, builtin modules that
  173 --  exist but that have not been imported.
  174 
  175 data CryptolFrame =
  176   CryptolFrame { fNamingEnv :: MR.NamingEnv
  177                , fImports :: [(ImportVisibility, C.Import)] 
  178                }
  179 
  180 initFrame :: CryptolFrame
  181 initFrame = CryptolFrame mempty mempty
  182 
  183 -- | A nonempty list of 'CryptolFrame's, where the first element is the
  184 --   "bottom" frame that takes highest precedence when looking up
  185 --   names. Each individual frame only contains values declared at
  186 --   exactly that level. The full scope is computed by collecting
  187 --   everything in this stack, via 'eExtraNaming' and 'eImports'.
  188 newtype CryptolEnv = CryptolEnv (NonEmpty CryptolFrame)
  189 
  190 initEnv :: CryptolEnv
  191 initEnv = CryptolEnv (initFrame :| [])
  192 
  193 isToplevel :: CryptolEnv -> Bool
  194 isToplevel (CryptolEnv (_ :| frames)) = null frames

  196 -- | Test if the scopes have the same number of frames pushed.
  197 sameHeight :: CryptolEnv -> CryptolEnv -> Bool
  198 sameHeight (CryptolEnv scope1) (CryptolEnv scope2) = 
  199   NE.length scope1 == NE.length scope2
  200 
  201 mapCurFrame :: 
  202   (CryptolFrame -> CryptolFrame) -> 
  203   CryptolEnv -> 
  204   CryptolEnv
  205 mapCurFrame f (CryptolEnv (frame :| frames)) =
  206   CryptolEnv (f frame :| frames)
  207 
  208 -- | Push a fresh frame onto the stack.
  209 pushScope :: CryptolEnv -> CryptolEnv
  210 pushScope (CryptolEnv frames) = CryptolEnv (initFrame <| frames)
  211 
  212 -- | Pop the current frame from the stack, discarding its
  213 --   contents. Panics if this is the only frame.
  214 popScope :: CryptolEnv -> CryptolEnv
  215 popScope (CryptolEnv frames) = case snd (NE.uncons frames) of
  216   Nothing -> panic "popScope" [ "Popping topmost scope"]
  217   Just frames' -> CryptolEnv frames'
  218 
  219 -- | Map the naming environment of the frame currently in scope.
  220 mapNaming :: 
  221   (MR.NamingEnv -> MR.NamingEnv) -> 
  222   CryptolEnv -> 
  223   CryptolEnv
  224 mapNaming f = mapCurFrame $ 
  225       \fr -> fr {fNamingEnv = f (fNamingEnv fr) }
  226 
  227 -- | Map the module imports of the frame currently in scope.
  228 mapImports :: 
  229   ([(ImportVisibility, C.Import)] -> [(ImportVisibility, C.Import)] ) -> 
  230   CryptolEnv -> 
  231   CryptolEnv
  232 mapImports f = mapCurFrame $ 
  233       \fr -> fr {fImports = f (fImports fr) }

  235 -- | Run the inner action bracketed new frame pushed/popped on
  236 --   the 'CryptolScope' stack.
  237 --   Fails if the inner action changes the scope height
  238 --   (i.e. it does not properly bracket its pushes and pops).
  239 withFreshScope :: 
  240   MonadFail m =>
  241   CryptolEnv -> 
  242   (CryptolEnv -> 
  243   m (a, CryptolEnv)) -> 
  244   m (a, CryptolEnv)
  245 withFreshScope env0 f = do
  246   let env1 = pushScope env0
  247   (a, env2) <- f env1
  248   unless (sameHeight env1 env2) $
  249     panic "withFreshScope" ["mismatched push/pops"]
  250   let env3 = popScope env2
  251   return (a, env3)

  253 -- | Access the 'C.Supply' in the global 'ME.ModuleEnv' for generating
  254 --   fresh names. More efficient than 'runModuleM', as it avoids any other
  255 --   bookkeeping.
  256 useModEnvSupply :: SharedContext -> (C.Supply -> (a, C.Supply)) -> IO a
  257 useModEnvSupply sc f = do
  258   modEnv <- eModuleEnv sc
  259   let (a, supply) = f $ ME.meSupply modEnv
  260   mapModEnv sc (\modEnv_ -> modEnv_ { ME.meSupply = supply } )
  261   return a

  275 -- The "getters" below were historically fields in 'CryptolEnv', which
  276 -- are now defined functions that access either the 'GlobalCryptolEnv'
  277 -- or the 'CryptolScope' as appropriate. In contrast, updates were
  278 -- historically made by directly accessing 'CryptolEnv' fields, and are
  279 -- now restricted to only adding new entries to the maps in
  280 -- 'GlobalCryptolEnv' (e.g. 'addExtraVars'). This enforces the policy
  281 -- that the global environment is intended to be a complete record of
  282 -- all work done during translation/import, regardless of scope. Since
  283 -- the maps in the global environment are all keyed on globally-unique
  284 -- values, these operations are expected to only add entries and never
  285 -- overwrite existing ones.
  286 --
  287 -- NOTE: We could enforce a write-once policy here, but it's
  288 -- not immediately obvious if we need to support key clashes
  289 -- for equal entries (not possible for
  290 -- some types like 'C.Expr'), or if it is even useful to do so.

  360 -- | Run an 'MM.ModuleM' action using the module environment from
  361 -- 'eModuleEnv'. If the action is successful, updates the module
  362 -- environment in the 'SharedContext' and returns the result of the
  363 -- action. Otherwise, the environment is left unchanged and a
  364 -- 'MM.ModuleError' is returned.
  365 runModuleM ::
  366  SharedContext -> MM.ModuleM a -> IO (Either MM.ModuleError a, [MM.ModuleWarning])
  367 runModuleM sc m = do
  368   fileReader <- scFileReader sc
  369   env <- eModuleEnv sc
  370   let minp solver = MM.ModuleInput {
  371           MM.minpCallStacks = True,
  372           MM.minpSaveRenamed = False,
  373           MM.minpEvalOpts = pure defaultEvalOpts,
  374           MM.minpByteReader = fileReader,
  375           MM.minpModuleEnv = env,
  376           MM.minpTCSolver = solver
  377       }
  378   (res, ws) <- SMT.withSolver (return ()) (meSolverConfig env) $ \solver ->
  379     MM.runModuleM (minp solver) m
  380   case res of
  381     Right (a, env') -> do
  382       mapGlobal sc $ \genv ->
  383         refreshCryptolEnv $ genv { geModuleEnv = env' }
  384       return (Right a, ws)
  385     Left err -> return (Left err, ws)

  412 -- | Map from Cryptol names to Cryptol types. This is
  413 -- used to call `fastTypeOf` and `fastSchemaOf` on Cryptol expressions
  414 -- to fetch their types. This table is implicitly updated after every
  415 -- 'runModuleM' action to ensure it is always up-to-date.
  416 --
  417 -- FUTURE: in principle we should be able to use the SAWCore types of
  418 -- the SAWCore terms after importing them, instead of `fastTypeOf` and
  419 -- `fastSchemaOf`, and drop the `eAllVars` table. In practice, doing
  420 -- that relies (in some cases) on being able to call `scCryptolType`
  421 -- to reconstruct the Cryptol-level type; that in turn requires, when
  422 -- inside a forall-binding, logic to intercept and lift SAWCore type
  423 -- variables back to their Cryptol parents. That requires a table we
  424 -- don't currently have, as well as additional lookup logic that
  425 -- doesn't currently exist. Furthermore, while we've fixed many of the
  426 -- ways the Cryptol -> SAWCore type mapping is noninjective, it still
  427 -- won't work for enumerations. And beyond that, when handling
  428 -- polymorphic type schemes we erase certain typeclasses in the
  429 -- translation, and that loses info, so we might need to translate
  430 -- those classes to placeholders instead of erasing them. It may then
  431 -- also be that the one use of `fastSchemaOf` can't actually be
  432 -- avoided; that isn't super clear.

  472 --

  474 -- | The "extra" naming environment that captures Cryptol names
  475 --   which don't correspond to any imported module. Generally these
  476 --   result from names created in SAW which have been reflected into
  477 --   the Cryptol environment.
  478 --   This is scoped content, where the accessible names are expected
  479 --   to be managed by SAW and correspond to the same SAW values that
  480 --   are in scope.
  481 --
  482 -- FUTURE: Cryptol has its own functionality for additional bindings
  483 -- (it uses it for things created from the Cryptol REPL) and we ought
  484 -- to be able to use it instead of bolting on our own additional layer
  485 -- of material. Doing so would avoid various inconsistencies and
  486 -- irregularities that can creep in when we reimplement Cryptol name
  487 -- resolution.
  488 eExtraNaming :: CryptolEnv -> MR.NamingEnv
  489 eExtraNaming (CryptolEnv (frame :| frames)) = 
  490   foldr (\fr ne -> ne `MR.shadowing` (fNamingEnv fr)) (fNamingEnv frame) frames

  492 -- | The list of Cryptol modules which have been brought into the
  493 --   current scope. This essentially acts as a filter on the full set
  494 --   of loaded modules in the global module environment ('eModuleEnv'),
  495 --   where the contents of the selected modules are brought into scope
  496 --   according to the associated 'ImportVisibility'. The modules here
  497 --   should only correspond to modules that are present in the module
  498 --   environment *and* have been translated into SAWCore.
  499 eImports :: CryptolEnv -> [(ImportVisibility, C.Import)]
  500 eImports (CryptolEnv frames) = 
  501   concat $ map fImports $ NE.toList frames

  504 -- | Refresh 'geAllVars' after updating the module environment.
  505 --   Previously (before 'GlobalCryptolEnv'), this would overwrite the
  506 --   'eAllVars' field. Now this will add new vars (i.e. from
  507 --   newly-added modules) but keep existing ones, as the
  508 --   'GlobalCryptolEnv' should never drop entries. If the module
  509 --   environment has been managed properly, we expect any clashing keys
  510 --   to have equal elements, but this is not checked/enforced. Not
  511 --   exported.
  512 refreshCryptolEnv :: GlobalCryptolEnv -> GlobalCryptolEnv
  513 refreshCryptolEnv genv =
  514      let modEnv = geModuleEnv genv
  515          ifaceDecls = getAllIfaceDecls modEnv
  516          newtypeCons = Map.fromList
  517                          [ con
  518                          | nt <- Map.elems (MI.ifNominalTypes ifaceDecls)
  519                          , con <- C.nominalTypeConTypes nt
  520                          ]
  521          vars = Map.map MI.ifDeclSig $ MI.ifDecls ifaceDecls
  522          -- note that we don't need to re-add geExtraVars, because
  523          -- it is already included in the existing geAllVars
  524          allvars = newtypeCons `Map.union` vars
  525          allvars' = Map.union allvars (geAllVars genv)
  526      in genv { geAllVars = allvars' }
