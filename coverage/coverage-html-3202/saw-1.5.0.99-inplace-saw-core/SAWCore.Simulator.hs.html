  105 -- Let-based terms
  106 
  107 -- | A representation of SAWCore terms with explicit @let@ bindings
  108 -- instead of using a 'TermIndex' on every subterm.
  109 -- This representation is designed to support efficient evaluation.
  110 -- Type 'TermIndex' is used to represent let-bound variable names.
  111 --
  112 -- Invariant: In any 'LLet', all let-variables occurring on the rhs of
  113 -- any binding must be strictly less than the index of the lhs; this
  114 -- ensures the term is well-founded.
  115 data LTerm
  116   = LFlat !(FlatTermF LTerm)
  117   | LApp !LTerm !LTerm
  118   | LLam !TermIndex !LTerm
  119     -- ^ The type is omitted because it is not needed for evaluation.
  120   | LLet !(IntMap LTerm) !LTerm
  121     -- ^ A non-recursive let, where smaller indices are in scope for
  122     -- entries of larger indices, but not vice versa.
  123   | LFun !LTerm !LTerm -- ^ Non-dependent function type.
  124   | LPi !VarIndex !LTerm !LTerm -- ^ Dependent function type.
  125   | LConst !Name
  126   | LLetVar !TermIndex -- ^ A variable bound in a 'LLet' term.
  127   | LBoundVar !VarIndex -- ^ A variable bound in a 'LLam' or 'LPi' term.
  128   deriving Show
  129 
  130 -- | The number of occurrences of a subterm.
  131 data Multiplicity = Single | Multiple
  132   deriving Eq
  133 
  134 -- | An 'LTerm' paired with the 'varTypes' field of the 'Term' it was
  135 -- derived from and a multiplicity.
  136 data LBinding = LBinding !LTerm !(IntMap Term) !Multiplicity
  137 
  138 usesVarName :: VarName -> LBinding -> Bool
  139 usesVarName vn (LBinding _ env _) = IntMap.member (vnIndex vn) env
  140 
  141 -- | Apply a let-variable substitution to an 'LTerm'.
  142 -- Only use entries marked as a single occurrence.
  143 -- The substitution is applied recursively, so that entries defined in
  144 -- terms of other entries will be completely unfolded.
  145 --
  146 -- Precondition: A binding at key index @i@ in the substitution may
  147 -- only contain 'LLetVar's with indices strictly less than @i@; this
  148 -- ensures that the substitution is well-founded.
  149 --
  150 -- Precondition: No entries in the substitution may share keys with
  151 -- any let-bindings in the term; this ensures that variable capture is
  152 -- impossible when substituting under 'LLet'.
  153 substLTerm :: IntMap LBinding -> LTerm -> LTerm
  154 substLTerm s = go
  155   where
  156     go :: LTerm -> LTerm
  157     go t =
  158       case t of
  159         LFlat ftf ->
  160           LFlat (fmap go ftf)
  161         LApp t1 t2 ->
  162           LApp (go t1) (go t2)
  163         LLam x t1 ->
  164           LLam x (go t1)
  165         LLet binds body ->
  166           LLet (fmap go binds) (go body)
  167         LFun t1 t2 ->
  168           LFun (go t1) (go t2)
  169         LPi x t1 t2 ->
  170           LPi x (go t1) (go t2)
  171         LConst {} ->
  172           t
  173         LLetVar i ->
  174           case IntMap.lookup i s of
  175             Nothing -> t
  176             Just (LBinding t' _ n) ->
  177               if n == Single then go t' else t
  178         LBoundVar {} ->
  179           t
  180 
  181 -- | Make a let expression after inlining all local bindings that are
  182 -- used only once.
  183 mkLLet :: IntMap LBinding -> LTerm -> LTerm
  184 mkLLet s body =
  185   if IntMap.null s2 then body' else LLet s2 body'
  186   where
  187     body' = substLTerm s body
  188     s1 = IntMap.filter (\(LBinding _ _ n) -> n == Multiple) s
  189     s2 = fmap (\(LBinding t _ _) -> substLTerm s t) s1
  190 
  191 -- | Translate a 'Term' to an 'LTerm'.
  192 -- Subterms that occur more than once are lifted out to the largest
  193 -- possible scope and collected into let expressions ('LLet').
  194 toLTerm :: Term -> LTerm
  195 toLTerm t0 =
  196   let (t', binds) = State.runState (go t0) mempty
  197   in mkLLet binds t'
  198   where
  199     -- | Translate a term @t@ into a let-variable.
  200     -- The 'IntMap' in the state monad is indexed by 'VarIndex', and
  201     -- records the 'LTerm' translations of all subterms of @t@ and
  202     -- other terms we've already seen in the current scope.
  203     go :: Term -> State.State (IntMap LBinding) LTerm
  204     go t =
  205       do binds <- State.get
  206          let i = termIndex t
  207          case IntMap.lookup i binds of
  208            Just (LBinding lt env _) ->
  209              -- Subterm has already been seen, so mark it as a multiple occurrence.
  210              do State.modify (IntMap.insert i (LBinding lt env Multiple))
  211                 pure (LLetVar i)
  212            Nothing ->
  213              -- New subterm: Traverse it and then record a single occurrence.
  214              do t' <- termf (unwrapTermF t)
  215                 State.modify (IntMap.insert i (LBinding t' (varTypes t) Single))
  216                 pure (LLetVar i)
  217 
  218     -- | Translate a 'TermF' into an 'LTerm'.
  219     -- If the input 'TermF' is a lambda or pi binder, then put all the
  220     -- bindings from the state monad that mention the bound variable
  221     -- into a 'LLet'.
  222     -- Bindings not mentioning the bound variable stay in the map and
  223     -- are thus lifted out of the scope of the binder.
  224     termf :: TermF Term -> State.State (IntMap LBinding) LTerm
  225     termf tf =
  226       case tf of
  227         FTermF ftf ->
  228           do ftf' <- traverse go ftf
  229              pure (LFlat ftf')
  230         App t1 t2 ->
  231           do t1' <- go t1
  232              t2' <- go t2
  233              pure (LApp t1' t2')
  234         Lambda x _ty body ->
  235           localScope x $
  236           do body' <- go body
  237              locals <- getLocalBindings x
  238              pure (LLam (vnIndex x) (mkLLet locals body'))
  239         Pi x t1 t2
  240           | IntMap.member (vnIndex x) (varTypes t2) ->
  241               do t1' <- go t1
  242                  localScope x $
  243                    do t2' <- go t2
  244                       locals <- getLocalBindings x
  245                       pure (LPi (vnIndex x) t1' (mkLLet locals t2'))
  246           | otherwise ->
  247               do t1' <- go t1
  248                  t2' <- go t2
  249                  pure (LFun t1' t2')
  250         Constant nm ->
  251           pure (LConst nm)
  252         Variable x _ty ->
  253           pure (LBoundVar (vnIndex x))
  254         Label _text t1 ->
  255           go t1
  256 
  257     -- | Temporarily remove bindings that mention x, run the inner
  258     -- computation where those bindings would be shadowed, and then
  259     -- put them back.
  260     localScope :: VarName -> State.State (IntMap LBinding) a -> State.State (IntMap LBinding) a
  261     localScope x action =
  262       do binds <- State.get
  263          let (shadowed, binds1) = IntMap.partition (usesVarName x) binds
  264          State.put binds1
  265          result <- action
  266          binds2 <- State.get
  267          State.put (shadowed <> binds2)
  268          pure result
  269 
  270     -- | Filter out all bindings mentioning x and return them.
  271     getLocalBindings :: VarName -> State.State (IntMap LBinding) (IntMap LBinding)
  272     getLocalBindings x =
  273       do binds <- State.get
  274          let (locals, binds') = IntMap.partition (usesVarName x) binds
  275          State.put binds'
  276          pure locals

  299 -- | Generic evaluator for 'LTerm's.
  300 evalLTerm ::
  301   forall l. (VMonadLazy l, MonadFix (EvalM l), Show (Extra l)) =>
  302   SimulatorConfig l ->
  303   IntMap (Thunk l) {- ^ Constant environment, indexed by 'VarIndex' -} ->
  304   IntMap (Thunk l) {- ^ Bound variable environment, indexed by 'VarIndex' -} ->
  305   IntMap (Thunk l) {- ^ Let-variable environment, indexed by 'TermIndex' -} ->
  306   LTerm -> MValue l
  307 evalLTerm cfg consts vars lets t0 =
  308   case t0 of
  309     LFlat ftf ->
  310       case ftf of
  311         Recursor r ->
  312           case simRecursor cfg (recursorDataType r) (recursorSort r) of
  313             Just v -> v
  314             Nothing ->
  315               case lookupVarIndexInMap (nameIndex (recursorDataType r)) (simModMap cfg) of
  316                 Just (ResolvedDataType dt) ->
  317                   do let nparams = recursorNumParams r
  318                      let nixs = recursorNumIxs r
  319                      let cnames = recursorCtorOrder r
  320                      vFunList nparams $ \_ps_thunks ->
  321                        pure $ VFun $ \_motive ->
  322                        vFunList (length cnames) $ \elim_thunks ->
  323                        do let vrec = VRecursor dt elim_thunks
  324                           vFunList nixs (\_ixs -> pure (evalRecursor vrec))
  325                 _ ->
  326                   panic "evalTermF"
  327                   [ "Data type not found for recursor: " <>
  328                     toAbsoluteName (nameInfo (recursorDataType r)) ]
  329         Sort s _h ->
  330           pure $ TValue (VSort s)
  331         ArrayValue _ tv ->
  332           VVector <$> traverse (delay . recEval) tv
  333         StringLit s ->
  334           pure $ VString s
  335 
  336     LApp t1 t2 ->
  337       do v1 <- recEval t1
  338          case v1 of
  339            VFun f ->
  340              do x <- delay (recEval t2)
  341                 f x
  342            _ -> panic "evalLTerm" ["Expected VFun"]
  343 
  344     LLam i t1 ->
  345       pure $ VFun (\x -> loop (IntMap.insert i x vars) lets t1)
  346 
  347     LLet binds body ->
  348       do lets' <-
  349            mfix $ \lets' ->
  350            do vs <- traverse (delay . loop vars lets') binds
  351               pure (lets <> vs)
  352          loop vars lets' body
  353 
  354     LFun t1 t2 ->
  355       do v1 <- toTValue <$> recEval t1
  356          v2 <- toTValue <$> recEval t2
  357          pure $ TValue $ VPiType v1 $ VNondependentPi v2
  358 
  359     LPi i t1 t2 ->
  360       do v1 <- toTValue <$> recEval t1
  361          pure $ TValue $ VPiType v1 $
  362            VDependentPi (\x -> toTValue <$> loop (IntMap.insert i x vars) lets t2)
  363 
  364     LConst nm ->
  365       case IntMap.lookup (nameIndex nm) consts of
  366         Just x -> force x
  367         Nothing -> panic "evalLTerm" ["Constant name not found", Text.pack (show nm)]
  368 
  369     LLetVar i ->
  370       case IntMap.lookup i lets of
  371         Just x -> force x
  372         Nothing ->
  373           panic "evalLTerm"
  374           ["Let variable not found", Text.pack (show i), Text.pack (show (IntMap.keys lets))]
  375 
  376     LBoundVar i ->
  377       case IntMap.lookup i vars of
  378         Just x -> force x
  379         Nothing ->
  380           panic "evalLTerm"
  381           ["Lambda/pi variable not found", Text.pack (show i), Text.pack (show (IntMap.keys vars))]
  382   where
  383     loop :: IntMap (Thunk l) -> IntMap (Thunk l) -> LTerm -> EvalM l (Value l)
  384     loop vars' lets' = evalLTerm cfg consts vars' lets'
  385 
  386     recEval :: LTerm -> EvalM l (Value l)
  387     recEval = loop vars lets
  388 
  389     toTValue :: HasCallStack => Value l -> TValue l
  390     toTValue (TValue x) = x
  391     toTValue t = panic "evalTermF / toTValue" ["Not a type value: " <> Text.pack (show t)]
  392 
  393     evalRecursor :: VRecursor l -> Value l
  394     evalRecursor vrec@(VRecursor dt elims) =
  395       vStrictFun $ \argv ->
  396       case argv of
  397         VCtorApp n _dep args
  398           | n < length elims ->
  399               do elimv <- force (elims !! n)
  400                  let ctor = dtCtors dt !! n
  401                  reduceRecursor (evalRecursor vrec) elimv args (ctorArgStruct ctor)
  402           | otherwise ->
  403               panic "evalTermF / evalRecursor"
  404               ["No eliminator for constructor: " <> Text.pack (show n)]
  405         VCtorMux branches ->
  406           do alts <- traverse (evalCtorMuxBranch vrec) (IntMap.assocs branches)
  407              combineAlts alts
  408         VBVToNat{} ->
  409           panic "evalTermF / evalRecursor"
  410           ["Unsupported symbolic recursor argument of type Nat"]
  411         _ ->
  412           panic "evalTermF / evalRecursor"
  413           ["Expected constructor for datatype: " <> toAbsoluteName (nameInfo (dtName dt))]
  414 
  415     evalCtorMuxBranch ::
  416       VRecursor l ->
  417       (Int, (VBool l, Muxability, [Thunk l])) ->
  418       EvalM l (VBool l, EvalM l (Value l))
  419     evalCtorMuxBranch r@(VRecursor dt elims) (i, (p, _m, args)) =
  420       do elimv <- force (elims !! i)
  421          let ctor = dtCtors dt !! i
  422          pure (p, reduceRecursor (evalRecursor r) elimv args (ctorArgStruct ctor))
  423 
  424     combineAlts :: [(VBool l, EvalM l (Value l))] -> EvalM l (Value l)
  425     combineAlts [] = panic "evalTermF / combineAlts" ["no alternatives"]
  426     combineAlts [(_, x)] = x
  427     combineAlts ((p, x) : alts) = simLazyMux cfg p x (combineAlts alts)
  428 
  429 -- | Compute whether the 'Ctor' has a type that allows argument-wise
  430 -- muxing.
  431 ctorMuxability :: Ctor -> Muxability
  432 ctorMuxability ctor =
  433   if nondep IntSet.empty (ctorArgs cas) then Muxable else NonMuxable
  434   where
  435     cas :: CtorArgStruct
  436     cas = ctorArgStruct ctor
  437     nondep :: IntSet -> [(VarName, CtorArg)] -> Bool
  438     nondep vs [] = all (IntSet.disjoint vs . freeVars) (ctorIndices cas)
  439     nondep vs ((vn, arg) : args) =
  440       IntSet.disjoint vs (freesCtorArg arg) &&
  441       nondep (IntSet.insert (vnIndex vn) vs) args
  442 
  443 -- | Compute the set of 'VarIndex'es of free variables in a 'CtorArg'.
  444 freesCtorArg :: CtorArg -> IntSet
  445 freesCtorArg (ConstArg t) = freeVars t
  446 freesCtorArg (RecursiveArg zs is) = go zs
  447   where
  448     go :: [(VarName, Term)] -> IntSet
  449     go [] = IntSet.unions (map freeVars is)
  450     go ((v, t) : ts) = freeVars t <> IntSet.delete (vnIndex v) (go ts)
  451 
  452 -- | Evaluate a recursor applied to a specific data constructor.
  453 reduceRecursor ::
  454   forall l. (VMonadLazy l, Show (Extra l)) =>
  455   Value l {- ^ recursor function expecting datatype argument -} ->
  456   Value l {- ^ constructor eliminator function -} ->
  457   [Thunk l] {- ^ constructor arguments -} ->
  458   CtorArgStruct {- ^ constructor formal argument descriptor -} ->
  459   MValue l
  460 reduceRecursor r elim c_args argstruct = go elim c_args (map snd (ctorArgs argstruct))
  461   where
  462     go :: Value l -> [Thunk l] -> [CtorArg] -> MValue l
  463     go e [] [] = pure e
  464     go e (x : xs) (arg : args) =
  465       case arg of
  466         ConstArg _ ->
  467           do e' <- apply e x
  468              go e' xs args
  469         RecursiveArg zs _ixs ->
  470           do e1 <- apply e x
  471              recx <- delay (mk_rec_arg (length zs) x)
  472              e2 <- apply e1 recx
  473              go e2 xs args
  474     go _ _ _ = panic "reduceRecursor" ["Wrong number of constructor arguments"]
  475 
  476     -- For a recursive argument, we need a value of the form
  477     -- > \z1 .. zk -> r (x z1 .. zk)
  478     mk_rec_arg :: Int -> Thunk l -> MValue l
  479     mk_rec_arg k x =
  480       vFunList k $ \zs ->
  481       do x_zs <- delay (force x >>= \v -> applyAll v zs)
  482          apply r x_zs

  544 -- | A variant of 'evalGlobal' that lets the uninterpreted function
  545 -- symbol and external-constant callbacks have access to the 'TermF'.
  546 evalGlobal' ::
  547   forall l. (VMonadLazy l, Show (Extra l)) =>
  548   ModuleMap ->
  549   -- | Implementations of 'Primitive' terms, plus overrides for 'Constant' and 'CtorApp' terms
  550   Map Ident (Prims.Prim l) ->
  551   -- | Implementations of free 'Variable' terms
  552   (Term -> VarName -> TValue l -> MValue l) ->
  553   -- | Overrides for Constant terms (e.g. uninterpreted functions)
  554   (Name -> TValue l -> Maybe (MValue l)) ->
  555   -- | Overrides for Recursor terms
  556   (Name -> Sort -> Maybe (Prims.Prim l)) ->
  557   -- | Handler for stuck primitives
  558   (Name -> Text -> [Thunk l] -> MValue l) ->
  559   -- | Lazy mux operation
  560   (VBool l -> MValue l -> MValue l -> MValue l) ->
  561   EvalM l (SimulatorConfig l)
  562 evalGlobal' modmap prims variable constant recursor primHandler lazymux =
  563   do checkPrimitives modmap prims
  564      return (SimulatorConfig primitive variable constant' recursor' modmap lazymux)
  565   where
  566     constant' :: Name -> TValue l -> Maybe (MValue l)
  567     constant' nm tv =
  568       case constant nm tv of
  569         Just v -> Just v
  570         Nothing ->
  571           case nameInfo nm of
  572             ModuleIdentifier ident ->
  573               evalPrim (primHandler nm) <$> Map.lookup ident prims
  574             ImportedName{} -> Nothing
  575 
  576     primitive :: Name -> MValue l
  577     primitive nm =
  578       case nameInfo nm of
  579         ImportedName {} ->
  580           panic "evalGlobal'" ["Unimplemented global: " <> toAbsoluteName (nameInfo nm)]
  581         ModuleIdentifier ident ->
  582           case Map.lookup ident prims of
  583             Just v  -> evalPrim (primHandler nm) v
  584             Nothing -> panic "evalGlobal'" ["Unimplemented global: " <> identText ident]
  585 
  586     recursor' :: Name -> Sort -> Maybe (MValue l)
  587     recursor' nm s = evalPrim (primHandler nm) <$> recursor nm s

  622 ----------------------------------------------------------------------
  623 -- The evaluation strategy for shared terms involves a preprocessing
  624 -- phase, where each term is translated to a special term type with
  625 -- explicit let bindings.
  626 -- The let-terms are then evaluated recursively using a set of three
  627 -- environment parameters:
  628 --
  629 -- * The constant environment contains a thunk for each constant used
  630 -- in the term, or in the definition of another used constant.
  631 --
  632 -- * The bound variable environment has a thunk for each lambda- or
  633 -- pi-bound variable in scope; it is pre-populated with translations
  634 -- of variables free in the top-level term.
  635 --
  636 -- * The let-variable environment has a thunk for each let-binding in
  637 -- scope.
  638 
  639 {-# SPECIALIZE evalSharedTerm ::
  640   Show (Extra l) => SimulatorConfigIn Id l -> Term -> MValueIn Id l #-}
  641 {-# SPECIALIZE evalSharedTerm ::
  642   Show (Extra l) => SimulatorConfigIn IO l -> Term -> MValueIn IO l #-}
  643 
  644 -- | Evaluator for shared terms.
  645 evalSharedTerm ::
  646   forall l. (VMonadLazy l, MonadFix (EvalM l), Show (Extra l)) =>
  647   SimulatorConfig l -> Term -> MValue l
  648 evalSharedTerm cfg t =
  649   do let names = collectConstants cfg t
  650      constThunks <-
  651        mfix $ \constThunks ->
  652        traverse (delay . evalConst constThunks) names
  653      let freevars = collectVariables t
  654      varThunks <-
  655        mfix $ \varThunks ->
  656        traverse (delay . evalVar constThunks varThunks) freevars
  657      evalLTerm cfg constThunks varThunks mempty (toLTerm t)
  658   where
  659     evalConst :: IntMap (Thunk l) -> Name -> MValue l
  660     evalConst consts nm =
  661       do let r = requireNameInMap nm (simModMap cfg)
  662          ty' <- toTValue <$> evalLTerm cfg consts mempty mempty (toLTerm (resolvedNameType r))
  663          case simConstant cfg nm ty' of
  664            Just override -> override
  665            Nothing ->
  666              case r of
  667                ResolvedCtor ctor ->
  668                  ctorValue (ctorNumber ctor) (ctorMuxability ctor) (ctorNumParams ctor) (ctorNumArgs ctor)
  669                ResolvedDataType dt ->
  670                  dtValue (nameInfo nm) (dtNumParams dt) (dtNumIndices dt)
  671                ResolvedDef d ->
  672                  case defBody d of
  673                    Just body -> evalLTerm cfg consts mempty mempty (toLTerm body)
  674                    Nothing -> simPrimitive cfg nm
  675 
  676     evalVar :: IntMap (Thunk l) -> IntMap (Thunk l) -> (VarName, Term) -> MValue l
  677     evalVar consts env (nm, tp) =
  678       do tv <- toTValue <$> evalLTerm cfg consts env mempty (toLTerm tp)
  679          simVariable cfg tp nm tv
  680 
  681     toTValue :: HasCallStack => Value l -> TValue l
  682     toTValue (TValue x) = x
  683     toTValue v = panic "evalTermF / toTValue" ["Not a type value: " <> Text.pack (show v)]
  684 
  685     ctorValue :: Int -> Muxability -> Int -> Int -> MValue l
  686     ctorValue k m i j =
  687       vFunList i $ \_params ->
  688       vFunList j $ \args ->
  689       pure $ VCtorApp k m args
  690 
  691     dtValue :: NameInfo -> Int -> Int -> MValue l
  692     dtValue nm i j =
  693       vStrictFunList i $ \params ->
  694       vStrictFunList j $ \idxs ->
  695       pure $ TValue $ VDataType nm params idxs
  696 
  697 -- | Precompute the set of constant names (indexed by 'VarIndex')
  698 -- required for evaluation of a 'Term'.
  699 collectConstants :: SimulatorConfig l -> Term -> IntMap Name
  700 collectConstants cfg t0 = snd $ go (IntSet.empty, IntMap.empty) t0
  701   where
  702     go :: (IntSet, IntMap Name) -> Term -> (IntSet, IntMap Name)
  703     go acc@(idxs, names) t
  704       | IntSet.member (termIndex t) idxs = acc
  705       | otherwise = termf (IntSet.insert (termIndex t) idxs, names) (unwrapTermF t)
  706 
  707     termf :: (IntSet, IntMap Name) -> TermF Term -> (IntSet, IntMap Name)
  708     termf acc@(idxs, names) tf =
  709       case tf of
  710         Constant nm ->
  711           case r of
  712             -- if tf is a defined constant, traverse the definition body and type
  713             ResolvedDef (defBody -> Just body) -> go (go acc' (resolvedNameType r)) body
  714             -- otherwise just traverse the type
  715             _ -> go acc' (resolvedNameType r)
  716           where
  717             acc' = (idxs, IntMap.insert (nameIndex nm) nm names)
  718             r = requireNameInMap nm (simModMap cfg)
  719         Lambda _x _ty body ->
  720           go acc body -- skip type, which is not used for simulation
  721         _ ->
  722           foldl' go acc tf
  723 
  724 -- | Precompute the set of variables (indexed by 'VarIndex') occurring
  725 -- free in a 'Term'.
  726 collectVariables :: Term -> IntMap (VarName, Term)
  727 collectVariables t0 =
  728   IntMap.fromList [ (vnIndex vn, (vn, t)) | (vn, t) <- Map.assocs (getAllVarsMap t0) ]

  731 {-# SPECIALIZE evalPrim ::
  732   Show (Extra l) =>
  733   (Text -> [ThunkIn Id l] -> MValueIn Id l) ->
  734   PrimIn Id l ->
  735   MValueIn Id l
  736  #-}
  737 {-# SPECIALIZE evalPrim ::
  738   Show (Extra l) =>
  739   (Text -> [ThunkIn IO l] -> MValueIn IO l) ->
  740   PrimIn IO l ->
  741   MValueIn IO l
  742  #-}
  743 evalPrim :: forall l. (VMonadLazy l, Show (Extra l)) =>
  744   (Text -> [Thunk l] -> MValue l) ->
  745   Prims.Prim l ->
  746   MValue l
  747 evalPrim fallback = loop []
  748   where
  749     loop :: [Thunk l] -> Prims.Prim l -> MValue l
  750     loop env (Prims.PrimFun f) =
  751       pure $ VFun $ \x ->
  752         loop (x : env) (f x)
  753 
  754     loop env (Prims.PrimStrict f) =
  755       pure $ vStrictFun $ \x ->
  756         loop (ready x : env) (f x)
  757 
  758     loop env (Prims.PrimFilterFun msg r f) =
  759       pure $ vStrictFun $ \x ->
  760         runMaybeT (r x) >>= \case
  761           Just v -> loop (ready x : env) (f v)
  762           _ -> fallback msg (ready x : env)
  763 
  764     loop env (Prims.PrimExcept m) =
  765       runExceptT m >>= \case
  766         Right v  -> pure v
  767         Left msg -> fallback msg env
  768 
  769     loop _env (Prims.Prim m) = m
  770     loop _env (Prims.PrimValue v) = pure v

  773 defaultPrimHandler ::
  774   (VMonadLazy l, MonadFail (EvalM l)) =>

  779   , "On argument " ++ show (length env)
