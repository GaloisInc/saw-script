   32 -- | An introduction rule of the form @forall ctxt. hyps -> concl@,
   33 -- along with a proof term inhabiting that type.
   34 --
   35 -- Invariant: The 'proof' field inhabits the same type returned by
   36 -- @scPiList sc ctxt =<< scFunAll sc hyps concl@.
   37 -- Invariant: All variables in 'ctxt' should occur free in 'concl'.
   38 data IntroRule =
   39   IntroRule
   40   { ctxt :: [(VarName, Term)]
   41   , hyps :: [Term]
   42   , concl :: Term
   43   , proof :: Term
   44   }
   45   deriving Eq
   46 
   47 mkIntroRule :: SharedContext -> Term -> IO IntroRule
   48 mkIntroRule sc t0 =
   49   do ty <- scTypeOf sc t0
   50      let (vs, hs, c) = go ty
   51      pure (IntroRule vs hs c t0)
   52   where
   53     go :: Term -> ([(VarName, Term)], [Term], Term)
   54     go t =
   55       case asPi t of
   56         Just (x, ty, body)
   57           | IntMap.member (vnIndex x) (varTypes body) ->
   58             let (vs, hs, c) = go body in ((x, ty) : vs, hs, c)
   59           | otherwise ->
   60             let (hs, c) = asFunAll body in ([], ty : hs, c)
   61         Nothing ->
   62           ([], [], t)
   63 
   64 -- | A set of 'IntroRule's, indexed by conclusion in a a
   65 -- 'TermNet.Net'.
   66 newtype IntroRuleSet = IntroRuleSet (TermNet.Net IntroRule)
   67 
   68 emptyIntroRuleSet :: IntroRuleSet
   69 emptyIntroRuleSet = IntroRuleSet TermNet.empty
   70 
   71 insertIntroRuleSet :: IntroRule -> IntroRuleSet -> IntroRuleSet
   72 insertIntroRuleSet r (IntroRuleSet net) =
   73   IntroRuleSet (TermNet.insert_term (termPat (concl r), r) net)
   74 
   75 -- | Attempt to construct a term inhabiting a given type using a set
   76 -- of introduction rules.
   77 -- A failing 'Left' result includes the type of a subgoal with no
   78 -- matching rule; a successful 'Right' result includes a proof term.
   79 proveWithIntros :: SharedContext -> IntroRuleSet -> Term -> IO (Either Term Term)
   80 proveWithIntros sc (IntroRuleSet net) t0 = runExceptT (solve t0)
   81   where
   82     solve :: Term -> ExceptT Term IO Term
   83     solve t = try (TermNet.match_term net (termPat t)) t
   84 
   85     try :: [IntroRule] -> Term -> ExceptT Term IO Term
   86     try [] t = throwError t
   87     try (rule : rules) t =
   88       do result <- lift $ scMatch sc (ctxt rule) (concl rule) t
   89          case result of
   90            Nothing -> try rules t
   91            Just inst ->
   92              do subgoals <- lift $ traverse (scInstantiate sc inst) (hyps rule)
   93                 subproofs <- traverse solve subgoals
   94                 let params = mapMaybe (\(x, _) -> IntMap.lookup (vnIndex x) inst) (ctxt rule)
   95                 lift $ scApplyAll sc (proof rule) (params ++ subproofs)
