  117 initTTEnv :: SharedContext -> CryptolEnv -> IO TTEnv
  118 initTTEnv sc env = do
  119   modEnv <- CrySAW.eModuleEnv sc
  120   case C.lookupModule C.preludeName modEnv of
  121     Just prelude -> do
  122       allPrims <- CrySAW.ePrims sc
  123       allPrimTypes <- CrySAW.ePrimTypes sc
  124       allTerms <- CrySAW.eAllTerms sc
  125       envVarsRef <- newIORef IntMap.empty
  126       usedNamesRef <- newIORef Set.empty
  127       eCacheRef <- newIORef IntMap.empty
  128       tCacheRef <- newIORef IntMap.empty
  129       constTypesRef <- newIORef Map.empty
  130       let
  131         pmap = C.ifacePrimMap $ C.lmInterface prelude
  132         exprPrims = fmap (\x -> C.lookupPrimDecl x pmap) $ revMap asConstant allPrims
  133         typePrims = fmap (\x -> C.lookupPrimType x pmap) $ revMap asConstant allPrimTypes
  134         gvmap = IntMap.fromList $
  135           mapMaybe (\(nm,t) -> (\(vn,_) -> (SAW.vnIndex vn, nm)) <$> asVariable t)
  136           (Map.toList allTerms)
  137       nenv <- CrySAW.getNamingEnv sc env
  138       return $
  139         TTEnv
  140           { ttAllEnvVars = envVarsRef
  141           , ttUsedNames = usedNamesRef
  142           , ttConstMap = Map.unions [revMap asConstant allTerms, exprPrims, typePrims]
  143           , ttExtras = Map.fromList (extraPrims pmap)
  144           , ttCryEnv = env
  145           , ttSc = sc
  146           , ttGlobalNamingEnv = nenv
  147           , ttGlobalVarMap = gvmap
  148           , ttBoundExprs = IntMap.empty
  149           , ttExprCache = [eCacheRef]
  150           , ttTypeCache = [tCacheRef]
  151           , ttVarTypes = Map.empty
  152           , ttConstTypes = constTypesRef
  153           }
  154     Nothing -> fail "initTTEnv: missing Cryptol prelude"

  156 checkConvertible :: Term -> Term -> TT ()
  157 checkConvertible t1 t2 = do
  158   sc <- asks ttSc
  159   (liftIO $ scConvertible sc t1 t2) >>= \case
  160     True -> return ()
  161     False -> do
  162       let ppts = do
  163             t1' <- liftIO $ prettyTerm sc t1
  164             t2' <- liftIO $ prettyTerm sc t2
  165             return $ PP.vcat [t1', PP.indent 2 "vs.",t2']
  166       withContext (CallContext  "checkConvertible" ppts) $ 
  167         errMsg "Terms are not convertible"

  172 revTopProofs :: C.Expr -> C.Expr
  173 revTopProofs = go []
  174   where
  175     unwind prfs e = case prfs of
  176       x:xs -> C.EProofAbs x (unwind xs e)
  177       [] -> e
  178     go prfs = \case
  179       C.ETAbs tp e -> C.ETAbs tp (go prfs e)
  180       C.ELocated loc e -> C.ELocated loc (go prfs e)
  181       C.EProofAbs prf e -> go (prf:prfs) e
  182       e -> unwind prfs e

  184 -- | SAW sometimes reverses the constraints importing expressions, in which case we need to reverse the
  185 --   result after type inference in order to recover the original type/term
  186 revConstraints :: (Expr, C.Expr, C.Schema) -> (Expr, C.Expr, C.Schema)
  187 revConstraints (pe, e,s) = (pe, revTopProofs e, s { C.sProps = reverse (C.sProps s)})

  189 validateImport :: Term -> (Expr, C.Expr, C.Schema) -> TT (Expr, C.Expr, C.Schema)
  190 validateImport t (pe, e, s) = do
  191   sc <- asks ttSc
  192   s' <- liftIO $ CrySAW.importSchema sc s
  193   e' <- liftIO $ CrySAW.importExpr sc e
  194   tT <- liftIO $ scTypeOf sc t
  195   checkConvertible tT s'
  196   checkConvertible e' t
  197   return (pe,e,s)

  223 doInferExprSchema ::
  224   C.NamingEnv ->
  225   Map C.Name C.Schema ->
  226   Map C.Name C.TySyn ->
  227   [(Name, P.Kind)] ->
  228   P.Expr P.PName ->
  229   MM.ModuleM (C.Expr, C.Schema)
  230 doInferExprSchema nameEnv0 extraVars extraTySyns tvars pexpr = do
  231   tparams <- mapM mkTParam tvars
  232   let 
  233     nms = foldMap (\((pnm,_),(nm,_)) -> (C.singletonNS (C.nameNamespace nm) pnm nm)) 
  234         (zip tvars tparams)
  235     nameEnv = nms <> nameEnv0
  236     tps = map snd tparams
  237 
  238   npe <- MM.interactive (MB.noPat pexpr)
  239   let npe' = MR.rename npe
  240   re <- MM.interactive (MB.rename C.interactiveName nameEnv npe')
  241   let range = fromMaybe P.emptyRange (P.getLoc re)
  242   prims <- MB.getPrimMap
  243   ifDecls <- CrySAW.getAllIfaceDecls <$> MM.getModuleEnv
  244   tcEnv <- MB.genInferInput range prims C.NoParams ifDecls
  245   let tcEnv' = tcEnv  { C.inpVars = Map.union extraVars (C.inpVars tcEnv)
  246                       , C.inpTSyns = Map.union extraTySyns (C.inpTSyns tcEnv)
  247                       }
  248   (out :: C.InferOutput (C.Expr,C.Schema)) <- MM.io $ C.runInferM tcEnv' $ do
  249     let loc = Pos.emptyRange
  250     fresh <- C.liftSupply $ C.mkDeclared C.NSValue (C.TopModule C.exprModName) C.SystemName
  251       (C.packIdent "(expression)") Nothing loc
  252     (res, goals) <- C.withTParams tps $ C.collectGoals $ C.inferBinds True False
  253       [ simplePBind fresh re ]
  254     let goalProps = map C.goal goals
  255     case res of
  256       [d] | C.DExpr e <- C.dDefinition d -> 
  257         return (foldr C.ETAbs e tps, (C.dSignature d){ C.sVars = tps ++ C.sVars (C.dSignature d), C.sProps = goalProps ++ (C.sProps (C.dSignature d)) })
  258       _ -> Panic.panic "doInferExprSchema" ("Unexpected result: ": map pp res)
  259   CrySAW.runInferOutput out

  261 inferSchemaExpr :: Term -> TT (Expr, C.Expr, C.Schema)
  262 inferSchemaExpr t = do
  263   sc <- asks ttSc
  264   let (vars, t') = asLambdaList t
  265   ((tvars, pe),ttout) <- listen $ translateLambda vars t'
  266   extraVars <- liftIO $ CrySAW.eExtraVars sc
  267   extraTySyns <- liftIO $ CrySAW.eExtraTySyns sc
  268   (res,_) <- liftIO $ CrySAW.runModuleM sc $ do
  269     doInferExprSchema (ttNamingEnv ttout) extraVars extraTySyns tvars pe
  270   case res of
  271     Left e -> errMsg (Text.unpack $ pp e)
  272     Right (expr,schema) -> do
  273       let r = (pe,expr,schema)
  274       -- the order of the constraints is somewhat inconsistent, so we try
  275       -- either the original or reverse orderings and return the one that validates
  276       -- in most cases the reversed order seems to be preferred
  277       validateImport t (revConstraints r) <|> validateImport t r

  313 propToLambda :: Term -> TT Term
  314 propToLambda t = withTermContext "propToLambda" t $ go t
  315   where
  316     go :: Term -> TT Term
  317     go e = case asPi e of
  318       Just (vn, tp, body) -> do
  319         sc <- asks ttSc
  320         case asEqTrue tp of
  321           Just tp' -> do
  322             body' <- go body
  323             imp <- lookupPrim "==>"
  324             liftIO $ scApplyAll sc imp [tp',body']
  325           Nothing -> do
  326             body' <- go body
  327             liftIO $ scLambda sc vn tp body'
  328       Nothing -> case asEqTrue e of
  329         Just e' -> return e'
  330         Nothing -> errMsg "Unexpected term shape"

  420 lookupVar :: VarIndex -> TT CryptolVar
  421 lookupVar idx = do
  422   m <- deref ttAllEnvVars
  423   vts <- asks ttVarTypes
  424   case IntMap.lookup idx m of
  425     Just nm -> do
  426       tyk <- mreturn $ Map.lookup nm vts
  427       case tyk of
  428         Left{} -> return $ CryParam nm
  429         Right{} -> return $ CryTParam nm
  430     Nothing -> do
  431       g <- asks ttGlobalVarMap
  432       case IntMap.lookup idx g of
  433         Just nm -> do
  434           nm' <- uncheckName nm
  435           case isValueName nm of
  436             True -> return $ CryParam nm'
  437             False -> return $ CryTParam nm'
  438         Nothing -> errMsg $ "lookupVarName: could not find variable: " ++ show idx

  440 constToName :: SAW.Name -> TT C.Name
  441 constToName nm = do
  442   m <- asks ttConstMap
  443   mT <- asks ttExtras

  445     [ mreturn $ Map.lookup nm m
  446     , do SAW.ModuleIdentifier ident <- return $ SAW.nameInfo nm
  447          mreturn $ Map.lookup ident mT

  452 mkFreshName :: Text -> TT Name
  453 mkFreshName txt0 = go 0
  454   where
  455     -- clean up anonymous type parameter names
  456     txt1 = case Text.uncons txt0 of
  457       Just (c, rest) | Text.all isDigit rest -> Text.singleton c
  458       _ -> txt0
  459     go :: Integer -> TT Name
  460     go i = do
  461       let 
  462         txt2 = if i == 0 then txt1 else (txt1 <> Text.pack (show i))
  463         nm = P.UnQual' (C.mkIdent txt2) C.SystemName
  464         nm' = P.UnQual' (C.mkIdent txt2) C.UserName
  465       m <- deref ttUsedNames
  466       ne <- asks ttGlobalNamingEnv
  467       case Set.member nm m of
  468         False | 
  469            Nothing <- C.lookupNS C.NSValue nm ne
  470          , Nothing <- C.lookupNS C.NSValue nm' ne
  471          , Nothing <- C.lookupNS C.NSType nm ne
  472          , Nothing <- C.lookupNS C.NSType nm' ne
  473          -> do
  474           usedNamesRef <- asks ttUsedNames
  475           liftIO $ modifyIORef' usedNamesRef (Set.insert nm)
  476           return nm
  477         _ -> go (i+1)

  479 deref :: (TTEnv -> IORef a) -> TT a
  480 deref f = do
  481   ref <- asks f
  482   liftIO $ readIORef ref

  484 mkVarName :: SAW.VarName -> TT Name
  485 mkVarName vn = do
  486   envVarsRef <- asks ttAllEnvVars
  487   envVars <- liftIO $ readIORef envVarsRef
  488   case IntMap.lookup (SAW.vnIndex vn) envVars of
  489     Just nm -> return nm
  490     Nothing -> do
  491       nm <- mkFreshName (SAW.vnName vn)
  492       liftIO $ modifyIORef' envVarsRef $
  493         IntMap.insert (SAW.vnIndex vn) nm
  494       return nm

  520 instance MonadIO TT where
  521   liftIO f = do
  522     mres <- TT $ liftIO (try f)

  524       Left (e :: IOException) -> throwError $ TTError (show e) [] True

  527 -- From Control.Monad.Error base 2.3
  528 withError :: MonadError e m => (e -> e) -> m a -> m a
  529 withError f g = catchError g (throwError . f)

  531 tryError :: MonadError e m => m a -> m (Either e a)
  532 tryError f = (Right <$> f) `catchError` (pure . Left)
  533 
  534 tryMaybe :: MonadError e m => m a -> m (Maybe a)
  535 tryMaybe f = (Just <$> f) `catchError` (\_ -> pure Nothing)

  537 -- | Commit to an alternative by considering any uncaught errors thrown
  538 --   by the sub-computation to be unrecoverable.
  539 commit :: TT a -> TT a
  540 commit = withError (\e -> (e {ttErrCommitted = True}))

  542 -- | Ignore committed errors in the sub-computation
  543 uncommit :: TT a -> TT a
  544 uncommit = withError (\e -> (e {ttErrCommitted = False}))

  557 -- Alternative branches are implicit try-catches as long as the
  558 -- thrown error is not committed (i.e. uncaught during a 'commit' action).
  559 instance Alternative TT where
  560   empty = throwError $ TTError "" [] False
  561   f <|> g =
  562     catchError f $ \e -> case ttErrCommitted e of
  563       -- re-throw any committed errors from 'f', as they are considered non-recoverable
  564       True -> throwError e
  565       -- otherwise, attempt the alternative 'g'
  566       False -> catchError g $ \e2 -> case e2 of
  567         -- if the second error is from "empty" (i.e. no more alternatives in an msum), then
  568         -- re-throw the original error, as it is likely to be more informative
  569         TTError "" [] False -> throwError e
  570         _ -> throwError e2

  593 lookupCache ::
  594   [IORef (IntMap a)] -> Int -> TT (Maybe a)
  595 lookupCache [] _ = return Nothing
  596 lookupCache (ref:refs) i = do
  597   ecached <- liftIO $ readIORef ref
  598   case IntMap.lookup i ecached of
  599     Just cr -> return $ Just cr
  600     Nothing -> lookupCache refs i

  602 translateCached ::
  603   (TTEnv -> [IORef (IntMap (CachedResult a))]) ->
  604   (Term -> TT a) ->
  605   Term ->
  606   TT a
  607 translateCached getrefs f t = do
  608   refs <- asks getrefs
  609   mres <- lookupCache refs (termIndex t)
  610   case mres of
  611     Just (Right (a,tout)) -> tell tout >> return a
  612     Just (Left err) -> throwError err
  613     Nothing -> do
  614       me <- tryError (listen $ f t)
  615       -- only save new results in the top of the stack
  616       case refs of
  617         [] -> return ()
  618         (ref:_) ->
  619           liftIO $ modifyIORef' ref (IntMap.insert (termIndex t) me)
  620       case me of
  621         Left err -> throwError err
  622         Right (a,tout) -> tell tout >> return a

  624 -- | NOTE: this will not reliably reproduce all of the constraints
  625 --   originally associated with this type. For this, we need to first translate
  626 --   into an 'Expr' and then use type inference (see 'termToSchemaExpr')
  627 translateAsSchema :: Term -> TT (P.Schema Name)
  628 translateAsSchema t | Just (vn, a, b) <- asPi t =
  629   alts "translateAsSchema" t
  630     [ do k <- translateAsKind a
  631          withFreshTVar vn k $ do
  632            CryTParam nm <- lookupVar (SAW.vnIndex vn)
  633            P.Forall tvs ps tT rng <- translateAsSchema b
  634            return $ P.Forall (P.TParam nm (Just k) Nothing:tvs) ps tT rng
  635     , do ta <- translateAsType a
  636          P.Forall tvs ps tb rng <- translateAsSchema b
  637          return $ P.Forall tvs ps (P.TFun ta tb) rng
  638     , do p <- translateAsConstraint a
  639          P.Forall tvs ps tb rng <- translateAsSchema b
  640          return $ P.Forall tvs (p:ps) tb rng
  641     ]
  642 
  643 translateAsSchema t = do
  644   t' <- translateAsType t
  645   return $ P.Forall [] [] t' Nothing

  647 translateAsType :: Term -> TT Type
  648 translateAsType =
  649   translateCached ttTypeCache translateAsType'

  651 translateAsType' :: Term -> TT Type
  652 translateAsType' t = alts "translateAsType" t
  653   [ translateAsInfixTypeApp t
  654   , do [n,a] <- mreturn $ asGlobalApply "Cryptol.seq" t
  655        commit $ do
  656          n' <- translateAsType n
  657          a' <- translateAsType a
  658          return $ P.TSeq n' a'
  659   , do [n] <- mreturn $ asGlobalApply "Cryptol.TCNum" t
  660        n' <- mreturn $ (asNat n <|> asPos n)
  661        return $ P.TNum (fromIntegral n')
  662   , do mreturn $ asBoolType t
  663        return $ P.TBit
  664     -- NOTE: this will intentionally fail if 'b' depends on 'a'
  665   , do (_, a, b) <- mreturn $ asPi t
  666        b' <- translateAsType b
  667        commit $ do
  668         a' <- translateAsType a
  669         return $ P.TFun a' b'
  670   , do (vn, _) <- mreturn $ asVariable t
  671        CryTParam nm <- lookupVar (SAW.vnIndex vn)
  672        return $ userT nm []
  673   , do (n,a) <- mreturn $ asVectorType t
  674        n' <- translateAsType n
  675        commit $ do
  676         a' <- translateAsType a
  677         return $ P.TSeq n' a'
  678   , do n <- mreturn $ (asNat t <|> asPos t)
  679        let i = fromIntegral n
  680        return $ P.TNum i
  681   , do ts@(_:_) <- mreturn $ asTupleType t
  682        commit $ do
  683          ts' <- mapM translateAsType ts
  684          return $ P.TTuple ts'
  685   , do flds@(_:_) <- mreturn $ asRecordType t
  686        commit $ do
  687           flds' <- forM flds $ \(fld,fldT) -> do
  688             fldT' <- translateAsType fldT
  689             return (C.mkIdent fld,(P.emptyRange, fldT'))
  690           return $ P.TRecord $ C.recordFromFields flds'
  691   , do (tc, nm, ts) <- translateAsTCon t
  692        case tc of
  693          C.TC _ -> return ()
  694          C.TF _ -> return ()
  695          _ -> errMsg $ "Unexpected type constructor: " ++ (show $ pp tc)
  696        commit $ do
  697          ts' <- mapM translateAsType ts
  698          return $ userT nm ts'
  699   ]

  740 nameAliases :: C.Namespace -> Name -> TT Int
  741 nameAliases ns nm = case nm of
  742   P.UnQual' ident _ -> do
  743     i <- nameUses ns (P.UnQual' ident C.SystemName)

  748 uncheckName :: C.Name -> TT Name
  749 uncheckName nm = do
  750   let checkAmbig pn = do
  751         i <- nameAliases (C.nameNamespace nm) pn
  752         if i > 1 then empty else return pn
  753   pnm <- msum
  754     [ checkAmbig (C.nameToDefPName nm)
  755     , checkAmbig (C.nameToPNameWithQualifiers nm)
  756     , do C.GlobalName _ og <- return $ C.nameInfo nm
  757          let mnm = C.topModuleFor $ C.ogModule og
  758          pnm <- case C.nameToPNameWithQualifiers nm of
  759            P.UnQual' i _ -> return $ P.mkQual mnm i
  760            P.Qual ps i -> 
  761              let mnm' = C.packModName $ C.modNameChunksText mnm ++ C.modNameChunksText ps
  762              in return $ P.mkQual mnm' i
  763            _ -> empty
  764          checkAmbig pnm
  765     , commit $ errMsg $ "Could not disambiguate name: " ++ show (pp nm)
  766     ]
  767   addName pnm nm
  768   return pnm

  775 translateApp :: Bool -> [Term] -> TT ([Type],[Expr])
  776 translateApp _ [] = return ([],[])
  777 translateApp useType (arg:args) = do
  778   (argTs,argEs) <- translateApp useType args
  779   alts "translateApp" arg
  780     [ do argT <- termType arg
  781          _ <- translateAsConstraint argT
  782          return (argTs, argEs)
  783     , do arg' <- if useType 
  784            then translateAsTypedExpr arg 
  785            else translateAsExpr arg
  786          return (argTs,arg':argEs)
  787     , do arg' <- translateAsType arg
  788          return (arg':argTs,argEs)
  789     ]

  822 withShared1 :: Term -> (Maybe (Name,Expr,Type) -> TT a) -> TT a
  823 withShared1 t f = do
  824   m <- asks ttBoundExprs
  825   case IntMap.member (termIndex t) m of
  826     True -> f Nothing
  827     False -> tryError (translateAsExpr t) >>= \case
  828       Right e | shouldMemoizeExpr e -> do
  829         nm <- mkFreshName "x"
  830         tT <- typeOfExpr e >>= \case
  831           Nothing -> termType t >>= translateAsType
  832           Just tT -> return tT
  833         local (\env -> env 
  834           { ttBoundExprs = IntMap.insert (termIndex t) nm (ttBoundExprs env) 
  835           }) $ local (bindVar nm (Left tT)) $ inCacheFrame $ f (Just (nm,e,tT))
  836       _ -> f Nothing

  853 -- | First extract any shared subterms (at this binding level) and generate
  854 --   where-bindings. Then translate with the fresh bindings in scope, where
  855 --   the bound name will be used in place of the shared term during
  856 --   translation.
  857 translateAsExprShared :: Term -> TT Expr
  858 translateAsExprShared t = do
  859   let shared = map fst $ IntMap.elems $ 
  860         IntMap.filter (\(t',cnt) -> cnt >= 2 && SAW.shouldMemoizeTerm t') $
  861         SAW.scTermCount False t
  862   go shared []
  863   where
  864     go :: [Term] -> [(Name,Expr,Type)] -> TT Expr
  865     go [] [] = translateAsExpr t
  866     go [] acc = do
  867       e <- translateAsExpr t
  868       return $ P.EWhere e $ map (\(nm,e',tT) -> (P.DBind $ mkBind nm e' tT)) acc
  869     go (t':ts) acc = withShared1 t' $ \case
  870       Nothing -> go ts acc
  871       Just (nm,e,tT) -> go ts ((nm,e,tT):acc)

  873 translateLambda :: [(SAW.VarName, Term)] -> Term -> TT ([(Name,P.Kind)], Expr)
  874 translateLambda vars fn = withVars vars $ do
  875   let asParam (vn,tT) =
  876         (do CryParam nm <- lookupVar (SAW.vnIndex vn)
  877             tT' <- translateAsType tT
  878             return $ Just $ P.PTyped (P.PVar (noLoc nm)) tT')
  879         <|> return Nothing
  880   let asTParam (vn,tT) =
  881         (do CryTParam nm <- lookupVar (SAW.vnIndex vn)
  882             tK <- translateAsKind tT
  883             return $ Just (nm, tK))
  884         <|> return Nothing
  885   vars' <- catMaybes <$> mapM asParam vars
  886   tvars' <- catMaybes <$> mapM asTParam vars
  887   fn1 <- translateAsExprShared fn
  888   let fn2 = case vars' of
  889         [] -> fn1
  890         _ -> P.EFun P.emptyFunDesc vars' fn1
  891   return (tvars', fn2)

  904 translateAsConst :: Term -> TT (Name, Maybe C.Fixity)

 1007 -- | Given a 'Term' as the source of a translated 'Expr', annotate
 1008 --   the 'Expr' with an appropriate type if it can't be deduced
 1009 --   from the context.
 1010 asTypedExpr :: Expr -> Term -> TT Expr
 1011 asTypedExpr e t = typeOfExpr e >>= \case
 1012   Just{} -> return e
 1013   Nothing -> do
 1014     tt <- termType t
 1015     tt' <- translateAsType tt
 1016     return $ P.ETyped e tt'

 1018 -- | Try to deduce the type of an 'Expr' from the current context.
 1019 --   Used to drop redundant type annotations as a post-processing step.
 1020 typeOfExpr :: Expr -> TT (Maybe Type)
 1021 typeOfExpr e0 = tryMaybe $ go e0
 1022   where
 1023     go :: Expr -> TT Type
 1024     go e = case e of
 1025       P.EVar nm -> do
 1026         vts <- asks ttVarTypes
 1027         case Map.lookup nm vts of
 1028           Just (Left ty) -> return ty
 1029           _ -> do
 1030             (P.Forall [] [] ty _) <- getConstType nm
 1031             return ty
 1032       P.ETyped _ ty -> return ty
 1033       P.EInfix e1 (Pos.thing -> nm) _ e2 -> do
 1034         isBinop nm >>= \case
 1035           True -> go e1 <|> go e2
 1036           False -> empty
 1037       P.EParens e1 -> go e1
 1038       P.ELocated e1 _ -> go e1
 1039       P.EList (es@(e1:_)) -> P.TSeq (P.TNum (fromIntegral $ length es)) <$> go e1
 1040       P.ETuple es -> P.TTuple <$> mapM go es
 1041       P.EFun desc ((P.PTyped (P.PVar (Pos.thing -> nm)) tArg):ps) e1 ->
 1042         local (bindVar nm (Left tArg)) $ do
 1043           t1 <- go (P.EFun desc ps e1)
 1044           return $ P.TFun tArg t1
 1045       P.EFun _ [] e1 -> go e1
 1046       P.EWhere e2 (P.DBind d:ds) |
 1047         (P.bSignature -> Just (P.thing -> P.Forall [] [] tT _)) <- d ->
 1048         let nm = Pos.thing (P.bName d)
 1049         in local (bindVar nm (Left tT)) $ go $ P.EWhere e2 ds
 1050       P.EWhere e2 [] -> go e2
 1051       -- NOTE: Cryptol has only two prefix operations, which are both
 1052       -- type-preserving
 1053       P.EPrefix _ e1 -> go e1
 1054       _ -> empty
 1055 
 1056 -- | Check if this constant is a
 1057 --   (possibly polymorphic) type-preserving binary operation
 1058 isBinop :: Name -> TT Bool
 1059 isBinop nm = fmap isJust $ tryMaybe $ do
 1060   P.Forall _ _ (P.TFun t1 (P.TFun t2 t3)) _ <- getConstType nm
 1061   True <- return $ t1 == t2 && t2 == t3
 1062   return ()

 1064 translateAsInfixExprApp :: Term -> TT Expr
 1065 translateAsInfixExprApp t = do
 1066   (fn, args@(_:_)) <- return $ asApplyAll t
 1067   (nm, fx) <- asInfixExprOp fn
 1068   binOp <- isBinop nm
 1069   (_,[e1,e2]) <- translateApp True args
 1070   -- if the operand is always a type-preserving binop, then
 1071   -- we can strip the individual type annotations
 1072   let (e1',e2') = if binOp then (dropType e1, dropType e2) else (e1,e2)
 1073   let t' = P.EInfix (eParens e1') (noLoc nm) fx (eParens e2')
 1074   asTypedExpr t' t

 1078   t' <- translateAsExpr t
 1079   case t' of
 1080     P.ETyped{} -> return t'
 1081     P.EVar{} -> return t'
 1082     _ -> asTypedExpr t' t

 1089 unNumber :: Expr -> TT Expr
 1090 unNumber e = case e of
 1091   P.EAppT (P.EVar nm) [P.PosInst val, P.PosInst rep] -> do
 1092     number <- (uncheckName =<< constToName =<< lookupSAWConst "Cryptol.ecNumber")
 1093     case nm == number of
 1094       True -> unNumber =<< eTyped (P.ETypeVal val) rep
 1095       False -> return e
 1096   P.ETyped (P.ETypeVal (P.TNum n)) rep -> do
 1097     let i = fromIntegral n
 1098     eTyped (P.ELit (P.ECNum i (P.DecLit (Text.pack $ show i)))) rep
 1099   _ -> return e

 1182 withVar :: SAW.VarName -> Term -> TT a -> TT a
 1183 withVar vn t f = alts "withVar"  t
 1184   [ do t' <- translateAsType t
 1185        uncommitNat $ withFreshVar vn t' f
 1186   , do _ <- translateAsConstraint t
 1187        commit $ f
 1188   , do t' <- translateAsKind t
 1189        uncommitNat $ withFreshTVar vn t' f
 1190   ]
 1191   where
 1192     -- Nat can be treated as both a value and a type, so
 1193     -- we may need to attempt both translations, ignoring
 1194     -- otherwise unrecoverable errors
 1195     uncommitNat :: TT a -> TT a
 1196     uncommitNat g = case asNatType t of
 1197       Just () -> uncommit g
 1198       Nothing -> commit g
