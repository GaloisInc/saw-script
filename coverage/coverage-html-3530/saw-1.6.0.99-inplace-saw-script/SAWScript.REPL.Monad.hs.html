   74 -- REPL state and state monad
   75 
   76 -- | The state maintained by the REPL.
   77 --
   78 -- rContinue starts True and remains true until someone runs @:q@,
   79 -- at which point the main REPL loop exits.
   80 --
   81 -- rIsBatch is True if we're executing the REPL from a file in
   82 -- batch mode (@saw -B@).
   83 --
   84 -- rTopLevelRO and rTopLevelRW are the SAWScript interpreter's
   85 -- TopLevel monad state.
   86 --
   87 -- rProofState is the SAWScript interpreter's ProofScript monad state;
   88 -- it should be `Nothing` when we're running an ordinary TopLevel REPL
   89 -- and `Just` when we're running a ProofScript REPL.
   90 --
   91 -- We carry the state around separately like this (rather than
   92 -- embedding the interpreter's monads) so we can run in either
   93 -- context. This could be done with a monad typeclass instead, though
   94 -- likely that would end up being a lot more complicated, and as we've
   95 -- seen in the interpreter core, it's very easy to mess up code like
   96 -- that so it executes some bits in the wrong monad, which will then
   97 -- generally typecheck and only fail at runtime if you happen to try
   98 -- the right things. So it's probably better this way, although I'd
   99 -- prefer that the internal partitioning of the interpreter state
  100 -- wasn't exposed.
  101 -- 
  102 data REPLState = REPLState
  103   { rContinue   :: Bool
  104   , rIsBatch    :: Bool
  105   , rTLHook     :: TopLevelShellHook
  106   , rPSHook     :: ProofScriptShellHook
  107   , rTCIncremental :: TCIncremental
  108   , rTopLevelRO :: TopLevelRO
  109   , rTopLevelRW :: TopLevelRW
  110   , rProofState :: Maybe ProofState
  111   }
  112 
  113 -- | Create an initial, empty environment.
  114 initREPL ::
  115     Bool -> Options -> TopLevelShellHook -> ProofScriptShellHook ->
  116     IO REPLState
  117 initREPL isBatch opts tlhook pshook =
  118   do (ro, rw, tcenv) <- buildTopLevelEnv opts [] tlhook pshook
  119      return REPLState
  120        { rContinue   = True
  121        , rIsBatch    = isBatch
  122        , rTLHook     = tlhook
  123        , rPSHook     = pshook
  124        , rTCIncremental = tcenv
  125        , rTopLevelRO = ro
  126        , rTopLevelRW = rw
  127        , rProofState  = Nothing
  128        }
  129 
  130 -- | Create an environment from an existing interpreter state.
  131 --
  132 -- FUTURE: it might be nice to be able to read subshell input from a
  133 -- file in batch mode. Or from the same file that's feeding the parent
  134 -- shell, though that's probably difficult. For now, assume any
  135 -- subshells are intended to be interactive.
  136 resumeREPL ::
  137     TopLevelShellHook -> ProofScriptShellHook -> TCIncremental ->
  138     TopLevelRO -> TopLevelRW -> Maybe ProofState ->
  139     REPLState
  140 resumeREPL tlhook pshook tcenv ro rw mpst =
  141     REPLState {
  142         rContinue   = True,
  143         rIsBatch    = False,
  144         rTLHook     = tlhook,
  145         rPSHook     = pshook,
  146         rTCIncremental = tcenv,
  147         rTopLevelRO = ro,
  148         rTopLevelRW = rw,
  149         rProofState = mpst
  150     }

  152 -- | REPL monad context.
  153 newtype REPL a = REPL { unREPL :: StateT REPLState IO a }
  154   deriving (
  155       Applicative, Functor, Monad, MonadThrow, MonadCatch, MonadMask, MonadFail
  156   )
  157 
  158 deriving instance MonadState REPLState REPL
  159 
  160 -- | Run a REPL action on a REPL state.
  161 runREPL :: REPL a -> REPLState -> IO (a, REPLState)
  162 runREPL m st = do
  163     runStateT (unREPL m) st
  164 
  165 -- | Run an IO action in a REPL context.
  166 instance MonadIO REPL where
  167   liftIO m = REPL (liftIO m)
  168 
  169 -- | Run a TopLevel action in a REPL context.
  170 liftTopLevel :: TopLevel a -> REPL a
  171 liftTopLevel m = do
  172     ro <- getTopLevelRO
  173     rw <- getTopLevelRW
  174     (result, rw') <- liftIO $ runTopLevel m ro rw
  175     modify (\st -> st { rTopLevelRW = rw' })
  176     return result
  177 
  178 -- | Run a ProofScript action in a REPL context.
  179 liftProofScript :: ProofScript a -> REPL a
  180 liftProofScript m = do
  181     mpst <- gets rProofState
  182     let pst = case mpst of
  183           Nothing -> panic "liftProofScript" ["Not in ProofScript mode"]
  184           Just ps -> ps
  185     (result, pst') <-
  186         liftTopLevel $ runStateT (runExceptT (unProofScript m)) pst
  187     modify (\st -> st { rProofState = Just pst' })
  188     liftTopLevel $ case result of
  189        Left (stats, cex) ->
  190          do ppOpts <- SV.getPPOpts
  191             fail (Text.unpack $ ppProofResult ppOpts (InvalidProof stats cex pst'))
  192        Right x -> return x

  231 captureError :: TopLevelCheckpoint -> Maybe String -> REPL ()
  232 captureError chk mmsg = do
  233     case mmsg of
  234         Nothing -> pure ()
  235         Just msg -> do
  236             liftIO $ putStrLn ""
  237             liftIO $ putStrLn msg
  238     liftTopLevel $ restoreCheckpoint chk

  240 -- | Inspect and handle or rethrow exceptions.
  241 --
  242 --   Note: this would be cleaner if it used `catches` with multiple
  243 --   cases; however, the docs say nothing about what `catches` does
  244 --   with overlapping cases, or if it actually works currently,
  245 --   whether that's a supported part of the interface.
  246 --
  247 --   FUTURE: with enough cleanup in lower-level code we could make
  248 --   this not need a blanket case and then `catches` becomes viable.
  249 --
  250 --   This logic controls which exceptions the REPL continues executing
  251 --   for. Anything we rethrow here is fatal and will cause the REPL to
  252 --   exit.
  253 --
  254 handleExceptions :: TopLevelCheckpoint -> X.SomeException -> REPL ()
  255 handleExceptions chk e
  256 
  257     -- Cons.Fatal: catch and continue. We don't need to print the
  258     -- exception; the point of `Fatal` vs. anything else is that
  259     -- we already printed the error messages.
  260   | Just (_ :: Cons.Fatal) <- X.fromException e =
  261         captureError chk Nothing
  262 
  263     -- IO exceptions: catch and continue on UserError (this includes
  264     -- `fail` calls), but treat anything else as fatal.
  265   | Just (e' :: X.IOException) <- X.fromException e =
  266         if isUserError e' then
  267             captureError chk $ Just $ ioeGetErrorString e'
  268         else
  269             throwM e'
  270 
  271     -- Async exceptions: treat as fatal.
  272   | Just (e' :: X.AsyncException) <- X.fromException e =
  273         throwM e'
  274 
  275     -- ExitCode arises when someone makes an explicit call to exit
  276     -- (`exitSuccess` or `exitWith`); assume they meant it.
  277   | Just (e' :: ExitCode) <- X.fromException e =
  278         throwM e'
  279 
  280     -- Exceptions generated by panic: treat as fatal.
  281   | Just (e' :: PanicSupport.PanicException) <- X.fromException e =
  282         throwM e'
  283 
  284     -- Trap anything and everything else. XXX: this is way too broad.
  285   | otherwise =
  286         captureError chk $ Just $ X.displayException e

  290 -- | Catch the exceptions we expect to catch while running a REPL
  291 --   action.
  292 exceptionProtect :: REPL () -> REPL ()
  293 exceptionProtect cmd = do
  294     chk <- liftTopLevel makeCheckpoint
  295     cmd `catch` handleExceptions chk
