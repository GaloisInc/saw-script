   23 basic_ss :: SharedContext -> IO (Simpset a)
   24 basic_ss sc =
   25   do rs1 <- concat <$> traverse (defRewrites sc) defs

   79 defRewrites :: SharedContext -> QualName -> IO [RewriteRule a]
   80 defRewrites sc qn =
   81   do mname <- scResolveQualName sc qn
   82      case mname of
   83        Nothing -> pure []
   84        Just nm ->
   85          do mm <- scGetModuleMap sc
   86             case lookupVarIndexInMap (nameIndex nm) mm of
   87               Just (ResolvedDef def) -> scDefRewriteRules sc def
   88               _ -> pure []
