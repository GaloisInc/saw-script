   85 -- | ImportVisibility is an enumeration that indicates how we handle
   86 -- the visibility of the symbols in an imported module.
   87 --
   88 -- `OnlyPublic` makes only the public/exported symbols of a module
   89 -- visible from SAW. `PublicAndPrivate` instead makes all symbols
   90 -- visible, as if one is working inside it. The latter is often useful
   91 -- (or necessary) for verification and code generation.
   92 --
   93 -- `PublicAndPrivate` is akin to setting the module focus in the
   94 -- Cryptol REPL; however, it uses a simpler internal mechanism and is
   95 -- only settable at import time.
   96 --
   97 -- (See 'CryptolEnv.importCryptolModule'.)
   98 --
   99 -- NOTE: this notion of public vs. private symbols is specific to
  100 -- SAWScript and distinct from Cryptol's notion of private
  101 -- definitions.
  102 --
  103 -- FUTURE: this should probably be replaced with a way to manipulate
  104 -- the module focus like the Cryptol REPL uses. What you really want
  105 -- is not to expose module innards that weren't meant to be exposed,
  106 -- but to go inside to prove things in the module's internal context.
  107 --
  108 data ImportVisibility
  109   = OnlyPublic       -- ^ behaves like a normal Cryptol @import@
  110   | PublicAndPrivate -- ^ allows viewing of both @private@ sections
  111                      --   and (arbitrarily nested) submodules.
  112   deriving (Eq, Show)
  113 
  114 -- | type synonym indicating module is nested (is submodule)
  115 type IsSubmodule = Bool

  125     }

  127 -- | The global environment for capturing the Cryptol state, both
  128 --   Cryptol's own state and the state associated with
  129 --   importing/translating into SAWCore. This is intended to be a
  130 --   write-once record of any work done during translation. Rather than
  131 --   using the fields of this datatype directly, it is accessed as
  132 --   'SharedContext' data via the setters and getters below.
  133 data GlobalCryptolEnv = GlobalCryptolEnv
  134   { geModuleEnv   :: ME.ModuleEnv
  135   -- | Invariant: This is a subset of 'geAllVars', which is
  136   --   enforced by 'addExtraVars'

  138   , geExtraTySyns :: Map C.Name C.TySyn
  139   , geAllVars     :: Map C.Name C.Schema
  140   , geAllTerms    :: Map C.Name Term
  141   , geRefPrims    :: Map C.PrimIdent C.Expr
  142   , gePrims       :: Map C.PrimIdent Term
  143   , gePrimTypes   :: Map C.PrimIdent Term
  144   , geFFITypes    :: Map NameInfo C.FFI
  145   , geInstances   :: Maybe IntroRuleSet
  146   }

  152     GlobalCryptolEnv modEnv

  158 -- | Restore a `GlobalCryptolEnv` from a checkpoint. The first argument
  159 --   @chkEnv@ is the `GlobalCryptolEnv` saved by / copied into the
  160 --   checkpoint; the second argument @newEnv@ is the current one
  161 --   we wish to overwrite by rolling back to the checkpoint.
  162 --   The 'ME.meNameSeeds' and 'ME.meSupply' from the
  163 --   module environment are not rolled back, to avoid re-using old
  164 --   names.
  165 --   NOTE: This effectively
  166 --   invalidates any translated 'Term's or Cryptol expressions created
  167 --   after the checkpoint. Attempting to use them in the restored
  168 --   environment will have unpredictable results, and likely will
  169 --   result in a panic. Similarly, 'CryptolEnv's captured after the
  170 --   checkpoint are no longer safe to use in the resulting environment.
  171 
  172 --   We also ought to invalidate terms constructed since the checkpoint
  173 --   was taken, like SAWCore does. See #2859.
  174 
  175 --   We could, for example, have 'CryptolScope' track which
  176 --   identifiers it references, and check that they are in a valid
  177 --   range with respect to the corresponding global environment
  178 --   before combining them into a 'CryptolEnv'.
  179   restoreMetadata chk now = return $
  180     let newMEnv = geModuleEnv now
  181         chkMEnv = geModuleEnv chk
  182     in chk { geModuleEnv = chkMEnv
  183                { ME.meNameSeeds = ME.meNameSeeds newMEnv
  184                , ME.meSupply = ME.meSupply newMEnv
  185                }
  186            }

  188 -- | A scope frame that captures which Cryptol names are accessible.
  189 --  `fNamingEnv` is the local naming environment, which can be extended
  190 --  ad-hoc with additional declarations. `fImports` is a list of all
  191 --  the modules that have been imported, and a visibility setting for
  192 --  each. This does not include, for example, builtin modules that
  193 --  exist but that have not been imported.
  194 
  195 data CryptolFrame =
  196   CryptolFrame { fNamingEnv :: MR.NamingEnv
  197                , fImports :: [ImportData]
  198                }
  199 
  200 initFrame :: CryptolFrame
  201 initFrame = CryptolFrame mempty mempty
  202 
  203 -- | A nonempty list of 'CryptolFrame's, where the first element is the
  204 --   "bottom" frame that takes highest precedence when looking up
  205 --   names. Each individual frame only contains values declared at
  206 --   exactly that level. The full scope is computed by collecting
  207 --   everything in this stack, via 'eExtraNaming' and 'eImports'.
  208 newtype CryptolEnv = CryptolEnv (NonEmpty CryptolFrame)
  209 
  210 initEnv :: CryptolEnv
  211 initEnv = CryptolEnv (initFrame :| [])
  212 
  213 isToplevel :: CryptolEnv -> Bool
  214 isToplevel (CryptolEnv (_ :| frames)) = null frames
  215 
  216 -- | Test if the scopes have the same number of frames pushed.
  217 sameHeight :: CryptolEnv -> CryptolEnv -> Bool
  218 sameHeight (CryptolEnv scope1) (CryptolEnv scope2) =
  219   NE.length scope1 == NE.length scope2
  220 
  221 mapCurFrame ::
  222   (CryptolFrame -> CryptolFrame) ->
  223   CryptolEnv ->
  224   CryptolEnv
  225 mapCurFrame f (CryptolEnv (frame :| frames)) =
  226   CryptolEnv (f frame :| frames)
  227 
  228 -- | Push a fresh frame onto the stack.
  229 pushScope :: CryptolEnv -> CryptolEnv
  230 pushScope (CryptolEnv frames) = CryptolEnv (initFrame <| frames)
  231 
  232 -- | Pop the current frame from the stack, discarding its
  233 --   contents. Panics if this is the only frame.
  234 popScope :: CryptolEnv -> CryptolEnv
  235 popScope (CryptolEnv frames) = case snd (NE.uncons frames) of
  236   Nothing -> panic "popScope" [ "Popping topmost scope"]
  237   Just frames' -> CryptolEnv frames'
  238 
  239 -- | Map the naming environment of the frame currently in scope.
  240 mapNaming ::
  241   (MR.NamingEnv -> MR.NamingEnv) ->
  242   CryptolEnv ->
  243   CryptolEnv
  244 mapNaming f = mapCurFrame $
  245       \fr -> fr {fNamingEnv = f (fNamingEnv fr) }
  246 
  247 -- | Map the module imports of the frame currently in scope.
  248 mapImports ::
  249   ([ImportData] -> [ImportData]) ->
  250   CryptolEnv ->
  251   CryptolEnv
  252 mapImports f = mapCurFrame $
  253       \fr -> fr {fImports = f (fImports fr) }

  255 -- | Run the inner action bracketed new frame pushed/popped on
  256 --   the 'CryptolScope' stack.
  257 --   Fails if the inner action changes the scope height
  258 --   (i.e. it does not properly bracket its pushes and pops).
  259 withFreshScope ::
  260   MonadFail m =>
  261   CryptolEnv ->
  262   (CryptolEnv ->
  263   m (a, CryptolEnv)) ->
  264   m (a, CryptolEnv)
  265 withFreshScope env0 f = do
  266   let env1 = pushScope env0
  267   (a, env2) <- f env1
  268   unless (sameHeight env1 env2) $
  269     panic "withFreshScope" ["mismatched push/pops"]
  270   let env3 = popScope env2
  271   return (a, env3)

  273 -- | Access the 'C.Supply' in the global 'ME.ModuleEnv' for generating
  274 --   fresh names. More efficient than 'runModuleM', as it avoids any other
  275 --   bookkeeping.
  276 useModEnvSupply :: SharedContext -> (C.Supply -> (a, C.Supply)) -> IO a
  277 useModEnvSupply sc f = do
  278   modEnv <- eModuleEnv sc
  279   let (a, supply) = f $ ME.meSupply modEnv
  280   mapModEnv sc (\modEnv_ -> modEnv_ { ME.meSupply = supply } )
  281   return a

  295 -- The "getters" below were historically fields in 'CryptolEnv', which
  296 -- are now defined functions that access either the 'GlobalCryptolEnv'
  297 -- or the 'CryptolScope' as appropriate. In contrast, updates were
  298 -- historically made by directly accessing 'CryptolEnv' fields, and are
  299 -- now restricted to only adding new entries to the maps in
  300 -- 'GlobalCryptolEnv' (e.g. 'addExtraVars'). This enforces the policy
  301 -- that the global environment is intended to be a complete record of
  302 -- all work done during translation/import, regardless of scope. Since
  303 -- the maps in the global environment are all keyed on globally-unique
  304 -- values, these operations are expected to only add entries and never
  305 -- overwrite existing ones.
  306 --
  307 -- NOTE: We could enforce a write-once policy here, but it's
  308 -- not immediately obvious if we need to support key clashes
  309 -- for equal entries (not possible for
  310 -- some types like 'C.Expr'), or if it is even useful to do so.

  380 -- | Run an 'MM.ModuleM' action using the module environment from
  381 -- 'eModuleEnv'. If the action is successful, updates the module
  382 -- environment in the 'SharedContext' and returns the result of the
  383 -- action. Otherwise, the environment is left unchanged and a
  384 -- 'MM.ModuleError' is returned.
  385 runModuleM ::
  386  SharedContext -> MM.ModuleM a -> IO (Either MM.ModuleError a, [MM.ModuleWarning])
  387 runModuleM sc m = do
  388   fileReader <- scFileReader sc
  389   env <- eModuleEnv sc
  390   let minp solver = MM.ModuleInput {
  391           MM.minpCallStacks = True,
  392           MM.minpSaveRenamed = False,
  393           MM.minpEvalOpts = pure defaultEvalOpts,
  394           MM.minpByteReader = fileReader,
  395           MM.minpModuleEnv = env,
  396           MM.minpTCSolver = solver
  397       }
  398   (res, ws) <- SMT.withSolver (return ()) (meSolverConfig env) $ \solver ->
  399     MM.runModuleM (minp solver) m
  400   case res of
  401     Right (a, env') -> do
  402       mapGlobal sc $ \genv ->
  403         refreshCryptolEnv $ genv { geModuleEnv = env' }
  404       return (Right a, ws)
  405     Left err -> return (Left err, ws)

  432 -- | Map from Cryptol names to Cryptol types. This is
  433 -- used to call `fastTypeOf` and `fastSchemaOf` on Cryptol expressions
  434 -- to fetch their types. This table is implicitly updated after every
  435 -- 'runModuleM' action to ensure it is always up-to-date.
  436 --
  437 -- FUTURE: in principle we should be able to use the SAWCore types of
  438 -- the SAWCore terms after importing them, instead of `fastTypeOf` and
  439 -- `fastSchemaOf`, and drop the `eAllVars` table. In practice, doing
  440 -- that relies (in some cases) on being able to call `scCryptolType`
  441 -- to reconstruct the Cryptol-level type; that in turn requires, when

  445 -- doesn't currently exist. Furthermore, while we've fixed many of the

  448 -- polymorphic type schemes we erase certain typeclasses in the
  449 -- translation, and that loses info, so we might need to translate

  505 -- | The "extra" naming environment that captures Cryptol names
  506 --   which don't correspond to any imported module. Generally these
  507 --   result from names created in SAW which have been reflected into
  508 --   the Cryptol environment.
  509 --   This is scoped content, where the accessible names are expected
  510 --   to be managed by SAW and correspond to the same SAW values that
  511 --   are in scope.
  512 --
  513 -- FUTURE: Cryptol has its own functionality for additional bindings
  514 -- (it uses it for things created from the Cryptol REPL) and we ought
  515 -- to be able to use it instead of bolting on our own additional layer
  516 -- of material. Doing so would avoid various inconsistencies and
  517 -- irregularities that can creep in when we reimplement Cryptol name
  518 -- resolution.
  519 eExtraNaming :: CryptolEnv -> MR.NamingEnv
  520 eExtraNaming (CryptolEnv (frame :| frames)) =
  521   foldr (\fr ne -> ne `MR.shadowing` (fNamingEnv fr)) (fNamingEnv frame) frames

  523 -- | The list of Cryptol modules which have been brought into the
  524 --   current scope. This essentially acts as a filter on the full set
  525 --   of loaded modules in the global module environment ('eModuleEnv'),
  526 --   where the contents of the selected modules are brought into scope
  527 --   according to the associated 'ImportVisibility'. The modules here
  528 --   should only correspond to modules that are present in the module
  529 --   environment *and* have been translated into SAWCore.
  530 eImports :: CryptolEnv -> [ImportData]
  531 eImports (CryptolEnv frames) =
  532   concat $ map fImports $ NE.toList frames

  535 -- | Refresh 'geAllVars' after updating the module environment.
  536 --   Previously (before 'GlobalCryptolEnv'), this would overwrite the
  537 --   'eAllVars' field. Now this will add new vars (i.e. from
  538 --   newly-added modules) but keep existing ones, as the
  539 --   'GlobalCryptolEnv' should never drop entries. If the module
  540 --   environment has been managed properly, we expect any clashing keys
  541 --   to have equal elements, but this is not checked/enforced. Not
  542 --   exported.
  543 refreshCryptolEnv :: GlobalCryptolEnv -> GlobalCryptolEnv
  544 refreshCryptolEnv genv =
  545      let modEnv = geModuleEnv genv
  546          ifaceDecls = getAllIfaceDecls modEnv
  547          newtypeCons = Map.fromList
  548                          [ con
  549                          | nt <- Map.elems (MI.ifNominalTypes ifaceDecls)
  550                          , con <- C.nominalTypeConTypes nt
  551                          ]
  552          vars = Map.map MI.ifDeclSig $ MI.ifDecls ifaceDecls
  553          -- note that we don't need to re-add geExtraVars, because
  554          -- it is already included in the existing geAllVars
  555          allvars = newtypeCons `Map.union` vars
  556          allvars' = Map.union allvars (geAllVars genv)
  557      in genv { geAllVars = allvars' }
