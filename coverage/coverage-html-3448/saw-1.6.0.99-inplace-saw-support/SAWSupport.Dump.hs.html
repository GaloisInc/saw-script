    8 Dump infrastructure.
    9 
   10 Dumping is different from prettyprinting. The goal of a prettyprinter
   11 is to recreate concrete syntax, including things like removing
   12 unnecessary parentheses, in pursuit of generating input that's ideally
   13 as readable as the original source text.
   14 
   15 The goal of a dumper is somewhat different. Readability is still a
   16 concern, but the primary goal is to make sure the output reflects
   17 exactly what the representation contains. Prettyprinter output will
   18 usually, for example, not distinguish the difference between
   19 ```
   20    Add (Add (Var x) (Var y)) (Var z)
   21 ```
   22 and
   23 ```
   24    Add (Var x) (Add (Var y) (Var z))
   25 ```
   26 Most of the time, that doesn't matter. However, when you're chasing
   27 after a problem where your optimizing pass is failing to see `y`, it
   28 matters a great deal. Dumps are intended to serve this debugging need.
   29 
   30 The schematic dump format supported by this module has the general
   31 format
   32 ```
   33     Heading
   34         Subelement
   35         Subelement
   36         Subelement
   37 ```
   38 
   39 or
   40 
   41 ```
   42     Heading
   43         field
   44             Element
   45         field
   46             Element
   47 ```
   48 
   49 where the heading is generally the AST constructor name plus any
   50 information that reliably fits on the same line.
   51 
   52 Small single subelements may be pulled up onto the heading line to
   53 save vertical space. This is automatic.
   54

   56 ```

   58     import SAWSupport.Dump (Dump)

   96 ------------------------------------------------------------
   97 -- Dump representation
   98 
   99 -- | Type for a collected but not yet printed dump, akin to a
  100 --   prettyprinter document. But we do a lot less formatting, so it
  101 --   can be a lot simpler.
  102 --
  103 --   There's nothing especially canonical about this representation;
  104 --   it's meant to serve observed needs.
  105 --
  106 --   `Phantom` is empty and produces no output, except in certain
  107 --   contexts it appears as "---".
  108 --
  109 --   `Line` is some text that fits on one line.
  110 --
  111 --   `List` is a series of dumps.
  112 --
  113 --   `Elements` is a heading and a series of dumps.
  114 --
  115 --   `Fields` is a heading and a series of labelled dumps.
  116 --
  117 --   `Surround` is like `Elements` but also has a footer.
  118 --
  119 data Dump
  120     = Phantom
  121     | Line Text
  122     | List [Dump]
  123     | Elements Text [Dump]
  124     | Fields Text [(Text, Dump)]
  125     | Surround Text [Dump] Text

  129 -- Support code
  130 
  131 -- | Convert an arbitrary string to a printable quoted string.
  132 --
  133 --   This differs from using the `Show` instance for `Text` or
  134 --   `String` (which adds quotes and escapes at least some characters)
  135 --   at least as follows:
  136 --      - it doesn't unpack or repack the whole string, and allocates
  137 --        only for non-printables;
  138 --      - it has well defined quoting and escaping behavior (the
  139 --        `Show` instances are understood to add quotes but are
  140 --        otherwise undocumented);
  141 --      - experimentation indicates that the `Show` instances escape
  142 --        at least some non-ASCII printables, which this doesn't.
  143 --
  144 --   Other reasons not to rely on the `Show` instances include:
  145 --      - it's easy to forget when making changes that the `Show`
  146 --        instances add quotes, because most analogous operations on
  147 --        strings (e.g. @pretty@) do not, and here we rely on it;
  148 --      - the fewer intentional uses of `Show` there are the easier it
  149 --        is to crack down on abuse.
  150 --
  151 stringize :: Text -> Text
  152 stringize txt0 =
  153     let visit txt results =
  154           let (keep, rest) = Text.span Char.isPrint txt in
  155           case Text.uncons rest of
  156               Nothing ->
  157                   -- reached the end
  158                   results
  159               Just (c, rest') ->
  160                   let again c' = visit rest' (c' : keep : results) in
  161                   case c of
  162                     '\a' -> again "\\a"
  163                     '\b' -> again "\\b"
  164                     '\t' -> again "\\t"
  165                     '\n' -> again "\\n"
  166                     '\v' -> again "\\v"
  167                     '\f' -> again "\\f"
  168                     '\r' -> again "\\r"
  169                     _ ->
  170                         let c' = Text.pack (showHex (Char.ord c) "") in
  171                          -- There are many ways we could print this.
  172                          -- This form is at least unambiguous...
  173                          again $ "\\(" <> c' <> ")"
  174     in
  175     Text.concat $ reverse $ ["\""] ++ visit txt0 ["\""]

  179 -- Basic dumps
  180 
  181 -- | Completely empty dump
  182 phantom :: Dump
  183 phantom = Phantom
  184 
  185 -- | Heading with no subelements. Special case of `subelements`.
  186 singleton :: Text -> Dump
  187 singleton heading = Elements heading []
  188 
  189 -- | Heading with one subelement. Special case of `subelements`.
  190 subelement :: Text -> Dump -> Dump
  191 subelement heading elt = Elements heading [elt]
  192     
  193 -- | Heading with a list of subelements.
  194 subelements :: Text -> [Dump] -> Dump
  195 subelements heading elts = Elements heading elts
  196 
  197 -- | Heading with a list of subelements, but each one comes with
  198 --   a text label.
  199 fields :: Text -> [(Text, Dump)] -> Dump
  200 fields heading items = Fields heading items
  201

  203 ------------------------------------------------------------
  204 -- Dumps for standard data types
  205 
  206 -- | Dump a Bool.
  207 bool :: Bool -> Dump
  208 bool True = Line "True"
  209 bool False = Line "False"
  210 
  211 -- | Dump an Int, Integer, or other Num.
  212 int :: (Show a, Num a) => a -> Dump
  213 int k = Line $ Text.pack $ show k
  214 
  215 -- | Dump some literal text. We don't interpret it, except to split it
  216 --   into lines if necessary so indent works.
  217 text :: Text -> Dump
  218 text txt = case Text.lines txt of
  219     [] -> Phantom
  220     [l] -> Line l
  221     ls -> List $ map Line ls
  222 
  223 -- | Dump some literal text, surrounding it with delimiters. We don't
  224 --   interpret it, except to split it into lines if necessary so
  225 --   indent works. This can be used for metasyntactic grouping symbols
  226 --   (parens, brackets, etc.) you want to have in the dump output. You
  227 --   don't want to use it for syntactic grouping; save that for the
  228 --   prettyprinting logic.
  229 surroundText :: Text -> Text -> Text -> Dump
  230 surroundText lhs txt rhs = case Text.lines txt of
  231     [] -> Line (lhs <> rhs)
  232     [l] -> Line (lhs <> l <> rhs)
  233     ls -> Surround lhs (map Line ls) rhs
  234 
  235 -- | Like `surroundText` except inserts spaces on the inside of the
  236 --   delimiters.
  237 surroundText' :: Text -> Text -> Text -> Dump
  238 surroundText' lhs txt rhs = case Text.lines txt of
  239     [] -> Line (lhs <> " " <> rhs)
  240     [l] -> Line (lhs <> " " <> l <> " " <> rhs)
  241     ls -> Surround lhs (map Line ls) rhs
  242 
  243 -- | Dump a string that's meant to be a quoted string literal.
  244 qstring :: Text -> Dump
  245 qstring txt = Line $ stringize txt
  246 
  247 -- | Dump a Maybe value using another dumper.
  248 maybe :: (a -> Dump) -> Maybe a -> Dump
  249 maybe dumpX mbX = case mbX of
  250     Nothing -> Phantom
  251     Just x -> dumpX x
  252 
  253 -- | Dump a list.
  254 list :: [Dump] -> Dump
  255 list xs = List xs
  256

  259 -- Other operations

  261 -- | Prepend some more text to the next thing. Use with discretion.
  262 prepend :: Text -> Dump -> Dump
  263 prepend newtxt d = case d of
  264     Phantom ->
  265         Line newtxt
  266     Line txt ->
  267         Line $ newtxt <> txt
  268     List ds ->
  269         Elements newtxt ds
  270     Elements txt ds ->
  271         Elements (newtxt <> txt) ds
  272     Fields txt flds ->
  273         Fields (newtxt <> txt) flds
  274     Surround lhs ds rhs ->
  275         Surround (newtxt <> lhs) ds rhs
  276

  279 -- Rendering
  280 
  281 -- | Guts of render.
  282 --
  283 --   We do the following special cases:
  284 --
  285 --   - If there's a single subelement and it's a single line, we pull
  286 --     it onto the same line as the header.
  287 --
  288 --   - If the value of a field is a single line, we pull it onto the
  289 --     same line as the field name.
  290 --
  291 --   - If the value of a field is `Phantom`, we print "---" instead of
  292 --     just leaving the field name hanging.
  293 --
  294 --   - We do _not_ do any of this for `Surround` because currently the
  295 --     only way to generate it is with a multiline text and the logic
  296 --     there already handles the short case directly.
  297 --
  298 renderAt :: Text -> Dump -> [Text]
  299 renderAt indent d = case d of
  300     Phantom ->
  301         []
  302     Line txt ->
  303         [indent <> txt]
  304     List ds ->
  305         -- List does _not_ indent any further.
  306         concatMap (renderAt indent) ds
  307     Elements header [d1] ->
  308         let indent' = "   " <> indent in
  309         case renderAt indent' d1 of
  310             [d1'] ->
  311                 let d1'' = Text.dropWhile Char.isSpace d1' in
  312                 [indent <> header <> " " <> d1'']
  313             ds' ->
  314                 (indent <> header) : ds'
  315     Elements header ds ->
  316         (indent <> header) : concatMap (renderAt ("   " <> indent)) ds
  317     Fields header flds ->
  318         let indent' = "   " <> indent
  319             indent'' = "      " <> indent
  320         in
  321         let renderField (name, d1) =
  322               case renderAt indent'' d1 of
  323                   [] ->
  324                       [indent' <> name <> ": ---"]
  325                   [d1'] ->
  326                       let d1'' = Text.dropWhile Char.isSpace d1' in
  327                       [indent' <> name <> ": " <> d1'']
  328                   ds' ->
  329                       (indent' <> name) : ds'
  330         in
  331         (indent <> header) : concatMap renderField flds
  332     Surround header ds footer ->
  333         let ds' = concatMap (renderAt ("   " <> indent)) ds in
  334         (indent <> header) : ds' ++ [indent <> footer]
  335 
  336 -- | Print to a list of (single) lines. There are no embedded newlines
  337 --   in the result.
  338 render :: Dump -> [Text]
  339 render dmp =
  340     renderAt "" dmp
  341 
  342 -- | `String` version of `render`, which we'll doubtless need.
  343 renderString :: Dump -> [String]
  344 renderString dmp = map Text.unpack $ render dmp
