  115 termOfTValue :: IsSymExprBuilder sym => SharedContext -> TValue (What4 sym) -> IO Term
  116 termOfTValue sc val =
  117   case val of
  118     VBoolType -> scBoolType sc
  119     VIntType -> scIntegerType sc
  120     VIntModType n -> scIntModType sc =<< scNat sc n
  121     VRationalType -> scRationalType sc
  122     VVecType n a ->
  123       do n' <- scNat sc n
  124          a' <- termOfTValue sc a
  125          scVecType sc n' a'
  126     VDataType (ModuleIdentifier "Prelude.UnitType") [] []
  127       -> scUnitType sc
  128     VDataType (ModuleIdentifier "Prelude.PairType") [TValue a, TValue b] []
  129       -> do a' <- termOfTValue sc a
  130             b' <- termOfTValue sc b
  131             scPairType sc a' b'
  132     VDataType (ModuleIdentifier "Prelude.EmptyType") [] []
  133       -> scRecordType sc []
  134     VDataType (ModuleIdentifier "Prelude.RecordType")
  135       [VString fname, TValue a, TValue b] []
  136       -> do fname' <- scString sc fname
  137             a' <- termOfTValue sc a
  138             b' <- termOfTValue sc b
  139             scGlobalApply sc "Prelude.RecordType" [fname', a', b']
  140     VDataType nmi ps vs ->
  141       do mnm <- scResolveQualName sc (toQualName nmi)
  142          case mnm of
  143            Just nm ->
  144              scConstApply sc nm =<< traverse (termOfSValue sc) (ps ++ vs)
  145            Nothing ->
  146              fail $ "termOfTValue: data type not found: " ++ show nmi
  147     _ -> fail $ "termOfTValue: " ++ show val
  148 
  149 termOfSValue :: IsSymExprBuilder sym => SharedContext -> SValue sym -> IO Term
  150 termOfSValue sc val =
  151   case val of
  152     VCtorApp 0 _ []
  153       -> scUnitValue sc
  154     VBool b
  155       | Just x <- W.asConstantPred b
  156       -> scBool sc x
  157     VNat n
  158       -> scNat sc n
  159     TValue tv -> termOfTValue sc tv
  160     _ -> fail $ "termOfSValue: " ++ show val
  161 
  162 termOfValue ::
  163   IsSymExprBuilder sym =>
  164   SharedContext ->
  165   TValue (What4 sym) ->
  166   SValue sym ->
  167   IO Term
  168 termOfValue sc ty val =
  169   case (ty, val) of
  170     (VDataType nmi ps _, VCtorApp n _ vv) ->
  171       do mnm <- scResolveQualName sc (toQualName nmi)
  172          case mnm of
  173            Just nm ->
  174              do mm <- scGetModuleMap sc
  175                 case lookupVarIndexInMap (nameIndex nm) mm of
  176                   Just (ResolvedDataType dt) ->
  177                     do let ctor = dtCtors dt !! n
  178                        ps' <- traverse (termOfSValue sc) ps
  179                        vv' <- traverse (\v -> termOfSValue sc =<< force v) vv
  180                        scConstApply sc (ctorName ctor) (ps' ++ vv')
  181                   _ ->
  182                     fail $ "termOfValue: data type not found: " ++ show nmi
  183            Nothing ->
  184              fail $ "termOfValue: data type not found: " ++ show nmi
  185     _ -> termOfSValue sc val
