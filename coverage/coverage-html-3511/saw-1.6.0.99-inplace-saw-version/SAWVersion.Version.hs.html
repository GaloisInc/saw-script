   14 module SAWVersion.Version (

   19 import Paths_saw (version)
   20 import SAWVersion.GitRev (foundGit, hash, branch)
   21 import Data.Version (showVersion)

   23 -- | Git branch info, in the form <hash> <branch-name>.
   24 --
   25 --   The git branch info is emitted into the autogenerated file
   26 --   @GitRev.hs@ by SAW's build system. Since in practice (despite
   27 --   various attempts to the contrary) this causes half of SAW to be
   28 --   rebuilt every time anything git-related changes, it is possible
   29 --   to suppress that update by setting @SAW_SUPPRESS_GITREV=yes@ in
   30 --   the process environment. Only do that if you need it because
   31 --   you're actively developing SAW, and be prepared to substitute
   32 --   correct current information into any panic messages you get.
   33 --
   34 gitText :: String
   35 gitText =
   36     if foundGit then
   37         case (hash, branch) of
   38             (Nothing, Nothing) -> "<non-dev build>"
   39             (Just h, Nothing) -> h ++ " <unknown-branch>"
   40             (Nothing, Just b) -> "<unknown-hash> " ++ b
   41             (Just h, Just b) -> h ++ " " ++ b
   42     else
   43         "<VCS-less build>"
   44 
   45 -- | The version number with git branch info.
   46 --   Has the form @1.6.0.99 (98b7423d master)@.
   47 shortVersionText :: String
   48 shortVersionText = showVersion version ++ " (" ++ gitText ++ ")"
   49 
   50 -- | The version text with the word \"version\" included.
   51 --   (Not sure why we bother to manifest this separately, especially
   52 --   since you'd expect to want both \"version 1.6\" and \"Version 1.6\".)
   53 versionText :: String
   54 versionText = "version " ++ shortVersionText
