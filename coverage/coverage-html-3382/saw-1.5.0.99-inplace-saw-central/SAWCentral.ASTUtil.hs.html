   26 -- NamedTyVars
   27 
   28 -- | namedTyVars is a type-class-polymorphic function for extracting named
   29 -- type variables from a type or type schema. It returns a set of Name
   30 -- (Name is just Text) manifested as a map from those Names to their
   31 -- originating positions.
   32 --
   33 -- We take the first position we see. The calls to `Map.union` are
   34 -- organized accordingly (it favors its left argument).
   35 --
   36 -- When we find a `TyVar`, we extract the position from the provenance
   37 -- with `getPos`. This is not sensible in general; however, we are
   38 -- called only from the typechecker's @generalize@ operation, which
   39 -- collects names from the bodies of polymorphic functions.
   40 --
   41 -- Those names can have two origins: they can be user-written names,
   42 -- in which case their provenance is `TypeExplicit` and the position
   43 -- is meaningful; or they can be the the results of running
   44 -- @generalize@. While we never run @generalize@ directly on the same
   45 -- function body more than once, we do (must) rescan the bodies of
   46 -- nested functions for occurrences of type variables properly
   47 -- belonging to the enclosing function. That can turn up generalized
   48 -- type variables that belong to the nested function.
   49 --
   50 -- We don't attempt to avoid extracting the positions from those type
   51 -- variables. However, we discard the results in the `Schema`
   52 -- instance: anything forall-bound within the schema is contained
   53 -- within it and shouldn't escape. So it doesn't matter if the
   54 -- positions we extract from those type variables are meaningful.
   55 -- However, it does mean extracting them shouldn't crash.
   56 --
   57 -- If things change such that extracting those variables is a problem
   58 -- we can, at the cost of a bunch of complexity, carry the
   59 -- forall-bound variables from the `Schema` into the code for `Type`
   60 -- and skip them when they appear.
   61 --
   62 -- FUTURE: given that we're now specific to being called from the
   63 -- typechecker, maybe this should get moved back to Typechecker.hs.
   64 --
   65 class NamedTyVars t where
   66   namedTyVars :: t -> Map Name Pos
   67 
   68 instance (Ord k, NamedTyVars a) => NamedTyVars (Map k a) where
   69   namedTyVars = namedTyVars . Map.elems
   70 
   71 instance (NamedTyVars a) => NamedTyVars [a] where
   72   namedTyVars ts = Map.unions $ map namedTyVars ts
   73 
   74 instance (NamedTyVars a) => NamedTyVars (Pos, a) where
   75   namedTyVars (_pos, x) = namedTyVars x
   76 
   77 instance NamedTyVars Type where
   78   namedTyVars t = case t of
   79     TyCon _ _ ts      -> namedTyVars ts
   80     TyFunc _ _ params namedParams ret ->
   81         let paramVars = namedTyVars params
   82             namedParamVars = namedTyVars namedParams
   83             retVars = namedTyVars ret
   84         in
   85         Map.unions [paramVars, namedParamVars, retVars]
   86     TyRecord _ tm     -> namedTyVars tm
   87     TyVar prov n      -> Map.singleton n (getPos prov)
   88     TyUnifyVar _ _    -> Map.empty
   89 
   90 instance NamedTyVars Schema where
   91   namedTyVars (Forall ns t) = namedTyVars t Map.\\ Map.fromList ns'
   92     where ns' = map (\(pos, n) -> (n, pos)) ns

   96 -- SubstituteTyVars
   97 
   98 --
   99 -- substituteTyVars is a typeclass-polymorphic function for
  100 -- substituting named type variables (such as those declared with
  101 -- typedef) in a Type.
  102 --
  103 -- Panics if we try to substitute a definition that isn't visible.
  104 --
  105 -- Note: substituteTyVars is reused by the interpreter as part of its
  106 -- handling of typedefs during execution as well as by the
  107 -- typechecker.
  108 --
  109 -- XXX: it's not clear that the instances for Maybe and List warrant
  110 -- setting up the typeclass.
  111 --
  112 
  113 class SubstituteTyVars t where
  114   -- | @substituteTyVars m x@ applies the (scoped) map @m@ to type variables in @x@.
  115   substituteTyVars ::
  116       Set PrimitiveLifecycle ->
  117       ScopedMap Name (PrimitiveLifecycle, NamedType) ->
  118       t -> t
  119 
  120 instance (SubstituteTyVars a) => SubstituteTyVars (Maybe a) where
  121   substituteTyVars avail tyenv = fmap (substituteTyVars avail tyenv)
  122 
  123 instance (SubstituteTyVars a) => SubstituteTyVars [a] where
  124   substituteTyVars avail tyenv = map (substituteTyVars avail tyenv)
  125 
  126 instance (SubstituteTyVars a) => SubstituteTyVars (Map k a) where
  127   substituteTyVars avail tyenv m = Map.map (substituteTyVars avail tyenv) m
  128 
  129 instance (SubstituteTyVars a) => SubstituteTyVars (pos, a) where
  130   substituteTyVars avail tyenv (pos, x) = (pos, substituteTyVars avail tyenv x)
  131 
  132 instance SubstituteTyVars Type where
  133   substituteTyVars avail tyenv ty = case ty of
  134     TyCon pos tc ts     -> TyCon pos tc (substituteTyVars avail tyenv ts)
  135     TyFunc pos nameinfo params namedParams ret ->
  136         let params' = substituteTyVars avail tyenv params
  137             namedParams' = substituteTyVars avail tyenv namedParams
  138             ret' = substituteTyVars avail tyenv ret
  139         in
  140         TyFunc pos nameinfo params' namedParams' ret'
  141     TyRecord pos fs     -> TyRecord pos (fmap (substituteTyVars avail tyenv) fs)
  142     TyUnifyVar _ _      -> ty
  143     TyVar _ n           ->
  144         case ScopedMap.lookup n tyenv of
  145             Nothing -> ty
  146             Just (lc, expansion) ->
  147                 if not (Set.member lc avail) then
  148                     panic "substituteTyVars" [
  149                         "Found reference to non-visible typedef: " <> n,
  150                         "Lifecycle setting: " <> Text.pack (show lc)
  151                     ]
  152                 else case expansion of
  153                     AbstractType _kind  -> ty
  154                     ConcreteType ty' -> ty'
  155 
  156 --
  157 -- The prime version uses an ordinary map.
  158 --
  159 -- This is used by the typechecker for the time being until the
  160 -- typechecker gets taught to use ScopedMap.
  161 --
  162 
  163 class SubstituteTyVars' t where
  164   -- | @substituteTyVars' m x@ applies the (ordinary) map @m@ to type variables in @x@.
  165   substituteTyVars' ::
  166       Set PrimitiveLifecycle ->
  167       Map Name (PrimitiveLifecycle, NamedType) ->
  168       t -> t
  169 
  170 instance (SubstituteTyVars' a) => SubstituteTyVars' (Maybe a) where
  171   substituteTyVars' avail tyenv = fmap (substituteTyVars' avail tyenv)
  172 
  173 instance (SubstituteTyVars' a) => SubstituteTyVars' [a] where
  174   substituteTyVars' avail tyenv = map (substituteTyVars' avail tyenv)
  175 
  176 instance (SubstituteTyVars' a) => SubstituteTyVars' (Map k a) where
  177   substituteTyVars' avail tyenv m = Map.map (substituteTyVars' avail tyenv) m
  178 
  179 instance (SubstituteTyVars' a) => SubstituteTyVars' (pos, a) where
  180   substituteTyVars' avail tyenv (pos, x) = (pos, substituteTyVars' avail tyenv x)
  181 
  182 instance SubstituteTyVars' Type where
  183   substituteTyVars' avail tyenv ty = case ty of
  184     TyCon pos tc ts     -> TyCon pos tc (substituteTyVars' avail tyenv ts)
  185     TyFunc pos nameinfo params namedParams ret ->
  186         let params' = substituteTyVars' avail tyenv params
  187             namedParams' = substituteTyVars' avail tyenv namedParams
  188             ret' = substituteTyVars' avail tyenv ret
  189         in
  190         TyFunc pos nameinfo params' namedParams' ret'
  191     TyRecord pos fs     -> TyRecord pos (fmap (substituteTyVars' avail tyenv) fs)
  192     TyUnifyVar _ _      -> ty
  193     TyVar _ n           ->
  194         case Map.lookup n tyenv of
  195             Nothing -> ty
  196             Just (lc, expansion) ->
  197                 if not (Set.member lc avail) then
  198                     panic "substituteTyVars'" [
  199                         "Found reference to non-visible typedef: " <> n,
  200                         "Lifecycle setting: " <> Text.pack (show lc)
  201                     ]
  202                 else case expansion of
  203                     AbstractType _kind -> ty
  204                     ConcreteType ty' -> ty'

  208 -- Deprecation
  209 
  210 isDeprecated :: PrimitiveLifecycle -> Bool
  211 isDeprecated lc = case lc of
  212     Current -> False
  213     WarnDeprecated -> True
  214     HideDeprecated -> True
  215     Experimental -> False
