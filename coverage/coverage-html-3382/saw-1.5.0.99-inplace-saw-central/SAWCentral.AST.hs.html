   95 -- | Position in the life cycle of a primitive.
   96 data PrimitiveLifecycle
   97   = Current         {- ^ Currently available in all modes. -}
   98   | WarnDeprecated  {- ^ Removal planned, available but causes a warning -}
   99   | HideDeprecated  {- ^ Will be removed soon, and available only when
  100                          requested. -}
  101   | Experimental    {- ^ Will be made @Current@ soon, but available only by
  102                          request at the moment. -}
  103   deriving (Eq, Ord, Show, Enum, Bounded)
  104 
  105 -- | Set of all lifecycle values.
  106 everythingAvailable :: Set PrimitiveLifecycle
  107 everythingAvailable = Set.fromList [minBound .. maxBound]
  108 
  109 -- | Default set of lifecycle values.
  110 --   Keep this with its type to make sure it stays current.
  111 defaultAvailable :: Set PrimitiveLifecycle
  112 defaultAvailable = Set.fromList [Current, WarnDeprecated]

  152 -- | Context for type provenance; basically, what kind of program
  153 --   element we were looking at when we inferred a type.
  154 data Tyctx
  155   = TyctxConstant  -- ^ Constants (special cases of expressions)
  156   | TyctxExpr      -- ^ Expressions
  157   | TyctxPat       -- ^ Patterns
  158   | TyctxStmt      -- ^ Statements
  159   | TyctxArgList   -- ^ Expression groups that are function argument lists
  160   deriving (Eq, Show)
  161 
  162 -- | Extended provenance/position information for types.
  163 --
  164 -- TypeExplicit means that the type was written in the input text
  165 --    at the given position.
  166 -- TypeFresh means that the type was generated at the given position
  167 --    to represent a type that that construct implies must exist.
  168 -- TypeFailed means that the type arose from a prior type error and
  169 --    is thus not itself very interesting.
  170 -- TypeFromForallNamed means that the type was forall-bound from the
  171 --    named variable at the given position, with the given name, in
  172 --    the function of the given name.
  173 -- TypeFromForallFresh means that the type was forall-bound from a
  174 --    fresh type variable implicitly generated at the given position,
  175 --    using the given typechecker-generated name, in the function of
  176 --    the given name.
  177 -- TypeFromElement means that the construct at the given position
  178 --    prompted us to choose the accompanying type; e.g. "[]" has type
  179 --    List t.
  180 -- TypeFromContext means that the usage of the construct at the given
  181 --    position prompted us to choose the accompanying type; e.g. in "f
  182 --    x", f has function type.
  183 data TypeProvenance
  184   = TypeExplicit Pos
  185   | TypeFresh Pos
  186   | TypeFailed Pos
  187   | TypeFromForallNamed Pos Text Text
  188   | TypeFromForallFresh Pos Text Text
  189   | TypeFromElement Pos Tyctx
  190   | TypeFromContext Pos Tyctx
  191   | TypeFromFuncWithSig Pos -- ^ Function header with lambda and explicit return type
  192   | TypeFromFuncWithBody Pos Pos -- ^ Function header with lambda and body expression
  193   deriving (Eq, Show)
  194

  198 -- | Type for the hardwired monad types. Note that these days the
  199 --   @LLVMSetup@, @JVMSetup@, and @MIRSetup@ monad types are ordinary

  201 data Context
  202   = ProofScript
  203   | TopLevel
  204   deriving (Eq, Ord, Show)

  223 -- | Information about the named parameters in a function type
  224 --   signature.
  225 --
  226 --   This form preserves the ordering of the names, though not their
  227 --   exact positions, and is used specifically to feed the builtin
  228 --   wrapping logic. (The builtin wrapping logic needs to map the
  229 --   named parameters onto the last N positional arguments of the
  230 --   underlying Haskell function; we must preserve the ordering or
  231 --   utter chaos results.)
  232 --
  233 --   The `Int` argument gives the number of positional arguments (all
  234 --   positional arguments come first in the underlying Haskell
  235 --   functions); the list argument gives the names of the named
  236 --   arguments.
  237 data NamedParamInfo = NamedParamInfo Int [Text]
  238   deriving Show
  239 
  240 -- | Dummy `NamedParamInfo` for use where the information is not
  241 --   needed, such as the typechecker. The only consumer of the
  242 --   `NamedParamInfo` is the builtin wrapper logic, which is
  243 --   downstream only from the type signature parser. Elsewhere,
  244 --   updating or generating the information is not entirely trivial
  245 --   and there is in general no reason to bother.
  246 --
  247 --   FUTURE: a cleaner solution would be to parse to a parse tree
  248 --   first, which would preserve the ordering (and also generally be
  249 --   tidier) then pull the info out in the one case we care about it
  250 --   and discard it otherwise, before lowering to the AST.
  251 noNames :: NamedParamInfo
  252 noNames = NamedParamInfo 0 []
  253 
  254 -- | Allow splicing two `NamedParamInfo` values together with `<>`.
  255 instance Semigroup NamedParamInfo where
  256     NamedParamInfo n1 nps1 <> NamedParamInfo n2 nps2 =
  257         NamedParamInfo (n1 + n2) (nps1 ++ nps2)

  259 -- | Types.
  260 --
  261 -- We carry around provenance for types, which wrap the source
  262 -- positions.
  263 --
  264 -- For types we infer, we want to record not just where but also how
  265 -- the inference happened, so that when we report this to the user
  266 -- they can see what's going on. (For example, if we infer that a type
  267 -- must be a function because it's applied to an argument, we record
  268 -- that it's inferred from context and the position of the context is
  269 -- the position of the term that was applied.) When the type flows
  270 -- around during type inference it carries the position info with it.
  271 --
  272 -- Note that for a non-primitive type the various layers of the type
  273 -- may have totally different provenance. (E.g. we might have List Int
  274 -- where List was inferred from a term "[x]" somewhere but Int came
  275 -- from an explicit annotation somewhere completely different.) So
  276 -- printing this information usefully requires some thought.
  277 --
  278 -- We have the following constraints that it would be nice to encode
  279 -- into the types, except it doesn't seem feasible:
  280 --
  281 --    - Prior to the @generalize@ step in typechecking,
  282 --      `TyVar` should always have `TypeExplicit` provenance, and
  283 --      only `TyUnifyVar` should ever have `TypeFailed` provenance.
  284 --
  285 --    - After @generalize@, `TyVar` can have any provenance, because
  286 --      @generalize@ converts remaining `TyUnifyVar` occurrences to
  287 --      `TyVar` occurrences. This includes `TypeFailed`.
  288 --
  289 --   - Like `TyUnifyVar`, `TypeFailed` should not escape the
  290 --     typechecker.
  291 --
  292 data Type
  293   = TyCon TypeProvenance TyCon [Type]
  294   | TyFunc TypeProvenance NamedParamInfo [Type] (Map Name Type) Type
  295   | TyRecord TypeProvenance (Map Name Type)
  296   | TyVar TypeProvenance Name
  297     -- | For internal typechecker use only.
  298   | TyUnifyVar TypeProvenance TypeIndex
  299   deriving Show
  300 
  301 -- | The positions in type schemes can be either explicit (the user
  302 --   gave a name at this position) or implicit (a fresh unification
  303 --   var was generated at this position and ended up getting forall-
  304 --   bound). XXX: this could use a shorter name.
  305 data SchemaNameProvenance = SchemaNameExplicit Pos | SchemaNameImplicit Pos
  306   deriving Show
  307 
  308 data Schema = Forall [(SchemaNameProvenance, Name)] Type
  309   deriving Show
  310 
  311 -- | A schema pattern is like a schema but has potentially multiple
  312 -- type entries that are meant to match fragments of a complete
  313 -- schema. (We don't, for now at least, need a separate type for type
  314 -- patterns and can just use Type.)
  315 data SchemaPattern = SchemaPattern [(SchemaNameProvenance, Name)] [Type]
  316 
  317 -- | The things a (named) TyVar can refer to by its name.
  318 --
  319 -- A ConcreteType is a direct substitution for the type variable,
  320 -- such as one generated by a typedef statement.
  321 --
  322 -- AbstractType is an opaque type whose only semantics are the
  323 -- operations available for it, if any. The name identifies it; the
  324 -- AbstractType constructor is a placeholder that only serves to
  325 -- carry the kind information.
  326 data NamedType = ConcreteType Type | AbstractType Kind

  330 -- Expressions
  331 
  332 data Expr
  333   -- Constants
  334   = Bool Pos Bool
  335   | String Pos Text
  336   | Int Pos Integer
  337   | Code Pos Text
  338   | CType Pos Text
  339   -- Structures
  340   | Array  Pos [Expr]
  341     -- | A do-block, with zero or more statements and a final expression.
  342     --   The body is a pair so it can be carried around as a single object,
  343     --   which is convenient in a few places.
  344   | Block  Pos ([Stmt], Expr)
  345   | Tuple  Pos [Expr]
  346   | Record Pos (Map Name Expr)
  347   -- Accessors
  348   | Index   Pos Expr Expr
  349   | Lookup  Pos Expr Name
  350   | TLookup Pos Expr Integer
  351   -- LC
  352   | Var Pos Name
  353   -- | All functions are handled as lambdas. We hang onto the name
  354   --   from the function declaration (if there was one) for use in
  355   --   stack traces. The second position (after the name) is the
  356   --   combined position of the whole parameter list. The list of
  357   --   patterns holdes the positional parameters; the map from text
  358   --   holds the optional named parameters. The elements of that map
  359   --   are:
  360   --      - the position of the name text
  361   --      - the overall position
  362   --      - the pattern with the local-facing name (which might be different)
  363   --        and any type
  364   --      - the default value
  365   --   The tuple nesting is supposed to make it possible to remember
  366   --   which position is which: the one belonging to the map key is
  367   --   the closest to it.
  368   | Lambda Pos (Maybe Name) Pos [Pattern] (Map Text (Pos, (Pos, Expr, Pattern))) Expr
  369   -- | Function arguments come with an optional name; if present,
  370   --   it includes the position of the name string.
  371   | Application Pos Expr [(Maybe (Pos, Text), Expr)]
  372   -- Sugar
  373   | Let Pos DeclGroup Expr
  374   | TSig Pos Expr Type
  375   | IfThenElse Pos Expr Expr Expr
  376   deriving Show
  377

  382 -- | Patterns.

  387 data Pattern

  397 -- | Tracking/state type for the @let rebindable@ behavior.
  398 data Rebindable
  399   = RebindableVar -- ^ produced by @let rebindable@
  400   | ReadOnlyVar   -- ^ produced by ordinary @let@ and by @rec@

  403 data Import = Import
  404   { iIsSubmodule :: Bool
  405   , iModule    :: Either FilePath P.ModName
  406   , iAs        :: Maybe P.ModName
  407   , iSpec      :: Maybe P.ImportSpec
  408   , iPos       :: Pos
  409   } deriving Show

  411 -- | Statements.
  412 --
  413 --   In `StmtCode` the first `Pos` is the position of the whole
  414 --   construct (including the initial "let") and the second is the
  415 --   position of just the Cryptol text.
  416 --
  417 --   Similarly, in `StmtTypedef` the first `Pos` is the position of
  418 --   the whole construct (including the expansion) and the second is
  419 --   the position of the name.
  420 --
  421 --   `StmtPushdir` and `StmtPopdir` have no concrete syntax; they are
  422 --   generated by the @include@ handling. See notes in the interpreter
  423 --   for why they're needed.
  424 --
  425 --   The `Bool` in `StmtInclude` is `True` for @include_once@ and
  426 --   `False` for ordinary @include@.
  427 --
  428 data Stmt
  429   = StmtBind     Pos Pattern Expr
  430   | StmtLet      Pos Rebindable DeclGroup
  431   | StmtCode     Pos Pos Text
  432   | StmtImport   Pos Import
  433   | StmtInclude  Pos Text Bool
  434   | StmtTypedef  Pos Pos Text Type
  435   | StmtPushdir  Pos FilePath
  436   | StmtPopdir   Pos
  437   deriving Show

  465 -- Position extraction
  466 
  467 -- | This instance should only be used for cases where we know
  468 --   the position is meaningful on its own.
  469 instance Positioned TypeProvenance where
  470   getPos prov = case prov of
  471       TypeExplicit pos -> pos
  472       TypeFresh pos -> pos
  473       TypeFailed pos -> pos
  474       TypeFromForallNamed pos _ _ -> pos
  475       TypeFromForallFresh pos _ _ -> pos
  476       TypeFromElement pos _ -> pos
  477       TypeFromContext pos _ -> pos
  478       TypeFromFuncWithSig pos -> pos
  479       TypeFromFuncWithBody pos _morepos -> pos
  480 
  481 -- | This is used by the parser where all the provenance is
  482 --   `TypeExplicit`, and should not really be used downstream from
  483 --   there. The locations without the accompanying provenance
  484 --   annotations aren't too meaningful and may be confusing.
  485 --
  486 instance Positioned Type where
  487   getPos ty = case ty of
  488       TyCon prov _ _ -> getPos prov
  489       TyFunc prov _ _ _ _ -> getPos prov
  490       TyRecord prov _ -> getPos prov
  491       TyVar prov _ -> getPos prov
  492       TyUnifyVar prov _ -> getPos prov
  493 
  494 instance Positioned Expr where
  495   getPos (Bool pos _) = pos
  496   getPos (String pos _) = pos
  497   getPos (Int pos _) = pos
  498   getPos (Code pos _) = pos
  499   getPos (CType pos _) = pos
  500   getPos (Array pos _) = pos
  501   getPos (Block pos _) = pos
  502   getPos (Tuple pos _) = pos
  503   getPos (Record pos _) = pos
  504   getPos (Index pos _ _) = pos
  505   getPos (Lookup pos _ _) = pos
  506   getPos (TLookup pos _ _) = pos
  507   getPos (Var pos _) = pos
  508   getPos (Lambda pos _ _ _ _ _) = pos
  509   getPos (Application pos _ _) = pos
  510   getPos (Let pos _ _) = pos
  511   getPos (TSig pos _ _) = pos
  512   getPos (IfThenElse pos _ _ _) = pos
  513 
  514 instance Positioned Pattern where
  515   getPos (PWild pos _) = pos
  516   getPos (PVar fullpos _namepos _ _) = fullpos
  517   getPos (PTuple pos _) = pos
  518 
  519 instance Positioned Import where
  520   getPos = iPos
  521 
  522 instance Positioned Stmt where
  523   getPos (StmtBind pos _ _)  = pos
  524   getPos (StmtLet pos _ _)       = pos
  525   getPos (StmtCode allpos _spos _str) = allpos
  526   getPos (StmtImport pos _)    = pos
  527   getPos (StmtInclude pos _ _)    = pos
  528   getPos (StmtTypedef allpos _apos _a _ty) = allpos
  529   getPos (StmtPushdir pos _) = pos
  530   getPos (StmtPopdir pos) = pos
  531 
  532 instance Positioned DeclGroup where
  533   getPos (Recursive ds) = maxSpan ds
  534   getPos (NonRecursive d) = getPos d
  535 
  536 instance Positioned Decl where
  537   getPos = dPos

  550 ppTyctx :: Tyctx -> Text
  551 ppTyctx ctx = case ctx of
  552     TyctxConstant -> "constant"
  553     TyctxExpr     -> "expression"
  554     TyctxPat      -> "pattern"
  555     TyctxStmt     -> "statement"
  556     TyctxArgList  -> "argument list"

  558 prettyTyctx :: Tyctx -> PP.Doc ann
  559 prettyTyctx ctx = PP.pretty $ ppTyctx ctx

  561 ppContext :: Context -> Text
  562 ppContext c = case c of
  563     ProofScript  -> "ProofScript"
  564     TopLevel     -> "TopLevel"
  565 
  566 -- XXX: currently the typechecker calls this directly; however, it
  567 -- would probably be better if it didn't, at which point we don't
  568 -- need the somewhat unfortunate cases for tuple/array/function and
  569 -- the rest can be folded into prettyType.
  570 prettyTyCon :: TyCon -> PP.Doc ann
  571 prettyTyCon tc = case tc of
  572     TupleCon n     -> PP.parens $ PPS.replicate (n - 1) $ PP.pretty ','
  573     ArrayCon       -> PP.parens $ PP.brackets $ PP.emptyDoc
  574     StringCon      -> "String"
  575     TermCon        -> "Term"
  576     TypeCon        -> "Type"
  577     BoolCon        -> "Bool"
  578     IntCon         -> "Int"
  579     AIGCon         -> "AIG"
  580     CFGCon         -> "CFG"
  581     JVMSpecCon     -> "JVMSpec"
  582     LLVMSpecCon    -> "LLVMSpec"
  583     MIRSpecCon     -> "MIRSpec"
  584     BlockCon       -> "<Block>"
  585     ContextCon cxt -> PP.pretty $ ppContext cxt
  586 
  587 ppTyCon :: PPS.Opts -> TyCon -> Text
  588 ppTyCon ppopts tc = PPS.renderText ppopts $ prettyTyCon tc
  589 
  590 prettyType :: PPS.Opts -> Type -> PPS.Doc
  591 prettyType ppopts = PP.group . visit 0
  592   where
  593     visit :: Int -> Type -> PPS.Doc
  594     visit prec ty0 = case ty0 of
  595       TyCon _ ctor args -> case (ctor, args) of
  596           (TupleCon n, _) ->
  597               if fromIntegral (length args) /= n then
  598                   -- These is no way to produce this state
  599                   croak "tuple" n args
  600               else
  601                   PP.align $ PP.parens $ PP.fillSep $ PP.punctuate "," $ map (visit 0) args
  602           (ArrayCon, [ty1]) ->
  603               PP.brackets $ visit 0 ty1
  604           (BlockCon, [m, arg]) ->
  605               let m' = visit 1 m
  606                   arg' = visit 2 arg
  607                   body = m' <+> arg'
  608               in
  609               if prec > 1 then PP.parens body else body
  610           (ArrayCon, _) -> croak "array" 1 args
  611           (BlockCon, _) -> croak "block" 2 args
  612           (_, _) ->
  613               let ctor' = prettyTyCon ctor in
  614               case args of
  615                   [] -> ctor'
  616                   _ ->
  617                       let ctor'' = PPS.renderText ppopts ctor' in
  618                       croak ctor'' 0 args
  619 
  620       TyFunc _ _ params namedParams ret ->
  621               let params' = map (\p -> visit 1 p <+> "->") params
  622                   oneNamed (n, p) = PP.pretty n <> "?" <> visit 1 p <+> "->"
  623                   namedParams' = map oneNamed $ Map.toList namedParams
  624                   ret' = visit 0 ret
  625                   body = PP.vsep (params' ++ namedParams') <> PP.line <> ret'
  626               in
  627               if prec > 0 then PP.parens (PP.group body) else body
  628       TyRecord _ fields ->
  629           let prettyField (name, ty) =
  630                 let name' = PP.pretty name
  631                     ty' = visit 0 ty
  632                 in
  633                 PP.group $ PPS.prettyTypeSig name' ty'
  634               fields' = map prettyField $ Map.assocs fields
  635               body = PP.sep $ PP.punctuate "," $ fields'
  636               body' = PP.flatAlt (PP.indent 3 body) body
  637           in
  638           PP.braces (PP.line <> body' <> PP.line)
  639 
  640       TyUnifyVar _ i ->
  641           "t." <> PP.pretty i
  642       TyVar _ n ->
  643           PP.pretty n
  644 
  645     croak :: Text -> Integer -> [Type] -> a
  646     croak what n args =
  647         let ppArg arg = "   " <> (PPS.renderText ppopts $ visit 0 arg)
  648             args' = map ppArg args
  649         in
  650         panic "prettyType" $ [
  651             "Malformed " <> what <> " type constructor",
  652             "Expected " <> Text.pack (show n) <> " arguments, found:"
  653         ] ++ args'

  685 prettyExpr :: PPS.Opts -> Expr -> PPS.Doc
  686 prettyExpr ppopts expr0 = case expr0 of
  687     Bool _ b   -> PP.viaShow b
  688     String _ s -> PP.pretty $ PPS.ppStringLiteral s
  689     Int _ i    -> PP.pretty i
  690     Code _ s   -> PP.braces $ PP.braces $ PP.pretty s
  691     CType _ s  -> PP.braces $ "|" <> PP.pretty s <> "|"
  692     Array _ xs ->
  693         PP.brackets $ PP.fillSep $ PP.punctuate "," (map (prettyExpr ppopts) xs)
  694     Block _ (stmts, lastexpr) ->
  695         let stmts' = map (prettyStmt ppopts) stmts
  696             lastexpr' = prettyExpr ppopts lastexpr <> ";"
  697             body = PP.align $ PP.vsep (stmts' ++ [lastexpr'])
  698             -- You would think this could unconditionally be `PP.nest 3
  699             -- body`. But that doesn't work. If you use `PP.nest`,
  700             -- `PP.group` throws away the indentation entirely (whether
  701             -- or not it groups successfully); if you use `PP.indent`
  702             -- instead, it indents when not grouped, but also generates
  703             -- spaces when grouped. Explicit use of `PP.flatAlt` seems
  704             -- to fix this, but ew. And you'd think this would work by
  705             -- default, since folding small blocks to single lines is
  706             -- one of the most basic prettyprinting operations.
  707             body' = PP.flatAlt (PP.indent 3 body) body
  708         in
  709         PP.group $ "do" <+> PP.braces (PP.line <> body' <> PP.line)
  710     Tuple _ exprs ->
  711         PP.parens $ PP.fillSep $ PP.punctuate "," (map (prettyExpr ppopts) exprs)
  712     Record _ members ->
  713         let prettyMember (name, value) =
  714                 PP.pretty name <+> "=" <+> prettyExpr ppopts value
  715             members' = map prettyMember $ Map.assocs members
  716             body = PP.sep $ PP.punctuate PP.comma members'
  717             body' = PP.flatAlt (PP.indent 3 body) body
  718         in
  719         PP.group $ PP.braces (PP.line <> body' <> PP.line)
  720     Index _ _ _ ->
  721         panic "prettyExpr" ["There is no concrete syntax for AST node 'Index'"]
  722     Lookup _ expr name ->
  723         let expr' = prettyExpr ppopts expr
  724             name' = PP.pretty name
  725         in
  726         expr' <> PP.dot <> name'
  727     TLookup _ expr n ->
  728         let expr' = prettyExpr ppopts expr
  729             n' = PP.viaShow n
  730         in      
  731         expr' <> PP.dot <> n'
  732     Var _ name ->
  733         PP.pretty name
  734     Lambda _ _mname _parampos params namedParams expr ->
  735         let onePositional pat =
  736                 let pat' = prettyPattern ppopts pat in
  737                 "\\" <+> pat' <+> "->"
  738             oneNamed (name, (_namepos, (_pos, def, pat))) =
  739                 let name' = PP.pretty name
  740                     def' = prettyExpr ppopts def
  741                     pat' = prettyPattern ppopts pat
  742                 in
  743                 "\\" <+> name' <+> "@" <+> pat' <+> "?=" <> def' <+> "->"
  744             params' = map onePositional params
  745             namedParams' = map oneNamed $ Map.toList namedParams
  746             expr' = prettyExpr ppopts expr
  747         in
  748         let lines_ = params' ++ namedParams' ++ [expr']
  749             -- Now indent each successive line by 3. As elsewhere,
  750             -- this needs to be done using PP.flatAlt or it comes out
  751             -- wrong.
  752             indent line rest =
  753                 PP.group (line <> PP.line <> PP.flatAlt (PP.indent 3 rest) rest)
  754         in
  755         -- This will print the last few arguments and the body
  756         -- together on the last line if they fit, which matches the
  757         -- older behavior. If we decide we don't like that, grouping
  758         -- it again will apparently put each piece on its own line if
  759         -- the whole thing doesn't fit on one.
  760         --
  761         -- Note: if you make changes here you probably want to make
  762         -- matching changes to the Value printer too.
  763         foldr1 indent lines_
  764     Application _ f args ->
  765         -- XXX FIXME: use precedence to minimize parentheses
  766         let f' = prettyExpr ppopts f
  767             once (mbName, arg) =
  768                 let arg' = prettyExpr ppopts arg in
  769                 case mbName of
  770                     Nothing -> arg'
  771                     Just (_pos, name) -> PP.pretty name <> "=" <> arg'
  772             args' = map once args
  773         in
  774         -- XXX: the following baloney is to avoid changing the behavior
  775         -- while doing other much more subtle changes elsewhere and should
  776         -- be simplified later.
  777         --
  778         -- Wrap f' in parens, then f' and the first arg, then that and the
  779         -- second, etc. Except, not the last.
  780         let pairify pieces = case pieces of
  781                 [] -> PP.emptyDoc
  782                 [a] -> a
  783                 a : b : more -> pairify ((PP.parens a <+> b) : more)
  784         in
  785         pairify (f' : args')
  786     Let _ (NonRecursive decl) expr ->
  787         let decl' = prettyDef ppopts decl
  788             expr' = prettyExpr ppopts expr
  789             -- Break after the "in" when it doesn't fit. Maybe I've
  790             -- gotten too used to reading OCaml?
  791             line1 = "let" <+> decl' <+> "in"
  792             line2 = expr'
  793         in
  794         PP.group $ line1 <> PP.line <> line2
  795     Let _ (Recursive decls) expr ->
  796         let decls' = map (prettyDef ppopts) decls
  797             expr' = prettyExpr ppopts expr
  798             decls'' = case decls' of
  799               [] -> []  -- (not actually possible)
  800               first : rest -> ("rec" <+> first) : map (\d -> "and" <+> d) rest
  801         in
  802         PP.vsep decls'' <> PP.hardline <> "in" <> PP.hardline <> PP.nest 3 expr'
  803     TSig _ expr ty ->
  804         let expr' = prettyExpr ppopts expr
  805             ty' = prettyType ppopts ty
  806         in
  807         PP.parens (expr' <+> PP.colon <+> ty')
  808     IfThenElse _ e1 e2 e3 ->
  809         let e1' = prettyExpr ppopts e1
  810             e2' = prettyExpr ppopts e2
  811             e3' = prettyExpr ppopts e3
  812             -- plan for four lines
  813             line1 = "if" <+> e1' <+> "then"
  814             line2 = PP.flatAlt (PP.indent 3 e2') e2'
  815             line3 = "else"
  816             line4 = PP.flatAlt (PP.indent 3 e3') e3'
  817         in
  818         -- Use PP.sep so it'll fold to one line if it fits
  819         PP.group $ PP.sep [line1, line2, line3, line4]
  820

  822 ppExpr ppopts e =
  823     PPS.renderText ppopts $ prettyExpr ppopts e

  825 prettyPattern :: PPS.Opts -> Pattern -> PPS.Doc
  826 prettyPattern ppopts pat =
  827     let prettyArg name' mty = case mty of
  828           Nothing -> name'
  829           Just ty -> PP.parens $ name' <+> PP.colon <+> prettyType ppopts ty
  830     in   
  831     case pat of
  832         PWild _ mty ->
  833           prettyArg "_" mty
  834         PVar _ _ name mty ->
  835           prettyArg (PP.pretty name) mty
  836         PTuple _ pats ->
  837           PP.parens $ PP.fillSep $ PP.punctuate "," $ map (prettyPattern ppopts) pats
  838 
  839 ppPattern :: PPS.Opts -> Pattern -> Text
  840 ppPattern ppopts pat =
  841   PPS.renderText ppopts $ prettyPattern ppopts pat

  843 prettyStmt :: PPS.Opts -> Stmt -> PPS.Doc
  844 prettyStmt ppopts s0 = case s0 of
  845     StmtBind _ (PWild _ _ty) expr ->
  846        -- Drop the _, even if it has an explicit type
  847        prettyExpr ppopts expr <> ";"
  848     StmtBind _ pat expr ->
  849        let pat' = prettyPattern ppopts pat
  850            expr' = prettyExpr ppopts expr
  851            line1 = pat' <+> "<-"
  852            line2 = PP.flatAlt (PP.indent 3 expr') expr'
  853        in
  854        PP.group $ line1 <> PP.line <> line2 <> ";"
  855     StmtLet _ rebindable (NonRecursive decl) ->
  856        let header = case rebindable of
  857              RebindableVar -> "let rebindable"
  858              ReadOnlyVar -> "let"
  859            decl' = prettyDef ppopts decl
  860        in
  861        PP.group $ header <+> decl' <> ";"
  862     StmtLet _ _ (Recursive decls) ->
  863        let decls' = map (prettyDef ppopts) decls
  864            decls'' = case decls' of
  865              [] -> []  -- (not actually possible)
  866              first : rest -> ("rec" <+> first) : map (\d -> "and" <+> d) rest
  867        in
  868        PP.vsep decls'' <> ";"
  869     StmtCode _ _ code ->
  870        let code' = PP.braces $ PP.braces $ PP.pretty code in
  871        "let" <+> code' <> ";"
  872     StmtImport _ imp ->
  873        let prettyNames names =
  874                let prettyIdent name = PP.pretty $ P.identText name
  875                    names' = PP.fillSep $ {- PP.punctuate "," $ -} map prettyIdent names
  876                    long = PP.parens $ PP.line <> PP.indent 3 names' <> PP.line
  877                    short = PP.parens names'
  878                in
  879                PP.flatAlt long short
  880            prettyModName mn =
  881                PP.pretty (intercalate "::" (P.modNameChunks mn))
  882            module' = case iModule imp of
  883                Left filepath -> PP.dquotes $ PP.pretty filepath
  884                Right modName -> prettyModName modName
  885            as' = case iAs imp of
  886                Nothing -> PP.emptyDoc
  887                Just modName -> " as" <+> prettyModName modName
  888            spec' = case iSpec imp of
  889                Nothing -> PP.emptyDoc
  890                Just (P.Hiding names) ->
  891                     " hiding" <+> prettyNames names
  892                Just (P.Only names) ->
  893                     " " <> prettyNames names
  894        in
  895        PP.group $ "import" <+> module' <> as' <> spec' <> ";"
  896     StmtInclude _ name once ->
  897         let inc = if once then "include_once" else "include"
  898             name' = PP.dquotes $ PP.pretty name
  899         in
  900         inc <+> name' <> ";"
  901     StmtTypedef _ _ name ty ->
  902        let name' = PP.pretty name
  903            ty' = prettyType ppopts ty
  904        in
  905        PP.group $ "typedef" <+> name' <+> "=" <+> ty' <> ";"
  906     StmtPushdir _ dir ->
  907        ".pushdir" <+> PP.pretty dir <> ";"
  908     StmtPopdir _ ->
  909        ".popdir;"
  910 
  911 prettyDef :: PPS.Opts -> Decl -> PPS.Doc
  912 prettyDef ppopts (Decl _ pat0 _ def) =
  913    let dissectLambda :: Expr -> ([Pattern], Map Text (Pos, (Pos, Expr, Pattern)), Expr)
  914        dissectLambda e0 = case e0 of
  915           Lambda _pos _name _parampos pats namedpats e1 ->
  916               let (morepats, morenamedpats, e1') = dissectLambda e1 in
  917               (pats ++ morepats, Map.union namedpats morenamedpats, e1')
  918           _ ->
  919               ([], Map.empty, e0)
  920        (params, namedParams, body) = dissectLambda def
  921        params' = map (prettyPattern ppopts) (pat0 : params)
  922        oneNamed (x, (_xpos, (_pos, defExpr, pat))) =
  923            let x' = PP.pretty x
  924                defExpr' = prettyExpr ppopts defExpr
  925                pat' = prettyPattern ppopts pat
  926            in
  927            x' <+> "@" <+> pat' <+> "?=" <> defExpr'
  928        namedParams' = map oneNamed (Map.toList namedParams)
  929        allParams' = PP.align $ PP.sep (params' ++ namedParams')
  930        body' = prettyExpr ppopts body
  931        body'' = PP.flatAlt (PP.indent 3 body') body'
  932    in
  933    allParams' <+> "=" <> PP.line <> body''
  934 
  935 prettyWholeModule :: PPS.Opts -> [Stmt] -> PPS.Doc
  936 prettyWholeModule ppopts stmts =
  937     let stmts' = PP.vsep $ map (prettyStmt ppopts) stmts in
  938     stmts' <> PP.line
