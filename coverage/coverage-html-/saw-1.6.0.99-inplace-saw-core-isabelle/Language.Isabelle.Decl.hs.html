   38 data Decl = 
   39     Definition 
   40       Bool {- ^ qualified -}
   41       Binding.Binding {- ^ const name and type -}
   42       [Binding.Binding] {- ^ argument name/types -}
   43       Expr.Expr {- ^ body -}
   44       [Name.Name] {- ^ extra dependencies -}
   45   | ConstDecl Bool Binding.Binding
   46   | TypeSyn [Expr.Type] Name.Name Expr.Type
   47   | TypeDecl [Expr.Type] Name.Name
   48   | Import Name.TheoryName
   49   | RecordDecl [String]
   50   | DatatypeDecl [Expr.Type] Name.Name [(Name.Name,[Expr.Type])]
   51   | Commented String Decl
   52   | FixpointLocale [Binding.Binding]
   53   | HashDecl 
   54       Name.Name {- ^ const name -}
   55       Int {- ^ deep cryptol hash -}
   56   | NoDecl
   57   deriving (Eq, Ord, Show)
   58 
   59 noName :: Name.Name
   60 noName = Name.SimpleName ""
   61 
   62 instance Name.HasName Decl where
   63   nameOf = \case
   64     Definition _ b _ _ _ -> Name.nameOf b
   65     ConstDecl _ b -> Name.nameOf b
   66     TypeSyn _ nm _ -> nm
   67     TypeDecl _ nm -> nm
   68     Import _ -> noName
   69     RecordDecl _ -> noName
   70     DatatypeDecl _ nm _ -> nm
   71     Commented _ d -> Name.nameOf d
   72     FixpointLocale _ -> noName
   73     HashDecl nm _ -> nm
   74     NoDecl -> noName
   75 
   76   mapName f d = case d of
   77     Definition q b args e deps -> Definition q (Name.mapName f b) args e deps
   78     ConstDecl q b -> ConstDecl q (Name.mapName f b)
   79     TypeSyn ts nm t -> TypeSyn ts (f nm) t
   80     TypeDecl ts nm -> TypeDecl ts (f nm)
   81     Import{} -> d
   82     RecordDecl{} -> d
   83     DatatypeDecl ts nm flds -> DatatypeDecl ts (f nm) flds
   84     Commented msg d' -> Commented msg (Name.mapName f d')
   85     FixpointLocale{} -> d
   86     HashDecl nm h -> HashDecl (f nm) h
   87     NoDecl -> d

   89 newtype Decls = Decls { toList :: [Decl] }
   90   deriving (Semigroup, Monoid, Eq, Ord)

   92 traverseDecls:: Monad m =>  Decls -> (Decl -> m Decl) -> m Decls
   93 traverseDecls (Decls ds) f = do
   94   ds' <- mapM f (List.nub ds)
   95   return $ Decls (List.nub ds')
   96 
   97 traverseImports :: Monad m =>  Decls -> (Name.TheoryName -> m Name.TheoryName) -> m Decls
   98 traverseImports ds f = do
   99   let (imports, Decls ds') = strip (\case Import{} -> True; _ -> False) ds
  100   imports' <- mapM (\case Import nm -> Import <$> f nm; _ -> undefined) (List.nub imports)
  101   return $ Decls (imports' ++ ds')

  103 liftDeclPred :: (Decl -> Bool) -> Decl -> Bool
  104 liftDeclPred f = \case
  105   Commented _ d -> liftDeclPred f d
  106   d -> f d

  108 strip :: (Decl -> Bool) -> Decls -> ([Decl], Decls)
  109 strip f (Decls ds) = let (ds_match, ds') = List.partition (liftDeclPred f) ds in (ds_match, Decls ds')
  110 
  111 filter :: (Decl -> Bool) -> Decls -> [Decl]
  112 filter f (Decls ds ) = List.filter (liftDeclPred f) ds

  114 mapMaybe :: (Decl -> Maybe a) -> Decls -> [a]
  115 mapMaybe f (Decls ds) = Maybe.mapMaybe f ds

  117 singleton :: Decl -> Decls

  138 recordName args =

  144 recordCmd :: Output.HasOutput => [String] -> Template.Template -> String
  145 recordCmd args template =
  146   let
  147     nm = Name.cleanName $ recordName (map Output.out args)
  148     commas = List.intercalate ","
  149     sorts s = commas $ List.replicate (length args + 1) s
  150     (ts, ts') = freshTs (length args + 1)
  151     coercion = mapRecord nm args (\(_,x) -> coerceExpr x)
  152     fields = map (\(anm,t) -> Output.out anm ++ " :: " ++ t) (zip args ts)
  153     coercible s = "(" ++ s ++ "::coercible" ++ ")"
  154   in Output.out $ Template.apply template $ 
  155     [ nm
  156     , commas (map coercible ts)
  157     , commas (map coercible ts')
  158     , commas (take (length args) ts)
  159     , sorts "coercible"
  160     , sorts "strip_type"
  161     , Output.indent 1 $ Output.lines $ fields
  162     , Output.out coercion
  163     , Output.out $ Output.spaces $ Output.addSep "," fields
  164     , sorts "zero"
  165     , Output.out $ recordCtr $ (map (\s -> (Name.SimpleName s, Expr.IntLit 0)) args) ++ [(dots, Expr.IntLit 0)]
  166     , sorts "_"
  167     , commas ts
  168     , sorts "coercible_atom"
  169     ]

  171 collectDeps :: Expr.Expr -> [Name.Name]
  172 collectDeps e = Expr.foldSubExprs (\case Expr.Ctr _ nm -> addNm nm; Expr.Var nm -> addNm nm; _ -> id) e []
  173   where
  174     addNm :: Name.Name -> [Name.Name] -> [Name.Name]
  175     addNm nm nms = (nm:nms)
  176 
  177 dependencies :: Decl -> [Name.Name]
  178 dependencies = \case
  179   Definition _ _ _ e extra -> extra ++ collectDeps e
  180   ConstDecl{} -> []
  181   Commented _ d -> dependencies d
  182   TypeDecl{} -> []
  183   TypeSyn _ _ rhs -> collectDeps rhs
  184   RecordDecl{} -> []
  185   DatatypeDecl targs _ flds -> concat $ map collectDeps $ ((concat $ map (\(_,ts) -> ts) flds) List.\\ targs)
  186   Import{} -> []
  187   FixpointLocale nms -> (map (asRecImpl . Binding.bindName) nms ++ map (asRecSpec . Binding.bindName) nms)
  188   HashDecl nm _ -> [nm]
  189   NoDecl -> []

  191 tyHeader :: Output.HasOutput => [Expr.Expr] -> Name.Name -> String
  192 tyHeader [] nm = Output.out nm
  193 tyHeader es nm = Output.brackets (List.intercalate "," (map Output.out es)) ++ " " ++ Output.out nm

  195 lcp2 :: Eq a => [a] -> [a] -> [a]
  196 lcp2 xs ys = map fst (takeWhile (\(x,y) -> x == y) (zip xs ys))

  198 lcp :: Eq a => [[a]]-> [a]
  199 lcp [] = []
  200 lcp (x:xs) = foldl lcp2 x xs

  208 freshenTVarBinds :: [Binding.Binding] -> [Binding.Binding]
  209 freshenTVarBinds = freshenVars
  210   (\b -> fst $ stripTAbs $ Binding.bindType b)
  211   (\tvars b ->
  212     let
  213       bT = Binding.bindType b
  214       baseT = snd $ stripTAbs bT
  215       tvarsT = map (\nm -> Expr.Var (Name.SimpleName ("'" ++ nm))) tvars
  216     in b { Binding.bindType = tAbs tvarsT baseT })

  218 fixPointLocale :: 
  219  Output.HasOutput => 
  220  [Binding.Binding] ->
  221  [String] ->
  222  Template.Template ->
  223  String
  224 fixPointLocale baseBinds0 path template =
  225   let
  226     baseBinds = freshenTVarBinds baseBinds0
  227     baseNames = map Binding.bindName baseBinds
  228     allvars = concat $ map (fst . stripTAbs . Binding.bindType) baseBinds
  229 
  230     impls = map asRecImpl baseBinds
  231     specs = map asRecSpec baseBinds
  232 
  233     eta_expand b =
  234       let
  235         (tvars,_) = stripTAbs (Binding.bindType b)
  236         args = map (Expr.Var . Name.SimpleName) tvars
  237       in appExpr (Expr.Var (Binding.bindName b)) args
  238 
  239     mkRhs b =
  240       let
  241         (tvars,_) = stripTAbs (Binding.bindType b)
  242         bvs = map (\nm -> Expr.Var (Name.SimpleName (nm ++ "'"))) tvars
  243         lvs = map (Expr.Var . Name.SimpleName) tvars
  244         args = zipWith useSnd lvs bvs
  245       in eAbs bvs (appExpr (Expr.Var (Binding.bindName b)) args)
  246 
  247     asms = map (\(lhs,rhs) -> eqExpr (eta_expand lhs) (eta_expand rhs))
  248       (zip specs impls)
  249     defs = map (\(lhs,rhs) ->
  250       eqExpr (Expr.Var (Name.unqualify (Binding.bindName lhs))) (mkRhs rhs))
  251       (zip baseBinds impls)
  252     def_template = (Output.getTemplate "Fixpoint_Locale_Def")
  253 
  254     def_cmds = map (\(b,e) ->
  255          Output.out $ Template.apply def_template
  256            [Output.out (Binding.bindName b), Output.out e])
  257          (zip impls defs)
  258 
  259     locale_name = localeName baseNames
  260     path_str = if path == [] then "" else outPath path ++ "."
  261 
  262     aliases = 
  263       map (\nm -> "alias " ++ Output.out (Name.unqualify nm) ++ " = "
  264                  ++ path_str
  265                  ++ locale_name ++ "." ++ Output.out (Name.unqualify nm)) baseNames
  266     hides = 
  267       map (\nm -> "hide_fact " ++ path_str ++ locale_name ++ "." ++ Output.out (Name.unqualify nm) ++ "_def") baseNames
  268   in Output.out $ Template.apply template $ 
  269      [ locale_name
  270      , Output.spaces (Output.addSep "and" (map (\e -> Output.quotes (Output.out e)) asms))
  271      , Output.lines def_cmds
  272      , Output.lines aliases
  273      , Output.lines hides
  274      , path_str
  275      , if null allvars then "" else " fixes " ++ Output.spaces (map (Output.out . Name.SimpleName) allvars)
  276      ]

  281 isTNum :: Expr.Type -> Bool
  282 isTNum t = case t of
  283   Expr.Var tnm | Name.TNum <- Name.nmKind tnm -> True
  284   _ -> False

  286 freshenVars :: Traversable t => (a -> [String]) -> ([String] -> a -> a) -> t a -> t a
  287 freshenVars toNms updNames xs_all =
  288   snd (mapAccumL go Set.empty xs_all)
  289   where
  290     mkNm :: String -> Int -> String
  291     mkNm nm i = case i of
  292       0 -> nm
  293       _ -> nm ++ show i
  294 
  295     inner i acc nm =
  296       let nm' = mkNm nm i
  297       in case Set.member nm' acc of
  298         True -> inner (i+1) acc nm
  299         False -> (Set.insert nm' acc, nm')
  300     go acc x =
  301       let
  302         nms = toNms x
  303         (acc', nms') = mapAccumL (inner 0) acc nms
  304       in (acc', updNames nms' x)

  306 freshVars :: [Expr.Type] -> [Expr.Type]
  307 freshVars tvs = drop (length tvs) $ freshenVars
  308   (\case Expr.Var nm -> [Name.nmStr nm]; _ -> [])
  309   (\nms t -> case (nms,t) of ([nm_raw],Expr.Var nm) -> Expr.Var (nm { Name.nmStr = nm_raw }); _ -> t)
  310   (tvs ++ tvs)

  312 mkSameType :: Output.HasOutput => (Expr.Type, Expr.Type) -> String
  313 mkSameType (t1, t2) = Output.brackets $ case isTNum t1 of
  314   True -> "LEN" ++ Output.brackets (Output.out t1) ++ " = " ++ "LEN" ++ Output.brackets (Output.out t2)
  315   False -> "same_type TYPE" ++ Output.brackets (Output.out t1) ++ " TYPE" ++ Output.brackets (Output.out t2)

  317 datatypeDecl :: 
  318  Output.HasOutput => 
  319  [Expr.Type] -> 
  320  Name.Name ->
  321  [(Name.Name,[Expr.Type])] ->
  322  [String] ->
  323  Template.Template ->
  324  String
  325 datatypeDecl targs nm flds0 path template =
  326   let targs' = freshVars targs
  327       flds = map (\(fnm,fts) -> (fnm,map tyargT targs ++ fts)) flds0
  328       defs = map (\(fnm,fts) -> Name.cleanName (Name.unqualify fnm) ++ " " ++ List.intercalate " "
  329         (map (Output.quotes . Output.out) fts)) flds
  330       sorts s = List.intercalate "," $ map (\t -> if isTNum t then "_" else s) targs
  331       commas xs = List.intercalate "," (map Output.out xs)
  332       shownat n = Output.brackets $ show n ++ "::nat"
  333       natcase n = case n of
  334         0 -> "0"
  335         _ -> "Suc (" ++ natcase (n-1) ++ ")"
  336       coercible s = case isTNum s of
  337         True -> Output.out s
  338         False -> "(" ++ Output.out s ++ "::coercible" ++ ")"
  339       mk_comp f n = Output.brackets $ List.intercalate " o " (replicate n f)
  340       mk_strip (n,(_,fts)) = Output.brackets $ case length fts of
  341         0 -> "strip (" ++ shownat n ++ ",())"
  342         1 -> "curry strip " ++ shownat n
  343         _ -> mk_comp "curry" (length fts - 1) ++ (Output.brackets $ "curry strip " ++ shownat n)
  344       mk_unstrip_rhs (fnm,fts) = case length fts of
  345         0 -> "\\<lambda> _. " ++ Output.out fnm
  346         1 -> Output.out fnm ++ " o unstrip"
  347         _ -> mk_comp "uncurry" (length fts - 1) ++ " "
  348                 ++ Output.out fnm ++ " o unstrip"
  349       mk_unstrip (n,r) = natcase n ++ " \\<Rightarrow> " ++ mk_unstrip_rhs r
  350       mk_strip_type tp = case tp of
  351         Expr.Var tnm | Name.TNum <- Name.nmKind tnm -> 
  352           "Tn LEN(" ++ Output.out tnm ++ ")"
  353         _ -> "strip_type TYPE(" ++ Output.out tp ++ ")"
  354       mk_coerce (fnm0,fts) = Output.brackets $
  355         let preamble = case targs of
  356               [] -> ""
  357               _ -> "\\<lambda>" ++ List.intercalate " " (map (\_ -> "_") targs) ++ ". "
  358             fnm = Output.brackets $ List.intercalate " " (Output.out fnm0:map (\_ -> "tyargCtr") targs)
  359         in preamble ++ case length fts of
  360         0 -> fnm
  361         1 -> fnm ++ " o coerce"
  362         _ -> let s = mk_comp "uncurry" (length fts - 1) ++ " " ++ fnm ++ " o coerce"
  363           in  mk_comp "curry" (length fts - 1) ++ " " ++ Output.brackets s
  364 
  365   in Output.out $ Template.apply template $ 
  366     [ Name.cleanName nm
  367     , commas targs
  368     , List.intercalate " | " defs
  369     , commas $ map mk_strip_type targs
  370     , commas targs'
  371     , sorts "coercible"
  372     , List.intercalate " " (map mk_strip (zip [0::Integer ..] flds))
  373     , List.intercalate " | " (map mk_unstrip (zip [0::Integer ..] flds))
  374     , sorts "_"
  375     , sorts "strip_type"
  376     , commas $ map coercible targs
  377     , commas $ map coercible targs'
  378     , List.intercalate " \\<and> " $ map mkSameType (zip targs targs')
  379     , if path == [] then "" else outPath path ++ "."
  380     , List.intercalate " " $ map mk_coerce flds0
  381     , sorts "coercible_atom"
  382     ]

  384 outDecl :: Output.HasOutput => Decl -> [String] -> String
  385 outDecl d qs = case d of
  386   Definition q b args e _ -> definitionCmd q (Name.unqualify b) (Just (args,e))
  387   ConstDecl q b -> definitionCmd q (Name.unqualify b) Nothing
  388   Import i -> Output.out (Name.thyNm i)
  389   TypeSyn targs nm rhs ->
  390       "type_synonym " ++ tyHeader targs (Name.unqualify nm) ++ " = " ++ Output.quotes (Output.out rhs)
  391   TypeDecl targs nm -> "typedecl " ++ tyHeader targs (Name.unqualify nm)
  392   RecordDecl args -> recordCmd args (Output.getTemplate "Coercible_Record")
  393   DatatypeDecl targs nm flds -> datatypeDecl targs nm flds qs (Output.getTemplate "Coercible_Datatype")
  394   FixpointLocale args -> fixPointLocale args qs (Output.getTemplate "Fixpoint_Locale")
  395   Commented msg d' -> "(* " ++ msg ++ "*)\n" ++ Output.out d'
  396   HashDecl nm h -> "declare [[set_const_hash " ++ Output.out nm ++ " 0x" ++ showHex (fromIntegral h :: Word) "" ++ "]]"
  397   NoDecl -> ""
  398 
  399 outPath :: Output.HasOutput => [String] -> String
  400 outPath qs = List.intercalate "." (map (Name.cleanName . Name.SimpleName) qs)

  409 instance Output.Output Decl where
  410   out d = case d of
  411     FixpointLocale args@(nm:_) -> case Name.stripQualifier nm of
  412       (_,[]) -> outDecl d []
  413       (_,qs) -> fixPointLocale args qs (Output.getTemplate "Fixpoint_Locale")
  414     _ -> case Name.stripQualifier d of
  415       (_,[]) -> outDecl d []
  416       (d',qs) -> qualifiedBlock qs $ outDecl d' qs
