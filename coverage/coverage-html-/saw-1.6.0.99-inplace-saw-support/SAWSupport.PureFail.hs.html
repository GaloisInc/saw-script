    1 {-# LANGUAGE OverloadedStrings #-}
    2 {- |
    3 Module      : SAWSupport.PureFail
    4 Description : Failure interface for legacy "pure" code
    5 License     : BSD3
    6 Maintainer  : saw@galois.com
    7 Stability   : provisional
    8 
    9 This module contains an exception and a throw function for it that can
   10 be used in existing pure code to fail without it being a panic.
   11 
   12 Replaces uses of `error`; `error` is specifically undesirable for
   13 several reasons.
   14 
   15 New code should not use this interface.
   16 
   17 Needless to say, actual pure code doesn't throw exceptions. Throwing
   18 exceptions from "pure" code is also fundamentally unsafe: it creates
   19 logic bombs that sit in unevaluated expressions and can move around,
   20 and in particular, can move out from under catches that were intended
   21 to intercept them.
   22 
   23 We are not supposed to be doing it at all in SAW, except for panic
   24 (which is itself a compromise) ... however, there's some code that
   25 needs major reworking to be able to fail properly, and for the moment
   26 removing uses of `error` from it is a higher priority.
   27 
   28 Usage: @pureFail location message@ where
   29 
   30 * @location@ (a `Text`) should say where in the source the call is.
   31   The recommended format is the module name and function, and for
   32   huge functions perhaps also a sub-function or sub-part.
   33 
   34   In principle this information could perhaps be extracted with
   35   `HasCallStack` (e.g. panics do that) but I'm not especially
   36   interested in reducing the friction associated with using this code.
   37 
   38 * @message@ (a `Text`) should be whatever you'd pass to a proper failure
   39   function.
   40 
   41 Does not return.
   42 
   43 The exception constructor is intentionally not exported.
   44 
   45 This module is pure fail and should be removed.
   46 -}
   47 
   48 module SAWSupport.PureFail (MadeOfFail, ppMadeOfFail, pureFail) where
   49 
   50 import Control.Exception (Exception, throw)
   51 import qualified Data.Text as Text
   52 import Data.Text (Text)
   53 
   54 
   55 data MadeOfFail = MadeOfFail Text Text
   56 
   57 ppMadeOfFail :: MadeOfFail -> Text
   58 ppMadeOfFail (MadeOfFail location message) =
   59     "pureFail call at " <> location <> ":\n" <> message
   60 
   61 instance Show MadeOfFail where
   62     show mof = Text.unpack $ ppMadeOfFail mof
   63 
   64 instance Exception MadeOfFail
   65 
   66 pureFail :: Text -> Text -> a
   67 pureFail location message =
   68     throw $ MadeOfFail location message
