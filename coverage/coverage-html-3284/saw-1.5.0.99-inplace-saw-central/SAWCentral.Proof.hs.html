  161 -- | A proposition is a saw-core type of type `Prop`.
  162 -- In particular, this includes any pi type whose result
  163 -- type is a proposition. The argument of a pi type represents
  164 -- a universally quantified variable.
  165 newtype Prop = Prop Term
  166   -- INVARIANT: The type of the given term is `Prop`
  167 
  168 unProp :: Prop -> Term
  169 unProp (Prop tm) = tm

  171 -- | Turn a SAWCore term into a proposition.
  172 -- If it already has type @Prop@, then it is returned as is.
  173 -- If it has a function type of any arity with a return type of
  174 -- @Bool@, then convert the function arguments to universal
  175 -- quantifiers.
  176 termToProp :: SharedContext -> Term -> IO Prop
  177 termToProp sc tm =
  178   do ty <- scWhnf sc =<< scTypeOf sc tm
  179      case asSort ty of
  180        Just s | s == propSort -> pure (Prop tm)
  181        -- If the term is not a Prop, then we check whether it is a
  182        -- function that returns a Bool.
  183        _ ->
  184          case asBoolType (snd (asPiList ty)) of
  185            Nothing ->
  186              do ty' <- ppTerm sc ty
  187                 fail $ unlines $
  188                   [ "Expected type Prop or function type returning Bool."
  189                   , "Found type: " ++ ty'
  190                   ]
  191            Just () ->
  192              predicateToProp sc Universal tm
  193 
  194 -- | Turn a saw-core term into a proposition under the type-as-propositions
  195 --   regime.  The given term must be a type, which means that its own type
  196 --   is a sort.  If it is not, return @Nothing@.
  197 termToMaybeProp :: SharedContext -> Term -> IO (Maybe Prop)
  198 termToMaybeProp sc tm =
  199    do ty <- scWhnf sc =<< scTypeOf sc tm
  200       case asSort ty of
  201         Just s | s == propSort -> return (Just (Prop tm))
  202         _ -> return Nothing
  203 
  204 -- | Turn a boolean-valued saw-core term into a proposition by asserting
  205 --   that it is equal to the true boolean value.  Generalize the proposition
  206 --   by universally quantifying over the variables given in the list.
  207 boolToProp :: SharedContext -> [(VarName, Term)] -> Term -> IO Prop
  208 boolToProp sc vars tm =
  209   do mmap <- scGetModuleMap sc
  210      ty <- scTypeOf sc tm
  211      case evalSharedTerm mmap mempty mempty ty of
  212        TValue VBoolType ->
  213          do p0 <- scEqTrue sc tm
  214             Prop <$> scPiList sc vars p0
  215        _ -> do
  216            tm' <- ppTerm sc tm
  217            ty' <- ppTerm sc ty
  218            fail $ unlines [ "boolToProp: Term is not a boolean", tm', ty' ]

  228 -- | Attempt to split a conjunctive proposition into two propositions.
  229 splitConj :: SharedContext -> Prop -> IO (Maybe (Prop, Prop))
  230 splitConj sc (Prop p) =
  231   do let (vars, body) = asPiList p
  232      case (isGlobalDef "Prelude.and" <@> return <@> return) =<< asEqTrue body of
  233        Nothing -> pure Nothing
  234        Just (_ :*: p1 :*: p2) ->
  235          do t1 <- scPiList sc vars =<< scEqTrue sc p1
  236             t2 <- scPiList sc vars =<< scEqTrue sc p2
  237             return (Just (Prop t1,Prop t2))
  238 
  239 -- | Attempt to split a disjunctive proposition into two propositions.
  240 splitDisj :: SharedContext -> Prop -> IO (Maybe (Prop, Prop))
  241 splitDisj sc (Prop p) =
  242   do let (vars, body) = asPiList p
  243      case (isGlobalDef "Prelude.or" <@> return <@> return) =<< asEqTrue body of
  244        Nothing -> pure Nothing
  245        Just (_ :*: p1 :*: p2) ->
  246          do t1 <- scPiList sc vars =<< scEqTrue sc p1
  247             t2 <- scPiList sc vars =<< scEqTrue sc p2
  248             return (Just (Prop t1,Prop t2))

  250 -- | Unfold all the constants appearing in the proposition
  251 --   whose VarIndex is found in the given set.
  252 unfoldProp :: SharedContext -> Set VarIndex -> Prop -> IO Prop
  253 unfoldProp sc unints (Prop tm) =
  254   do let unfold nm = Set.member (nameIndex nm) unints
  255      tm' <- scUnfoldConstants sc unfold tm
  256      return (Prop tm')
  257 
  258 -- | Unfold one time all the fixpoint constants appearing in the proposition
  259 -- whose VarIndex is found in the given set.
  260 unfoldFixOnceProp :: SharedContext -> Set VarIndex -> Prop -> IO Prop
  261 unfoldFixOnceProp sc unints (Prop tm) =
  262   do tm' <- scUnfoldOnceFixConstantSet sc True unints tm
  263      return (Prop tm')

  265 -- | Rewrite the proposition using the provided Simpset
  266 simplifyProp :: Monoid a => SharedContext -> Simpset a -> Prop -> IO (a, Prop)
  267 simplifyProp sc ss (Prop tm) =
  268   do (a, tm') <- rewriteSharedTerm sc ss tm
  269      return (a, Prop tm')

  271 hoistIfsInProp :: SharedContext -> Prop -> IO Prop
  272 hoistIfsInProp sc (Prop p) = do
  273   let (vars, body) = asPiList p
  274   body' <-
  275     case asEqTrue body of
  276       Just t -> pure t
  277       Nothing -> fail "hoistIfsInProp: expected EqTrue"
  278   t1 <- hoistIfs sc body'
  279   t2 <- scEqTrue sc t1
  280   t3 <- scPiList sc vars t2
  281   return (Prop t3)

  283 -- | Evaluate the given proposition by round-tripping
  284 --   through the What4 formula representation.  This will
  285 --   perform a variety of simplifications and rewrites.
  286 evalProp :: SharedContext -> Bool -> Set VarIndex -> Prop -> IO Prop
  287 evalProp sc what4PushMuxOps unints p =
  288   do let (vars, body) = asPiList (unProp p)
  289      body' <-
  290        case asEqTrue body of
  291          Just t -> pure t
  292          Nothing -> do
  293            p' <- ppTerm sc (unProp p)
  294            fail $ "goal_eval: expected EqTrue\n" ++ p'
  295 
  296      eb <- W4Sim.newSAWCoreExprBuilder sc what4PushMuxOps
  297      let st = eb ^. W4.userState
  298      (_names, (_mlabels, p')) <- W4Sim.w4Eval eb st sc mempty unints body'
  299      t1 <- W4Sim.toSC eb st p'
  300      t2 <- scEqTrue sc t1
  301      -- turn the free variables we generated back into pi-bound variables
  302      t3 <- scPiList sc vars t2
  303      return (Prop t3)

  319 trivialProofTerm :: SharedContext -> Prop -> IO (Either String Term)
  320 trivialProofTerm sc (Prop p) = runExceptT (loop =<< lift (scWhnf sc p))
  321   where
  322     loop t =
  323       case asPi t of
  324         Just (nm, tp, body) ->
  325           do pf <- loop =<< lift (scWhnf sc body)
  326              lift $ scLambda sc nm tp pf
  327         Nothing ->
  328           case asEq t of
  329             Just (tp, x, _y) ->
  330               -- NB, we don't check if x is convertable to y here, as that will
  331               -- be done later in @tacticTrivial@ during the type-checking step
  332               lift $ scGlobalApply sc "Prelude.Refl" [tp, x]
  333             Nothing -> do
  334               p' <- lift $ ppTerm sc p
  335               throwError $ unlines
  336                 [ "The trivial tactic can only prove quantified equalities, but"
  337                 , "the given goal is not in the correct form."
  338                 , p'
  339                 ]

  392 -- | Check that a @Prop@ value is actually a proposition.
  393 --   This check should always succeed, unless some programming
  394 --   mistake has allowed us to build an ill-typed Prop.
  395 checkProp :: SharedContext -> Prop -> IO ()
  396 checkProp sc (Prop t) =
  397   do ty <- scTypeOf sc t
  398      case asSort ty of
  399         Just s | s == propSort -> return ()
  400         _ -> do
  401           t' <- ppTerm sc t
  402           ty' <- ppTerm sc ty
  403           fail $ unlines ["Term is not a prop!", t', ty']

  417 -- | A theorem is a proposition which has been wrapped in a
  418 --   constructor indicating that it has already been proved,

  435 -- | A theorem database is intended to track theorems that may be used
  436 --   in the proof of later theorems or verification conditions. This is
  437 --   ultimately used to produce verification summaries, which capture
  438 --   the dependency graph between theorems and verifications.
  439 data TheoremDB =
  440   TheoremDB
  441   -- TODO, maybe this should be a summary or something simpler?
  442   { theoremMap :: Map TheoremNonce Theorem
  443   }
  444 
  445 emptyTheoremDB :: TheoremDB
  446 emptyTheoremDB = TheoremDB mempty
  447 
  448 recordTheorem :: TheoremDB -> Theorem -> TheoremDB
  449 recordTheorem db thm@Theorem{ _thmNonce = n } = TheoremDB (Map.insert n thm (theoremMap db))

  451 -- | Given a set of root values, find all the theorems in this database
  452 --   that are transitively used in the proofs of those theorems.
  453 --   This function will panic if any of the roots or reachable theorems
  454 --   are not found in the database.
  455 reachableTheorems :: TheoremDB -> Set TheoremNonce -> Map TheoremNonce Theorem
  456 reachableTheorems db roots = Set.foldl' (loop (theoremMap db)) mempty roots
  457 
  458  where
  459    loop m visited curr
  460      | Just _ <- Map.lookup curr visited = visited
  461 
  462      | Just thm <- Map.lookup curr m =
  463          Set.foldl' (loop m)
  464             (Map.insert curr thm visited)
  465             (thmDepends thm)
  466 
  467      | otherwise =
  468          let curr' = Text.pack (show (indexValue curr))
  469              m' = map (\(k, _v) -> Text.pack (show k)) $ Map.toList m
  470          in
  471          panic "reachableTheorems" [
  472              "Could not find theorem with identifier: " <> curr',
  473              "Theorems in database: " <> Text.intercalate " " m'
  474          ]

  477 -- | Check that the purported theorem is valid.
  478 --
  479 --   This validates that the included evidence actually supports
  480 --   the proposition.  Note, however, this validation procedure
  481 --   does not totally guarantee the theorem is true, as it does
  482 --   not verify any solver-provided proofs, and it accepts admitted
  483 --   propositions and quickchecked propositions as valid.
  484 validateTheorem :: SharedContext -> Bool -> TheoremDB -> Theorem -> IO ()
  485 
  486 validateTheorem sc what4PushMuxOps db Theorem{ _thmProp = p, _thmEvidence = e, _thmDepends = thmDep } =
  487    do let knownThms = Map.keysSet (theoremMap db)
  488       (deps, _, hyps) <- checkEvidence sc what4PushMuxOps e p
  489       unless (Set.isSubsetOf deps thmDep && Set.isSubsetOf thmDep knownThms)
  490              (fail $ unlines ["Theorem failed to declare its dependencies correctly"
  491                              , show deps, show thmDep ])
  492       ppHyps <- traverse (ppTerm sc) (HashSet.toList hyps)
  493       unless (HashSet.null hyps) $
  494         fail $ unlines $ ["Theorem depends on undischarged hypotheses:"] ++ ppHyps

  496 data TheoremSummary
  497   = AdmittedTheorem Text
  498   | TestedTheorem Integer
  499   | ProvedTheorem SolverStats

  501 instance Monoid TheoremSummary where
  502   mempty = ProvedTheorem mempty

  504 instance Semigroup TheoremSummary where
  505   AdmittedTheorem msg <> _ = AdmittedTheorem msg
  506   _ <> AdmittedTheorem msg = AdmittedTheorem msg
  507   TestedTheorem x <> TestedTheorem y = TestedTheorem (min x y)
  508   TestedTheorem x <> _ = TestedTheorem x
  509   _ <> TestedTheorem y = TestedTheorem y
  510   ProvedTheorem s1 <> ProvedTheorem s2 = ProvedTheorem (s1<>s2)

  525 -- | This datatype records evidence for the truth of a proposition.
  526 data Evidence
  527   = -- | The given term provides a direct programs-as-proofs witness
  528     --   for the truth of its type (qua proposition). This will
  529     --   succeed when applied to sequent with a conclusion focus whose
  530     --   statement matches the type of the given term.
  531     ProofTerm !Term
  532 
  533     -- | This type of evidence is produced when the given goal
  534     --   has been dispatched to a solver which has indicated that it
  535     --   was able to prove the goal. The included @SolverStats@
  536     --   give some details about the solver run.
  537   | SolverEvidence !SolverStats !Prop
  538 
  539     -- | This type of evidence is produced when the given goal
  540     --   has been randomly tested against input vectors in the style
  541     --   of quickcheck. The included number is the number of successfully
  542     --   passed test vectors.
  543   | QuickcheckEvidence !Integer !Prop
  544 
  545     -- | This type of evidence is produced when the given goal
  546     --   has been explicitly assumed without other evidence, at the
  547     --   user's direction.
  548   | Admitted !Text !Pos !Prop
  549 
  550     -- | This type of evidence is produced when a previously-proved
  551     -- theorem is applied via backward reasoning to prove a goal.
  552     -- Pi-quantified variables of the theorem may be specialized
  553     -- either by giving an explicit @Term@ to instantiate the
  554     -- variable, or by giving @Evidence@ for @Prop@ hypotheses.
  555     -- After specializing the given @Theorem@ the result must match
  556     -- the current goal.
  557   | ApplyEvidence !Theorem ![Either Term Evidence]
  558 
  559     -- | This type of evidence is used to prove a universally-
  560     -- quantified goal.
  561     -- The included 'VarName' should be a fresh variable used to
  562     -- instantiate the quantified goal.
  563   | IntroEvidence !VarName !Term !Evidence
  564 
  565     -- | This type of evidence is used to modify a goal to prove via
  566     -- rewriting.
  567     -- The property is rewritten by the given simpset; then the
  568     -- provided evidence is used to check the modified goal.
  569   | RewriteEvidence !(Simpset TheoremAnnotation) !Evidence
  570 
  571     -- | This type of evidence is used to modify a goal via unfolding
  572     -- constant definitions.
  573     -- The goal is modified by unfolding constants identified via the
  574     -- given set of @VarIndex@; then the provided evidence is used to
  575     -- check the modified goal.
  576   | UnfoldEvidence !(Set VarIndex) !Evidence
  577 
  578     -- | This type of evidence is used to modify a goal via unfolding
  579     -- fixpoint constant definitions once.
  580     -- The goal is modified by unfolding constants identified via the
  581     -- given set of @VarIndex@; then the provided evidence is used to
  582     -- check the modified goal.
  583   | UnfoldFixOnceEvidence !(Set VarIndex) !Evidence
  584 
  585     -- | This type of evidence is used to modify a goal via evaluation
  586     -- into the the What4 formula representation.
  587     -- During evaluation, the constants identified by the given set of
  588     -- @VarIndex@ are held uninterpreted (i.e., will not be unfolded).
  589     -- Then, the provided evidence is use to check the modified goal.
  590   | EvalEvidence !(Set VarIndex) !Evidence
  591 
  592     -- | This type of evidence is used to modify a goal.
  593     -- The modified goal should be equivalent up to conversion.
  594   | ConversionEvidence !Prop !Evidence
  595 
  596     -- | This type of evidence is used to modify a goal to prove by applying
  597     -- 'hoistIfsInProp'.
  598   | HoistIfsEvidence !Evidence
  599 
  600     -- | Modify the goal by invoking the term evaluator on it.
  601     -- Treat the given variable indexes as opaque.
  602   | NormalizePropEvidence !(Set VarIndex) !Evidence

  857 -- | Convert a term with a function type of any arity into a pi type.
  858 -- Negate the term if the result type is @Bool@ and the quantification
  859 -- is 'Existential'.
  860 predicateToProp :: SharedContext -> Quantification -> Term -> IO Prop
  861 predicateToProp sc quant = loop
  862   where
  863   loop t =
  864     case asLambda t of
  865       Just (x, ty, body) ->
  866         do Prop body' <- loop body
  867            Prop <$> scPi sc x ty body'
  868       Nothing ->
  869         do (argTs, resT) <- asPiList <$> scTypeOf sc t
  870            let toPi env [] t0 =
  871                  case asBoolType resT of
  872                    Nothing -> do
  873                        resT' <- ppTerm sc resT
  874                        fail $ unlines [
  875                            "predicateToProp : Expected boolean result type but got",
  876                            resT'
  877                         ]
  878                    Just () ->
  879                      case quant of
  880                        Universal -> scEqTrue sc =<< scInstantiate sc env t0
  881                        Existential -> scEqTrue sc =<< scNot sc =<< scInstantiate sc env t0
  882                toPi env ((x, xT) : tys) t1 =
  883                  do x' <- scFreshVarName sc (vnName x)
  884                     xT' <- scInstantiate sc env xT
  885                     v <- scVariable sc x' xT'
  886                     t2 <- scApply sc t1 v
  887                     t3 <- toPi (IntMap.insert (vnIndex x) v env) tys t2
  888                     scPi sc x' xT' t3
  889            Prop <$> toPi IntMap.empty argTs t

  913 -- forall x in ps1, exists y in ps2 where x == y
  914 propsSubset :: SharedContext -> [Prop] -> [Prop] -> IO Bool
  915 propsSubset sc ps1 ps2 =
  916   -- For each x, check if x exists in ps2 by checking term identity using stAppIndex
  917   -- If the check succeeds, return True. Otherwise use propsElem to do the more expensive
  918   -- convertibility check.
  919   and <$> sequence [ if idSubset (unProp x) then pure True else propsElem sc x ps2 | x <- ps1 ]
  920   where
  921     ps2Ids = foldr (\x ids -> Set.insert (termIndex (unProp x)) ids) Set.empty ps2
  922     idSubset t = Set.member (termIndex t) ps2Ids
  923 
  924 -- exists y in ps where x == y
  925 propsElem :: SharedContext -> Prop -> [Prop] -> IO Bool
  926 propsElem sc x ps =
  927   or <$> sequence [ scConvertible sc (unProp x) (unProp y) | y <- ps ]

  929 -- | Tests that the first prop subsumes the second.
  930 -- This is a shallow syntactic check that is sufficient to show that a proof
  931 -- of the first prop is sufficient to prove the second.
  932 propSubsumes :: SharedContext -> Prop -> Prop -> IO Bool
  933 propSubsumes sc g1 g2 =
  934   propsSubset sc [g2] [g1] -- proves no *less*

  936 -- | Verify that the given evidence in fact supports the given proposition.
  937 --   Returns the identifiers of all the theorems depended on while checking evidence
  938 --   Also return the context of all types assumed to be inhabited.
  939 checkEvidence ::
  940   SharedContext -> Bool -> Evidence -> Prop ->
  941   IO (Set TheoremNonce, TheoremSummary, Hypotheses)
  942 checkEvidence sc what4PushMuxOps = \e p -> do
  943                               nenv <- scGetNamingEnv sc
  944                               check nenv e p
  945 
  946   where
  947     checkApply _nenv (Prop p) [] = return (mempty, mempty, p, termHypotheses p)
  948 
  949     -- Check a theorem applied to "Evidence".
  950     -- The given prop must be an implication
  951     -- (i.e., nondependent Pi quantifying over a Prop)
  952     -- and the given evidence must match the expected prop.
  953     checkApply nenv (Prop p) (Right e:es)
  954       | Just (lnm, tp, body) <- asPi p
  955       , IntSet.notMember (vnIndex lnm) (freeVars body)
  956       = do (d1, sy1, hyps1) <- check nenv e =<< termToProp sc tp
  957            (d2, sy2, p', hyps2) <- checkApply nenv (Prop body) es
  958            return (Set.union d1 d2, sy1 <> sy2, p', hyps1 <> hyps2)
  959       | otherwise = do
  960            p' <- ppTerm sc p
  961            fail $ unlines [
  962                "Apply evidence mismatch: non-function or dependent function",
  963                p'
  964             ]
  965 
  966     -- Check a theorem applied to a term. This explicitly instantiates
  967     -- a Pi binder with the given term.
  968     checkApply nenv (Prop p) (Left tm:es) =
  969       do p1 <- reducePi sc p tm
  970          (deps, sy, p2, hyps) <- checkApply nenv (Prop p1) es
  971          pure (deps, sy, p2, hyps <> termHypotheses tm)
  972 
  973     check ::
  974       DisplayNameEnv ->
  975       Evidence ->
  976       Prop ->
  977       IO (Set TheoremNonce, TheoremSummary, HashSet Term)
  978     check nenv e sqt = case e of
  979       ProofTerm tm ->
  980         case sqt of
  981           Prop ptm ->
  982             do ty <- scTypeOf sc tm
  983                ok <- scConvertible sc ptm ty
  984                unless ok $ do
  985                    ptm' <- ppTerm sc ptm
  986                    tm' <- ppTerm sc tm
  987                    fail $ unlines [
  988                        "Proof term does not prove the required proposition",
  989                        ptm',
  990                        tm'
  991                     ]
  992                return (mempty, ProvedTheorem mempty, termHypotheses tm)
  993 
  994       SolverEvidence stats sqt' ->
  995         do ok <- propSubsumes sc sqt' sqt
  996            unless ok $ do
  997              ppopts <- scGetPPOpts sc
  998              fail $ PPS.render ppopts $ PP.vsep
  999                [ "Solver proof does not prove the required sequent"
 1000                , prettyProp ppopts nenv sqt
 1001                , prettyProp ppopts nenv sqt'
 1002                ]
 1003            return (mempty, ProvedTheorem stats, propHypotheses sqt')
 1004 
 1005       Admitted msg pos sqt' ->
 1006         do ok <- propSubsumes sc sqt' sqt
 1007            unless ok $ do
 1008              ppopts <- scGetPPOpts sc
 1009              let pos' = prettyPosition pos
 1010              fail $ PPS.render ppopts $ PP.vsep
 1011                [ "Admitted proof does not match the required sequent" <+> pos'
 1012                , pretty msg
 1013                , prettyProp ppopts nenv sqt
 1014                , prettyProp ppopts nenv sqt'
 1015                ]
 1016            return (mempty, AdmittedTheorem msg, propHypotheses sqt')
 1017 
 1018       QuickcheckEvidence n sqt' ->
 1019         do ok <- propSubsumes sc sqt' sqt
 1020            unless ok $ do
 1021              ppopts <- scGetPPOpts sc
 1022              fail $ PPS.render ppopts $ PP.vsep
 1023                [ "Quickcheck evidence does not match the required sequent"
 1024                , prettyProp ppopts nenv sqt
 1025                , prettyProp ppopts nenv sqt'
 1026                ]
 1027            return (mempty, TestedTheorem n, propHypotheses sqt')
 1028 
 1029       ApplyEvidence thm es ->
 1030         case sqt of
 1031           p ->
 1032             do (d, sy, p', hyps) <- checkApply nenv (thmProp thm) es
 1033                ok <- scConvertible sc (unProp p) p'
 1034                unless ok $ do
 1035                    sp <- ppTerm sc (unProp p)
 1036                    sp' <- ppTerm sc p'
 1037                    fail $ unlines [
 1038                        "Apply evidence does not match the required proposition",
 1039                        sp,
 1040                        sp'
 1041                     ]
 1042                return (Set.insert (thmNonce thm) d, sy, thmHyps thm <> hyps)
 1043 
 1044       UnfoldEvidence vars e' ->
 1045         do sqt' <- unfoldProp sc vars sqt
 1046            check nenv e' sqt'
 1047 
 1048       UnfoldFixOnceEvidence vars e' ->
 1049         do sqt' <- unfoldFixOnceProp sc vars sqt
 1050            check nenv e' sqt'
 1051 
 1052       NormalizePropEvidence opqueSet e' ->
 1053         do sqt' <- normalizeProp sc opqueSet sqt
 1054            check nenv e' sqt'
 1055 
 1056       RewriteEvidence ss e' ->
 1057         do (TheoremAnnotation d1 h1 s1, sqt') <- simplifyProp sc ss sqt
 1058            (d2, s2, h2) <- check nenv e' sqt'
 1059            return (d1 <> d2, s1 <> s2, h1 <> h2)
 1060 
 1061       HoistIfsEvidence e' ->
 1062         do sqt' <- hoistIfsInProp sc sqt
 1063            check nenv e' sqt'
 1064 
 1065       EvalEvidence vars e' ->
 1066         do sqt' <- evalProp sc what4PushMuxOps vars sqt
 1067            check nenv e' sqt'
 1068 
 1069       ConversionEvidence sqt' e' ->
 1070         do ok <- convertibleProps sc sqt sqt'
 1071            unless ok $ do
 1072                ppopts <- scGetPPOpts sc
 1073                fail $ PPS.render ppopts $ PP.vsep [
 1074                    "Converted sequent does not match goal",
 1075                    prettyProp ppopts nenv sqt,
 1076                    prettyProp ppopts nenv sqt'
 1077                 ]
 1078            check nenv e' sqt'
 1079 
 1080       IntroEvidence x xty e' ->
 1081         -- TODO! Check that the given VarName is fresh for the sequent.
 1082         --
 1083         --   On soundness: I am concerned that just checking that 'x' is fresh for 'sqt'
 1084         --   isn't enough, as 'x' may nonetheless appear in other values in the ambient
 1085         --   context, such as defined constants, or in the type of other things, etc.
 1086         --
 1087         --   The most reliable way to actually do this freshness check, then, is to produce
 1088         --   a brand-new guaranteed fresh value (call it 'y') and replace 'x' with 'y'
 1089         --   everywhere in the remaining evidence checking process. This is going to require
 1090         --   quite a bit of additional infrastructure to do the necessary replacements, and we
 1091         --   will need to be pretty careful if we want to avoid repeated traversals (which
 1092         --   could cause substantial performance issues).
 1093         case sqt of
 1094           Prop ptm ->
 1095             case asPi ptm of
 1096               Nothing -> do
 1097                   ptm' <- ppTerm sc ptm
 1098                   fail $ unlines ["Intro evidence expected function prop", ptm']
 1099               Just (nm, ty, body) ->
 1100                 do ok <- scConvertible sc ty xty
 1101                    unless ok $ do
 1102                        xty' <- ppTerm sc xty
 1103                        ty' <- ppTerm sc ty
 1104                        fail $ unlines [
 1105                            "Intro evidence types do not match",
 1106                            xty',
 1107                            ty'
 1108                         ]
 1109                    x' <- scVariable sc x xty
 1110                    body' <- scInstantiate sc (IntMap.singleton (vnIndex nm) x') body
 1111                    (deps, sy, hyps) <- check nenv e' (Prop body')
 1112                    let hyps' = HashSet.delete xty hyps
 1113                    pure (deps, sy, hyps')

 1136 -- | Attempt to complete a proof by checking that all subgoals have been discharged,
 1137 --   and validate the computed evidence to ensure that it supports the original
 1138 --   proposition.  If successful, return the completed @Theorem@ and a summary
 1139 --   of solver resources used in the proof.
 1140 --
 1141 --   If first boolean argument is @False@, the resulting theorem will not be
 1142 --   recorded in the theorem database. This should only be done when you are
 1143 --   sure that the theorem will not be used as part of the proof of other theorems,
 1144 --   or later steps will fail. This is intended for proofs of verification conditions,
 1145 --   which are not exposed for reuse, and where it requires a significant memory
 1146 --   burden to record them. In particular commands like @llvm_verify@, @jvm_verify@, etc
 1147 --   that produce and verify verification conditions should set this argument to
 1148 --   @False@ to reduce memory pressure.
 1149 --
 1150 --   The final boolean argument indicates if the proof state needs a sequent normalization
 1151 --   step as the final step in its evidence chain to check.  This is useful for goals that
 1152 --   start with a nontrivial sequent (e.g., when enable_sequent_goals is turned on). For some
 1153 --   goals, this step is expensive, so we avoid it unless necessary.
 1154 finishProof ::
 1155   SharedContext ->
 1156   TheoremDB ->
 1157   Prop ->
 1158   ProofState ->
 1159   Bool {- ^ should we record the theorem in the database? -} ->
 1160   Bool {- ^ If 'True', push certain @ExprBuilder@ operations (e.g., @zext@) down
 1161             to the branches of @ite@ expressions -} ->
 1162   IO (ProofResult, TheoremDB)
 1163 finishProof sc db conclProp
 1164     ps@(ProofState gs (_concl,loc,ploc,rsn) stats _ checkEv start)
 1165     recordThm what4PushMuxOps =
 1166   case gs of
 1167     [] ->
 1168       do e <- checkEv []
 1169          (deps, sy, hyps) <- checkEvidence sc what4PushMuxOps e conclProp
 1170          -- Fail if hyps includes any types that do not correspond to a
 1171          -- free variable in the conclusion
 1172          let extraHyps = HashSet.difference hyps (propHypotheses conclProp)
 1173          ppHyps <- traverse (ppTerm sc) (HashSet.toList extraHyps)
 1174          unless (HashSet.null extraHyps) $
 1175            fail $ unlines $ ["Theorem depends on undischarged hypotheses:"] ++ ppHyps
 1176          n <- freshNonce globalNonceGenerator
 1177          end <- getCurrentTime
 1178          let theorem =
 1179                    Theorem
 1180                    { _thmProp = conclProp
 1181                    , _thmHyps = hyps
 1182                    , _thmStats = stats
 1183                    , _thmEvidence = e
 1184                    , _thmLocation = loc
 1185                    , _thmProgramLoc = ploc
 1186                    , _thmReason = rsn
 1187                    , _thmNonce = n
 1188                    , _thmDepends = deps
 1189                    , _thmElapsedTime = diffUTCTime end start
 1190                    , _thmSummary = sy
 1191                    }
 1192          let db' = if recordThm then recordTheorem db theorem else db
 1193          pure (ValidProof stats theorem, db')
 1194     _ : _ ->
 1195          pure (UnfinishedProof ps, db)

 1261 -- | A @Tactic@ is a computation that examines, simplifies
 1262 --   and/or solves a proof goal.  Given a goal, it does some
 1263 --   work and returns 0 or more subgoals which, if they are all proved,
 1264 --   imply the original goal.  Moreover, it returns a way to compute
 1265 --   evidence for the original goal when given evidence for the generated
 1266 --   subgoal.  An important special case is a tactic that returns 0 subgoals,
 1267 --   and therefore completely solves the goal.
 1268 newtype Tactic m a =
 1269   Tactic (ProofGoal -> ExceptT (SolverStats, CEX) m (a, SolverStats, [ProofGoal], [Evidence] -> IO Evidence))

 1271 -- | Choose the first subgoal in the current proof state and apply the given
 1272 --   proof tactic.
 1273 withFirstGoal :: F.MonadFail m => Tactic m a -> ProofState -> m (Either (SolverStats, CEX) (a, ProofState))
 1274 withFirstGoal (Tactic f) (ProofState goals concl stats timeout evidenceCont start) =
 1275      case goals of
 1276        [] -> fail "ProofScript failed: no subgoal"
 1277        g : gs -> runExceptT (f g) >>= \case
 1278          Left cex -> return (Left cex)
 1279          Right (x, stats', gs', buildTacticEvidence) ->
 1280            do let evidenceCont' es =
 1281                       do let (es1, es2) = splitAt (length gs') es
 1282                          e <- buildTacticEvidence es1
 1283                          evidenceCont (e:es2)
 1284               let ps' = ProofState (gs' <> gs) concl (stats <> stats') timeout evidenceCont' start
 1285               seq ps' (return (Right (x, ps')))

 1291        (finalVars, tm') <- processTerm mmap initVars tm0

 1422 -- | Given a prop to prove, attempt to apply another given proposition, producing
 1423 --   new subgoals for any necessary hypotheses of the proposition.  Returns
 1424 --   @Nothing@ if the given proposition does not apply to the goal.
 1425 propApply ::
 1426   SharedContext ->
 1427   Prop {- ^ propsition to apply (usually a quantified and/or implication term) -} ->
 1428   Prop {- ^ goal to apply the proposition to -} ->
 1429   IO (Maybe [Either Term Prop])
 1430 propApply sc rule goal = applyFirst (asPiLists (unProp rule))
 1431   where
 1432     applyFirst :: [([(VarName, Term)], Term)] -> IO (Maybe [Either Term Prop])
 1433     applyFirst [] = pure Nothing
 1434     applyFirst ((ruleArgs, ruleConcl) : rest) =
 1435       do result <- scMatch sc ruleArgs ruleConcl (unProp goal)
 1436          case result of
 1437            Nothing -> applyFirst rest
 1438            Just inst ->
 1439              do let mkNewGoal :: (VarName, Term) -> IO (Either Term Prop)
 1440                     mkNewGoal (vn, tp) =
 1441                       case IntMap.lookup (vnIndex vn) inst of
 1442                         Nothing ->
 1443                           -- this argument not solved by unification, so make it a goal
 1444                           do c0 <- scInstantiate sc inst tp
 1445                              mp <- termToMaybeProp sc c0
 1446                              let nm = vnName vn
 1447                              case mp of
 1448                                Nothing ->
 1449                                  fail ("goal_apply: could not find instantiation for " ++ show nm)
 1450                                Just p -> pure (Right p)
 1451                         Just tm ->
 1452                           pure (Left tm)
 1453                 Just <$> traverse mkNewGoal ruleArgs
 1454 
 1455     asPiLists :: Term -> [([(VarName, Term)], Term)]
 1456     asPiLists t =
 1457       case asPi t of
 1458         Nothing -> [([], t)]
 1459         Just (nm, tp, body) ->
 1460           [ ((nm, tp) : args, concl) | (args, concl) <- asPiLists body ] ++ [([], t)]

 1485 -- | Attempt to prove a goal by applying the given theorem.  Any hypotheses of
 1486 --   the theorem will generate additional subgoals.
 1487 tacticApply :: (F.MonadFail m, MonadIO m) => SharedContext -> Theorem -> Tactic m ()
 1488 tacticApply sc thm = Tactic \goal ->
 1489   case goalProp goal of
 1490     gl ->
 1491       liftIO (propApply sc (thmProp thm) gl) >>= \case
 1492         Nothing -> fail "apply tactic failed: no match"
 1493         Just newterms ->
 1494           let newgoals =
 1495                 [ goal{ goalProp = p, goalType = goalType goal ++ ".subgoal" ++ show i }
 1496                 | Right p <- newterms
 1497                 | i <- [0::Integer ..]
 1498                 ] in
 1499           return ((), mempty, newgoals, \es -> ApplyEvidence thm <$> processEvidence newterms es)

 1502    processEvidence :: [Either Term Prop] -> [Evidence] -> IO [Either Term Evidence]
 1503    processEvidence (Left tm : xs) es     = (Left tm :) <$> processEvidence xs es
 1504    processEvidence (Right _ : xs) (e:es) = (Right e :) <$> processEvidence xs es
 1505    processEvidence []             []     = pure []
 1506    processEvidence _ _ = fail "apply tactic failed: evidence mismatch"

 1561 -- | Attempt to solve the given goal, usually via an automatic solver.
 1562 --   If the goal is discharged, return evidence for the goal.  If there
 1563 --   is a counterexample for the goal, the counterexample will be used
 1564 --   to indicate the goal is unsolvable. Otherwise, the goal will remain unchanged.
 1565 tacticSolve :: Monad m => (ProofGoal -> m (SolverStats, SolveResult)) -> Tactic m ()
 1566 tacticSolve f = Tactic \gl ->
 1567   do (stats, sres) <- lift (f gl)
 1568      case sres of
 1569        SolveSuccess e -> return ((), stats, [], leafEvidence e)
 1570        SolveUnknown   -> return ((), stats, [gl], passthroughEvidence)
 1571        SolveCounterexample cex -> throwError (stats, cex)
