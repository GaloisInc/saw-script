   35 -- | Parse an untyped module declaration from a (lazy) Text
   36 readModule :: FilePath -> FilePath -> TL.Text -> Either Text Un.Module
   37 readModule base path b =
   38   case Un.parseSAW base path b of
   39     Right m -> Right m
   40     Left err -> Left $ Un.ppParseError err
   41 
   42 -- | Parse an untyped module from file
   43 readModuleFromFile :: FilePath -> IO (Either Text Un.Module)
   44 readModuleFromFile path =
   45   do base <- getCurrentDirectory
   46      txt <- TLIO.readFile path
   47      return $ readModule base path txt
   48 
   49 -- | Parse an untyped module from file and panic if it fails
   50 readBuiltinModuleFromFile :: FilePath -> IO Un.Module
   51 readBuiltinModuleFromFile path = do
   52   result <- readModuleFromFile path
   53   case result of
   54     Right m -> pure m
   55     Left msg ->
   56         panic "readBuiltinModuleFromFile" [
   57             "Module parsing failed:",
   58             msg
   59         ]
   60 
   61 -- | Record @path@ as a dependency for this compilation.
   62 addDep :: FilePath -> Q ()
   63 #if MIN_VERSION_template_haskell(2,7,0)
   64 addDep path = qAddDependentFile path
   65 #else
   66 addDep path = pure ()
   67 #endif
   68 
   69 -- | Read a SAWCore module from @file@ at compile time, and add a TH
   70 -- declaration of the name @str@ that is bound to that module (with
   71 -- Haskell type 'Un.Module') at runtime.
   72 defineModuleFromFile :: String -> FilePath -> Q [Dec]
   73 defineModuleFromFile decNameStr path =
   74   do addDep path
   75      m <- runIO $ readBuiltinModuleFromFile path
   76      let decName = mkName decNameStr
   77      moduleTp <- [t| Un.Module |]
   78      body <- TH.lift m
   79      pure [ SigD decName moduleTp
   80           , FunD decName [ Clause [] (NormalB body) [] ]
   81           ]
