   10 Queries over a Cryptol 'ME.ModuleEnv' for browsing loaded modules and
   11 submodules (as done by the SAW REPL's @:cbrowse@ command).
   12 
   13 Any loaded (sub)module can be browsed, whether or not it has been
   14 imported; it is named by its original name (not by an @import ... as@
   15 qualifier).  Private definitions are always included; the virtual
   16 submodules that Cryptol generates are not.

   55 -- | Why a (sub)module could not be browsed.  Each constructor holds
   56 --   the name as given.
   57 data BrowseError
   58   = ModuleNotLoaded Text
   59   | SubmoduleNotLoaded Text
   60   | SubmoduleNotFound Text
   61   | SubmoduleAmbiguous Text [Text]
   62       -- ^ with the fully qualified names of the candidates
   63 
   64 browseErrorText :: BrowseError -> Text
   65 browseErrorText err =
   66   case err of
   67     ModuleNotLoaded nm    -> "Module `" <> nm <> "' is not loaded."
   68     SubmoduleNotLoaded nm -> "Submodule `" <> nm <> "' is not loaded."
   69     SubmoduleNotFound nm  -> "Submodule `" <> nm <> "' not found."
   70     SubmoduleAmbiguous nm cands ->
   71       Text.intercalate "\n" $
   72         ("Submodule `" <> nm <> "' is ambiguous:")
   73         : map ("  " <>) cands
   74 
   75 -- | All loaded top-level modules (loaded directly or indirectly), in
   76 --   order of module name.
   77 loadedModules :: ME.ModuleEnv -> [ME.LoadedModule]
   78 loadedModules modEnv =
   79   sortOn ME.lmName $ ME.getLoadedModules (ME.meLoadedModules modEnv)
   80 
   81 -- | The names of all loaded top-level modules, in order.
   82 loadedModuleNames :: ME.ModuleEnv -> [Text]
   83 loadedModuleNames = map (CI.modNameToText . ME.lmName) . loadedModules
   84 
   85 -- | The names and contexts of all loaded top-level modules, in order.
   86 allModuleContexts :: ME.ModuleEnv -> [(Text, ME.ModContext)]
   87 allModuleContexts modEnv = map (topContext modEnv) (loadedModules modEnv)
   88 
   89 -- | The (fully qualified) name and context of a loaded top-level
   90 --   module (loaded directly or indirectly).
   91 moduleContext ::
   92   ME.ModuleEnv -> Text -> Either BrowseError (Text, ME.ModContext)
   93 moduleContext modEnv modName =
   94   case ME.lookupModule (CI.textToModName modName) modEnv of
   95     Nothing -> Left (ModuleNotLoaded modName)
   96     Just lm -> Right (topContext modEnv lm)
   97 
   98 topContext :: ME.ModuleEnv -> ME.LoadedModule -> (Text, ME.ModContext)
   99 topContext modEnv lm =
  100   ( CI.modNameToText (ME.lmName lm)
  101   , withoutVirtuals modEnv (ME.lmModContextWith MI.ifsDefines modEnv lm)
  102   )
  103 
  104 -- | The fully qualified name and context of a loaded submodule,
  105 --   submodule alias, or submodule functor (or a submodule nested in a
  106 --   functor), named by its path within its top-level module (e.g.
  107 --   @S1::S2@), or by its fully qualified path (e.g. @M::S1::S2@).
  108 --   Interface submodules are not supported (Cryptol's
  109 --   'ME.modContextOfWith' does not support them).
  110 submoduleContext ::
  111   ME.ModuleEnv -> Text -> Either BrowseError (Text, ME.ModContext)
  112 submoduleContext modEnv modName =
  113   case cands of
  114     [] -> Left (SubmoduleNotLoaded modName)
  115     [nm] ->
  116       case ME.modContextOfWith MI.ifsDefines (CP.ImpNested nm) modEnv of
  117         Nothing  -> Left (SubmoduleNotFound modName)
  118         Just ctx -> Right (fullName nm, withoutVirtuals modEnv ctx)
  119     nms -> Left (SubmoduleAmbiguous modName (map fullName nms))
  120   where
  121   want = Text.splitOn "::" modName
  122   cands = nub [ nm | lm <- ME.getLoadedModules (ME.meLoadedModules modEnv)
  123                    , nm <- nestedIn (ME.lmModule lm)
  124                    , want `elem` subPaths nm
  125               ]
  126   fullName nm = Text.intercalate "::" (last (subPaths nm))
  127 
  128 -- | The ways to name a submodule: its path within its top-level
  129 --   module, and its fully qualified path.
  130 subPaths :: MN.Name -> [[Text]]
  131 subPaths nm =
  132   case MN.nameModPathMaybe nm of
  133     Nothing -> []
  134     Just p ->
  135       let (top, ids) = CI.modPathSplit p
  136           rel = map CI.identText (ids ++ [MN.nameIdent nm])
  137       in  [rel, CI.modNameChunksText top ++ rel]
  138 
  139 -- | The submodules, submodule aliases, and submodule functors in a
  140 --   module, including those nested in functors.  (Those nested in
  141 --   ordinary submodules are already included in the module's maps.)
  142 nestedIn :: T.ModuleG a -> [MN.Name]
  143 nestedIn m =
  144      Map.keys (T.mSubmodules m)
  145   ++ Map.keys (T.mModAliases m)
  146   ++ Map.keys (T.mFunctors m)
  147   ++ concatMap nestedIn (Map.elems (T.mFunctors m))
  148 
  149 -- | The virtual submodules in all loaded modules (including those
  150 --   nested in functors): these are generated by Cryptol, e.g., to hold
  151 --   the arguments of a functor instantiation.
  152 virtuals :: ME.ModuleEnv -> Set MN.Name
  153 virtuals modEnv =
  154   Set.fromList
  155     [ nm | lm <- ME.getLoadedModules (ME.meLoadedModules modEnv)
  156          , nm <- virtualIn (ME.lmModule lm)
  157     ]
  158   where
  159   virtualIn :: T.ModuleG a -> [MN.Name]
  160   virtualIn m =
  161        Map.keys (Map.filter T.smVirtual (T.mSubmodules m))
  162     ++ concatMap virtualIn (Map.elems (T.mFunctors m))
  163 
  164 -- | Don't show virtual submodules (which Cryptol keeps private, but
  165 --   we show private names).
  166 withoutVirtuals :: ME.ModuleEnv -> ME.ModContext -> ME.ModContext
  167 withoutVirtuals modEnv ctx =
  168   ctx { ME.mctxExported =
  169           Set.difference (ME.mctxExported ctx) (virtuals modEnv) }
  170 
  171 -- | Display the names in a context's 'ME.mctxExported': Cryptol's
  172 --   @:browse@ sections, plus a section for submodule aliases.
  173 browseDoc :: ME.ModContext -> PP.Doc ann
  174 browseDoc ctx =
  175   PP.unAnnotate $
  176     CB.browseModContextWith sections CB.BrowseExported ctx
  177   where
  178   sections =
  179     [ CB.browseSignatures
  180     , CB.browseMods
  181     , CB.browseModAliases
  182     , CB.browseFunctors
  183     , CB.browseTSyns
  184     , CB.browsePrimTys
  185     , CB.browseNominalTypes
  186     , CB.browseVars
  187     ]
