   41 cbrowseCmdHelp :: Text
   42 cbrowseCmdHelp =
   43   "list modules '-l', or browse modules ['MODULE' | 'submodule SUBMODULE']"
   44 
   45 -- | Display the contents of a Cryptol module (@:cbrowse MODULENAME@)
   46 --   or submodule (@:cbrowse submodule SUBMODULENAME@).  Any loaded
   47 --   (sub)module can be browsed, whether or not it has been imported;
   48 --   it is named by its original name (not by an @import ... as@
   49 --   qualifier).  Private definitions are always included.  With no
   50 --   arguments (@:cbrowse@), all loaded top-level modules are shown;
   51 --   with @:cbrowse -l@, just their names are listed.
   52 --   See "CryptolSAWCore.Browse".
   53 cbrowseCmd :: [Text] -> REPL ()
   54 cbrowseCmd args =
   55   case args of
   56     []                     -> browseAll
   57     ["-l"]                 -> mapM_ say . Browse.loadedModuleNames
   58                                 =<< getModEnv
   59     [modName] | modName /= "submodule"
   60                            -> browseWith Browse.moduleContext modName
   61     ["submodule", modName] -> browseWith Browse.submoduleContext modName
   62     _ -> say $
   63            "Usage: :cbrowse -l"
   64            <> " | :cbrowse [MODULENAME]"
   65            <> " | :cbrowse submodule SUBMODULENAME"
   66   where
   67   browseAll = do
   68     ctxs <- Browse.allModuleContexts <$> getModEnv
   69     sequence_ $ intersperse (say "") $ map (uncurry showCtx) ctxs
   70 
   71   browseWith find modName = do
   72     modEnv <- getModEnv
   73     case find modEnv modName of
   74       Left err          -> say (Browse.browseErrorText err)
   75       Right (name, ctx) -> showCtx name ctx
   76 
   77   say = liftIO . TextIO.putStrLn
   78 
   79   -- Show a (sub)module's contents under a "From MODNAME" header,
   80   -- underlined, with the contents indented by 2.
   81   showCtx :: Text -> Browse.ModContext -> REPL ()
   82   showCtx modName ctx = do
   83     ppopts <- getPPOpts
   84     let header = "From " <> modName
   85         -- Cryptol's output has whitespace-only lines and trailing
   86         -- blank lines; trim them.
   87         body = dropWhileEnd Text.null $ map Text.stripEnd $
   88                  Text.lines $ PPS.renderText ppopts (Browse.browseDoc ctx)
   89         indent l = if Text.null l then l else "  " <> l
   90     say $ Text.intercalate "\n" $
   91       header : Text.replicate (Text.length header) "-" : map indent body
   92 
   93 getModEnv :: REPL Browse.ModuleEnv
   94 getModEnv = do
   95   rw <- getTopLevelRW
   96   liftIO $ eModuleEnv (rwSharedContext rw)
