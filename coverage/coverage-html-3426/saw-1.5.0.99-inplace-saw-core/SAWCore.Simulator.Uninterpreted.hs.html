   43 -- | Given a set of 'VarIndex'es representing constants with
   44 -- higher-order function types, search a term and collect every
   45 -- subterm that is a partial application of one such constant.
   46 -- The partial application should include as many arguments as
   47 -- necessary to make the resulting function first-order.
   48 --
   49 -- For each such partial application in the term, replace it with a
   50 -- fresh free variable that has a first-order function type.
   51 -- Also create a substitution suitable for use with 'scInstantiate'
   52 -- that maps each new variable to the original subterm.
   53 --
   54 -- For example, the SAWCore function @foldl@ has the following type:
   55 --
   56 -- @foldl : (a b : sort 0) -> (n : Nat) -> (b -> a -> b) -> b -> Vec n a -> b@
   57 --
   58 -- Say the term has a subterm with @foldl@ applied to six arguments:
   59 -- @foldl Bool Bool 8 xor False 0x55@.
   60 -- The partial application @foldl Bool Bool 8 xor@ has the minimum
   61 -- number of arguments required to give it a first-order function
   62 -- type: @Bool -> Vec 8 -> Bool@.
   63 --
   64 -- Thus all occurrences of the subterm @foldl Bool Bool 8 xor@ are
   65 -- replaced with a new free variable @u@ of type @Bool -> Vec 8 Bool
   66 -- -> Bool@.
   67 --
   68 -- The final result will be a term containing @u False 0x55@, and
   69 -- a substitution mapping @u@ to @foldl Bool Bool 8 xor@.
   70 --
   71 -- The preprocessed term should now be suitable for use with a
   72 -- symbolic evaluation backend that supports uninterpreted first-order
   73 -- functions.
   74 
   75 generalizeHigherOrderFunctions ::
   76   SharedContext -> Set VarIndex -> Term -> IO (Term, IntMap Term)
   77 generalizeHigherOrderFunctions sc vs t0 =
   78   do vs' <- filterCandidates sc vs
   79      pats <- findFirstOrderApps sc vs' t0
   80      (ss, sub) <- mkSimpsetSub sc pats
   81      ((), t1) <- rewriteSharedTerm sc ss t0
   82      pure (t1, sub)
   83 
   84 -- | Given a set of SAWCore constants identified by 'VarIndex', filter
   85 -- them and return the set of constants with higher-order function
   86 -- types.
   87 filterCandidates :: SharedContext -> Set VarIndex -> IO (Set VarIndex)
   88 filterCandidates sc vs =
   89   Set.fromList <$> filterM (isHigherOrderVarIndex sc) (Set.toList vs)
   90 
   91 -- | Look up the type of a SAWCore constant by the 'VarIndex' of its
   92 -- name, and return 'True' if it has a higher-order function type.
   93 isHigherOrderVarIndex :: SharedContext -> VarIndex -> IO Bool
   94 isHigherOrderVarIndex sc i =
   95   do mm <- scGetModuleMap sc
   96      case lookupVarIndexInMap i mm of
   97        Just r -> not <$> isFirstOrderType sc (resolvedNameType r)
   98        Nothing -> pure False

  100 -- | Given a list of 'Term' patterns, invent a variable for each.
  101 -- Build a 'Simpset' to turn the patterns into the corresponding
  102 -- variables and a substitution to convert them back.
  103 mkSimpsetSub :: SharedContext -> [Term] -> IO (Simpset a, IntMap Term)
  104 mkSimpsetSub _ [] = pure (emptySimpset, IntMap.empty)
  105 mkSimpsetSub sc (t : ts) =
  106   do (ss, sub) <- mkSimpsetSub sc ts
  107      vn <- scFreshVarName sc "u"
  108      let vars = getAllVars t
  109      ty <- scTypeOf sc t
  110      ty' <- scPiList sc vars ty
  111      v <- scVariable sc vn ty'
  112      rhs <- scApplyAll sc v =<< scVariables sc vars
  113      eq <- scGlobalApply sc "Prelude.Eq" [ty, t, rhs]
  114      prop <- scPiList sc vars eq
  115      let rule = ruleOfTerm prop Nothing
  116      let ss' = addRule rule ss
  117      f <- scLambdaList sc vars t
  118      let sub' = IntMap.insert (vnIndex vn) f sub
  119      pure (ss', sub')

  121 -- | Given a set of 'VarIndex'es representing constants with
  122 -- higher-order function types, search a term and collect every
  123 -- subterm that is a partial application of one such constant.
  124 -- The partial application should include as many arguments as
  125 -- necessary to make the resulting function first-order.
  126 --
  127 -- For example, the SAWCore function @foldl@ has the following type:
  128 --
  129 -- @foldl : (a b : sort 0) -> (n : Nat) -> (b -> a -> b) -> b -> Vec n a -> b@
  130 --
  131 -- Say the term has a subterm with @foldl@ applied to six arguments:
  132 -- @foldl Bool Bool 8 xor False 0x55@.
  133 -- The final two arguments, @False@ and @0x55@, have simple non-function
  134 -- types, so they can be skipped over.
  135 -- However, the @xor@ argument has a function type, so the partial
  136 -- application must include it.
  137 -- Thus the result list will include @foldl Bool Bool 8 xor@, which
  138 -- has the first-order function type @Bool -> Vec 8 Bool -> Bool@.
  139 findFirstOrderApps :: SharedContext -> Set VarIndex -> Term -> IO [Term]
  140 findFirstOrderApps sc vs t0 = snd <$> go (IntSet.empty, []) t0
  141   where
  142     -- foldlM :: (Foldable t, Monad m) => (b -> a -> m b) -> b -> t a -> m b
  143     go :: (IntSet, [Term]) -> Term -> IO (IntSet, [Term])
  144     go acc@(seen, ts) t
  145       | IntSet.member (termIndex t) seen = pure acc
  146       | otherwise =
  147           let seen' = IntSet.insert (termIndex t) seen in
  148           -- Is this term a designated higher-order function applied to arguments?
  149           case asConstant (fst (asApplyAll t)) of
  150             Just nm
  151               | Set.member (nameIndex nm) vs ->
  152                   -- If so, check whether the outermost argument has a
  153                   -- simple type.
  154                   case asApp t of
  155                     Nothing -> pure (seen', ts)
  156                     Just (_t1, t2) ->
  157                       do ty2 <- scTypeOf sc t2
  158                          simple <- isSimpleType sc ty2
  159                          case simple of
  160                            -- If it's simple, then we don't need to
  161                            -- record a pattern; recurse.
  162                            True -> foldlM go (seen', ts) (unwrapTermF t)
  163                            -- If it's *not* simple, then record a
  164                            -- pattern.
  165                            False -> pure (seen', t : ts)
  166             _ -> foldlM go (seen', ts) (unwrapTermF t)

  169 -- | Return true for any function type where all the argument types
  170 -- and the result type are simple types; return false otherwise.
  171 isFirstOrderType :: SharedContext -> Term -> IO Bool
  172 isFirstOrderType sc t =
  173   do t' <- scWhnf sc t
  174      case asPi t' of
  175        Just (_, a, b) ->
  176          do ok <- isSimpleType sc a
  177             if ok then isFirstOrderType sc b else pure False
  178        Nothing ->
  179          isSimpleType sc t'
  180 
  181 -- | Return false for any type containing a function type.
  182 -- Return true for primitive base types, pairs or vectors of simple
  183 -- types, and non-recursive datatypes with only simple constructor
  184 -- argument types.
  185 -- Precondition: The type of the term argument should be a sort.
  186 isSimpleType :: SharedContext -> Term -> IO Bool
  187 isSimpleType sc t =
  188   do t' <- scWhnf sc t
  189      case t' of
  190        -- We consider type variables to be simple, optimistically
  191        -- assuming that they will be instantiated with simple types.
  192        (asVariable -> Just _) -> pure True
  193        -- Pi types are not simple.
  194        (asPi -> Just _) -> pure False
  195        -- Sorts are simple.
  196        (asSort -> Just _) -> pure True
  197        -- Primitive base types are simple.
  198        (asGlobalDef -> Just "Prelude.String") -> pure True
  199        (asGlobalDef -> Just "Prelude.Float") -> pure True
  200        (asGlobalDef -> Just "Prelude.Double") -> pure True
  201        (asRationalType -> Just ()) -> pure True
  202        (asIntegerType -> Just ()) -> pure True
  203        (asIntModType -> Just _) -> pure True
  204        -- Booleans and naturals are simple.
  205        (asBoolType -> Just ()) -> pure True
  206        (asNatType -> Just ()) -> pure True
  207        -- Vectors and pairs are simple if their arguments are.
  208        (asVectorType -> Just (_n, a)) -> isSimpleType sc a
  209        (asPairType -> Just (a, b)) ->
  210          allM (isSimpleType sc) [a, b]
  211        -- Nonrecursive datatypes are simple if all their constructor
  212        -- argument types are.
  213        (asApplyAll -> (asConstant -> Just nm, args)) ->
  214          do mm <- scGetModuleMap sc
  215             -- look up name; if it's a datatype, then check all its arguments
  216             case lookupVarIndexInMap (nameIndex nm) mm of
  217               Just (ResolvedDataType dt) ->
  218                 do let sub = IntMap.fromList (zip (map (vnIndex . fst) (dtParams dt)) args)
  219                    allM (checkCtorArgStruct sub . ctorArgStruct) (dtCtors dt)
  220               _ -> pure False
  221        -- No other cases are considered simple.
  222        _ -> pure False
  223   where
  224     checkCtorArgStruct :: IntMap Term -> CtorArgStruct -> IO Bool
  225     checkCtorArgStruct sub cas =
  226       allM (checkCtorArg sub . snd) (ctorArgs cas)
  227 
  228     checkCtorArg :: IntMap Term -> CtorArg -> IO Bool
  229     checkCtorArg sub carg =
  230       case carg of
  231         ConstArg ty -> isSimpleType sc =<< scInstantiate sc sub ty
  232         RecursiveArg{} -> pure False
  233

  235 allM :: (Monad m) => (a -> m Bool) -> [a] -> m Bool
  236 allM _ [] = pure True
  237 allM f (x : xs) =
  238   do b <- f x
  239      if b then allM f xs else pure False
