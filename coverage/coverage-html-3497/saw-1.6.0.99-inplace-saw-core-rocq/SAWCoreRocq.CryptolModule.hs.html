   29 -- | Translate a list of named terms with their types to a Rocq definitions
   30 translateTypedTermMap ::
   31   TermTranslation.TermTranslationMonad m => [(Name,Term,Term)] -> m [Rocq.Decl]
   32 translateTypedTermMap defs = forM defs translateAndRegisterEntry
   33   where
   34     translateAndRegisterEntry (name, t, tp) = do
   35       let nameStr = Rocq.Ident (identText (nameIdent name))
   36       decl <-
   37         do t_trans <- TermTranslation.translateTerm t
   38            tp_trans <- TermTranslation.translateTerm tp
   39            return $ TermTranslation.mkDefinition nameStr t_trans tp_trans
   40       modify $ over TermTranslation.globalDeclarations (nameStr :)
   41       return decl
   42 
   43 -- | create a `Map Name TypedTerm` from the Cryptol.Module
   44 --   - this only extracts the exported values!
   45 --
   46 --   - FIXME: Thus, we see why, per #3491, the Rocq exporter does not
   47 --     export submodules. This is where (or close to) the fix should
   48 --     be made.
   49 
   50 cryptolModuleToMapOfTypedTerm
   51   :: SharedContext -> Cryptol.Module -> IO (Map.Map Name TypedTerm)
   52 cryptolModuleToMapOfTypedTerm sc m =
   53   do
   54   modEnv   <- eModuleEnv sc
   55   allterms <- eAllTerms sc
   56   let
   57       ifaceDecls = CSC.getAllIfaceDecls modEnv
   58       types = Map.map Cryptol.ifDeclSig (Cryptol.ifDecls ifaceDecls)
   59 
   60       -- we're keeping only the Value exports of `m`:
   61       vNameSet = Cryptol.exported NSValue (Cryptol.mExports m)
   62 
   63   return $ Map.restrictKeys
   64              (Map.intersectionWith
   65                (\t x -> TypedTerm (TypedTermSchema t) x)
   66                  types
   67                  allterms)
   68              vNameSet

   70 -- | Translates a Cryptol.Module into a list of Rocq declarations.  This is done
   71 -- by going through the term map corresponding to the module, translating all
   72 -- terms, and accumulating the translated declarations of all top-level
   73 -- declarations encountered.
   74 translateCryptolModule ::
   75   SharedContext ->
   76   TranslationConfiguration ->
   77   -- | List of already translated global declarations
   78   [Rocq.Ident] ->
   79   Cryptol.Module ->
   80   IO (Either TranslationError [Rocq.Decl])
   81 translateCryptolModule sc configuration globalDecls cm =
   82   do tm <- cryptolModuleToMapOfTypedTerm sc cm
   83      defs <-
   84        forM (Map.assocs tm) $ \(nm, t) ->
   85          do tp <- ttTypeAsTerm sc t
   86             return (nm, ttTerm t, tp)
   87      mm <- scGetModuleMap sc
   88      return
   89        (reverse . view TermTranslation.topLevelDeclarations . snd <$>
   90         TermTranslation.runTermTranslationMonad
   91         configuration
   92         Nothing -- TODO: this should be Just no?
   93         mm
   94         globalDecls
   95         []
   96         (translateTypedTermMap defs))
