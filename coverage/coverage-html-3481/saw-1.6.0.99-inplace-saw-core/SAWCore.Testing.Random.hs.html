   54 randomFirstOrderValue :: (Applicative m, Functor m, MonadRandom m) =>
   55   FirstOrderType -> Compose Maybe m FirstOrderValue
   56 randomFirstOrderValue FOTBit =
   57   Compose (Just (FOVBit <$> getRandom))
   58 randomFirstOrderValue FOTInt =
   59   Compose (Just (FOVInt <$> randomInt))
   60 randomFirstOrderValue (FOTIntMod m) =
   61   Compose (Just (FOVIntMod m <$> getRandomR (0, toInteger m - 1)))
   62 randomFirstOrderValue FOTRational =
   63   Compose (Just (FOVRational <$> randomRational))
   64 randomFirstOrderValue (FOTFloat e p) =
   65   Compose (Just ((FOVFloat . FirstOrderFloat e p) <$> randomBigFloat e p))
   66 randomFirstOrderValue (FOTVec n FOTBit) =
   67   Compose (Just (FOVWord n <$> getRandomR (0, 2^n - 1)))
   68 randomFirstOrderValue (FOTVec n t) =
   69   FOVVec t <$> replicateM (fromIntegral n) (randomFirstOrderValue t)
   70 randomFirstOrderValue (FOTTuple ts) =
   71   FOVTuple <$> traverse randomFirstOrderValue ts
   72 randomFirstOrderValue (FOTRec fs) =
   73   FOVRec <$> traverse randomFirstOrderValue fs
   74 randomFirstOrderValue (FOTArray _ _) = Compose Nothing

   78 randomInt :: MonadRandom m => m Integer
   79 randomInt = getRandomR (-10^(6::Int), 10^(6::Int))

   81 randomRational :: MonadRandom m => m Rational
   82 randomRational = do
   83   numer <- randomInt
   84   denom <- getRandomR (1, 10^(6 :: Int))
   85   pure (numer % denom)

   87 randomBigFloat ::
   88   forall m.
   89   MonadRandom m =>
   90   -- | Exponent width
   91   Natural ->
   92   -- | Precision width
   93   Natural ->
   94   m BigFloat
   95 randomBigFloat e p = do
   96   let sz :: Integer
   97       sz = 5
   98   x <- getRandomR (0, 10*(sz+1))
   99   if | x < 2    -> pure bfNaN
  100      | x < 4    -> pure bfPosInf
  101      | x < 6    -> pure bfNegInf
  102      | x < 8    -> pure bfPosZero
  103      | x < 10   -> pure bfNegZero
  104      | x <= sz       -> randomSubnormal  -- about 10% of the time
  105      | x <= 4*(sz+1) -> randomBinary     -- about 40%
  106      | otherwise     -> randomNormal     -- remaining ~50%
  107   where
  108     opts = fpOpts e p NearEven
  109 
  110     eInt = fromIntegral @Natural @Int e
  111     pInt = fromIntegral @Natural @Int p
  112 
  113     -- Generates floats uniformly chosen from among all bitpatterns.
  114     randomBinary :: m BigFloat
  115     randomBinary = do
  116       -- NB: Use the size (e+p) below: 1 bit for the sign bit, e bits for the
  117       -- exponent, and (p - 1) bits for the mantissa for a total of
  118       -- (1 + e + (p - 1)) = (e+p) bits.
  119       v <- getRandomR (0, bit (eInt+pInt) - 1)
  120       pure $ bfFromBits opts v
  121 
  122     -- Generates floats corresponding to subnormal values. These are values
  123     -- with 0 biased exponent and nonzero mantissa.
  124     randomSubnormal :: m BigFloat
  125     randomSubnormal = do
  126       sgn <- getRandom
  127       -- NB: Use size (p - 1) bits below. `p` includes the implicit leading bit
  128       -- of the mantissa, which isn't explicitly included in the overall bit
  129       -- pattern.
  130       v <- getRandomR (1, bit (pInt - 1) - 1)
  131       let bf = bfFromBits opts v
  132       pure $ if sgn then bfNeg bf else bf
  133 
  134     -- Generates floats corresponding to normal values. These are values where
  135     -- the exponent bits are not all zeros and not all ones.
  136     randomNormal :: m BigFloat
  137     randomNormal = do
  138       sgn <- getRandom
  139       ex <- getRandomR (1, bit eInt - 2)
  140       si <- getRandomR (0, bit (pInt - 1) - 1)
  141       let v = (ex `shiftL` (pInt - 1)) .|. si
  142       let bf = bfFromBits opts v
  143       pure $ if sgn then bfNeg bf else bf

  145 execTest ::
  146   (F.MonadFail m, MonadRandom m, MonadIO m) =>
  147   SharedContext ->
  148   ModuleMap ->
  149   Map VarName (m FirstOrderValue) ->
  150   Term ->
  151   m (Maybe [(VarName, FirstOrderValue)])
  152 execTest sc mmap vars tm =
  153   do testVec <- sequence vars
  154      tm' <- liftIO $
  155              do argMap0 <- traverse (scFirstOrderValue sc) testVec
  156                 let argMap = IntMap.fromList [ (vnIndex x, v) | (x, v) <- Map.toList argMap0 ]
  157                 scInstantiate sc argMap tm
  158      case evalSharedTerm mmap Map.empty Map.empty tm' of
  159        -- satisfaible, return counterexample
  160        VBool True  -> return (Just (Map.toList testVec))
  161        -- not satisfied by this test vector
  162        VBool False -> return Nothing
  163        _ -> fail "execTest: expected boolean result from random testing"
  164 
  165 prepareSATQuery ::
  166   (MonadRandom m, F.MonadFail m, MonadIO m) =>
  167   SharedContext ->
  168   SATQuery ->
  169   IO (m (Maybe [(VarName, FirstOrderValue)]))
  170 prepareSATQuery sc satq
  171   | Set.null (satUninterp satq) =
  172        do varmap <- traverse prepareVar (satVariables satq)
  173           t <- satQueryAsTerm sc satq
  174           mmap <- scGetModuleMap sc
  175           return (execTest sc mmap varmap t)
  176   | otherwise = fail "Random testing cannot handle uninterpreted functions"
  177 
  178  where
  179    prepareVar fot =
  180      case randomFirstOrderValue fot of
  181        Compose (Just v) -> pure v
  182        _ -> fail ("Cannot randomly test argument of type: " ++ show fot)

  184 runManyTests ::
  185   RandT TFGen IO (Maybe [(VarName, FirstOrderValue)]) ->
  186   Integer ->
  187   IO (Maybe [(VarName, FirstOrderValue)])
  188 runManyTests m numtests = evalRandT (loop numtests) =<< newTFGen
  189   where
  190     loop n
  191       | n > 0 =
  192           m >>= \case
  193             Nothing  -> loop (n-1)
  194             Just cex -> return (Just cex)
  195 
  196       | otherwise = return Nothing
