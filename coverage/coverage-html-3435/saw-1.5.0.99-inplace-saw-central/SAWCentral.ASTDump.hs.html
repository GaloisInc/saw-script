    8 This is a dumper (not prettyprinter) for the SAWScript abstract syntax.
    9 
   10 For the moment it lives in SAWCentral so it's accessible from
   11 SAWCentral.Value.
   12 
   13 It mostly doesn't print source positions on the grounds that they're
   14 generally not interesting and take up space. (It _does_ print type
   15 provenance, though, including those positions, as that's less
   16 straightforward.)
   17 -}

   50 -- Cryptol doesn't have dumps so far, so provide some pretend ones.
   51 -- Unfortunately the guts of these things are mostly not exported...
   52 
   53 dumpIdent :: Cry.Ident -> Dump
   54 dumpIdent x = Dump.singleton $ Cry.identText x
   55 
   56 dumpModName :: Cry.ModName -> Dump
   57 dumpModName mn = Dump.singleton $ Cry.modNameToText mn
   58 
   59 dumpImportSpec :: Cry.ImportSpec -> Dump
   60 dumpImportSpec spec = case spec of
   61     Cry.Hiding names -> Dump.subelements "Hiding" $ map dumpIdent names
   62     Cry.Only names -> Dump.subelements "Only" $ map dumpIdent names

   66 -- SAWScript AST
   67 
   68 dumpKind :: Kind -> Dump
   69 dumpKind kind =
   70     Dump.singleton ("Kind " <> Text.pack (show $ kindNumArgs kind))
   71 
   72 dumpPosition :: Pos -> Dump
   73 dumpPosition pos =
   74     Dump.text $ ppPosition pos
   75 
   76 dumpTypeProvenance :: TypeProvenance -> Dump
   77 dumpTypeProvenance prov = case prov of
   78     TypeExplicit pos ->
   79         Dump.singleton ("Prov: Explicit at " <> ppPosition pos)
   80     TypeFresh pos ->
   81         Dump.singleton ("Prov: Fresh at " <> ppPosition pos)
   82     TypeFailed pos ->
   83         Dump.singleton ("Prov: Failed at " <> ppPosition pos)
   84     TypeFromForallNamed pos x f ->
   85         let pos' = ppPosition pos
   86             heading' = x <> " in " <> f
   87         in
   88         Dump.singleton ("Prov: Forall at " <> pos' <> ": named " <> heading')
   89     TypeFromForallFresh pos x f ->
   90         let pos' = ppPosition pos
   91             heading' = x <> " in " <> f
   92         in
   93         Dump.singleton ("Prov: Forall at " <> pos' <> ": fresh " <> heading')
   94     TypeFromElement pos tyctx ->
   95         let pos' = ppPosition pos
   96             tyctx' = ppTyCtx tyctx
   97         in
   98         Dump.singleton ("Prov: From element " <> tyctx' <> " at " <> pos')
   99     TypeFromContext pos tyctx ->
  100         let pos' = ppPosition pos
  101             tyctx' = ppTyCtx tyctx
  102         in
  103         Dump.singleton ("Prov: From context " <> tyctx' <> " at " <> pos')
  104     TypeFromFuncWithSig pos ->
  105         Dump.singleton ("Prov: From function header at " <> ppPosition pos)
  106     TypeFromFuncWithBody pos1 pos2 ->
  107         let pos1' = ppPosition pos1
  108             pos2' = ppPosition pos2
  109         in
  110         Dump.singleton ("Prov: From function header at " <> pos1' <>
  111                         " with body at " <> pos2')

  113 -- The only place this is used identifies it (it's in a field value)
  114 -- so it doesn't have to identify itself.
  115 dumpNamedParamInfo :: NamedParamInfo -> Dump
  116 dumpNamedParamInfo (NamedParamInfo n names) =
  117     let n' = Text.pack $ show n
  118         names' = Text.concat $ map (\name -> " " <> name) names
  119     in
  120     Dump.singleton (n' <> names')
  121 
  122 dumpType :: Type -> Dump
  123 dumpType ty0 = case ty0 of
  124     TyCon prov tycon args ->
  125         let prov' = dumpTypeProvenance prov in
  126         let tycon' = case tycon of
  127               TupleCon k -> "Tuple " <> Text.pack (show k)
  128               ArrayCon -> "Array"
  129               StringCon -> "String"
  130               TermCon -> "Term"
  131               TypeCon -> "Type"
  132               BoolCon -> "Bool"
  133               IntCon -> "Int"
  134               BlockCon -> "Block"
  135               AIGCon -> "AIG"
  136               CFGCon -> "CFG"
  137               JVMSpecCon -> "JVMSpec"
  138               LLVMSpecCon -> "LLVMSpec"
  139               MIRSpecCon -> "MIRSpec"
  140               ContextCon ProofScript -> "ProofScript"
  141               ContextCon TopLevel -> "TopLevel"
  142         in
  143         Dump.subelements ("TyCon " <> tycon') (prov' : map dumpType args)
  144     TyFunc prov npi params namedParams ret ->
  145         let prov' = dumpTypeProvenance prov in
  146         let dumpNamedParam (name, ty) =
  147                 Dump.subelement ("named " <> name) $ dumpType ty
  148         in
  149         let namedParams' = map dumpNamedParam $ Map.toList namedParams in
  150         Dump.fields "TyFunc" [
  151             ("provenance", prov'),
  152             ("NamedParamInfo", dumpNamedParamInfo npi),
  153             ("params", Dump.list $ map dumpType params),
  154             ("namedParams", Dump.list namedParams'),
  155             ("ret", dumpType ret)
  156         ]
  157     TyRecord prov members ->
  158         let dumpMember (name, ty) = (name, dumpType ty) in
  159         let prov' = dumpTypeProvenance prov
  160             members' = map dumpMember $ Map.toList members
  161         in
  162         Dump.fields "TyRecord" (("(provenance", prov') : members')
  163     TyVar prov name ->
  164         let prov' = dumpTypeProvenance prov in
  165         Dump.subelement ("TyVar " <> name) prov'
  166     TyUnifyVar prov n ->
  167         let prov' = dumpTypeProvenance prov in
  168         Dump.subelement ("TyUnifyVar " <> Text.pack (show n)) prov'
  169 
  170 dumpSchema :: Schema -> Dump
  171 dumpSchema (Forall foralls ty) =
  172     let foralls' = Text.concat $ map (\(_prov, name) -> name) foralls in
  173     Dump.subelement ("Schema {" <> foralls' <> "}") $ dumpType ty
  174 
  175 dumpSchemaPattern :: SchemaPattern -> Dump
  176 dumpSchemaPattern (SchemaPattern foralls tys) =
  177     let foralls' = Text.concat $ map (\(_prov, name) -> name) foralls in
  178     Dump.subelements ("SchemaPattern {" <> foralls' <> "}") $ map dumpType tys
  179 
  180 dumpExpr :: Expr -> Dump
  181 dumpExpr e0 = case e0 of
  182     Bool _ b ->
  183         Dump.subelement "Bool" $ Dump.bool b
  184     String _ txt ->
  185         Dump.subelement "String" $ Dump.qstring txt
  186     Int _ k ->
  187         Dump.subelement "Int" $ Dump.int k
  188     Code _ txt ->
  189         Dump.subelement "Code" $ Dump.surroundText' "{{" txt "}}"
  190     CType _ txt ->
  191         Dump.subelement "CType" $ Dump.surroundText' "{{" txt "}}"
  192     Array _ es ->
  193         Dump.subelements "Array" $ map dumpExpr es
  194     Block _ (stmts, final) ->
  195         Dump.subelements "Block" $ map dumpStmt stmts ++ [dumpExpr final]
  196     Tuple _ es ->
  197         Dump.subelements "Tuple" $ map dumpExpr es
  198     Record _ tbl ->
  199         Dump.fields "Record" $ map (\(n, e) -> (n, dumpExpr e)) $ Map.toList tbl
  200     Index _ e1 e2 ->
  201         Dump.fields "Array" [("e1", dumpExpr e1), ("e2", dumpExpr e2)]
  202     Lookup _ e1 name ->
  203         Dump.subelement ("Lookup " <> name) $ dumpExpr e1
  204     TLookup _ e1 n ->
  205         Dump.subelement ("TLookup " <> (Text.pack $ show n)) $ dumpExpr e1
  206     Var _ x ->
  207         Dump.singleton ("Var " <> x)
  208     Lambda _ (mname) _ params namedParams body ->
  209         let dumpNamedParam (name, (_namepos, (_fullpos, def, pat))) =
  210                 Dump.fields ("named " <> name) [
  211                     ("default", dumpExpr def),
  212                     ("pattern", dumpPattern pat)
  213                 ]
  214         in
  215         let heading' = case mname of
  216               Nothing -> "Lambda"
  217               Just name -> "Lambda " <> name
  218             params' = map dumpPattern params
  219             namedParams' = map dumpNamedParam $ Map.toList namedParams
  220             body' = dumpExpr body
  221         in
  222         Dump.fields heading' [
  223             ("params", Dump.list (params' ++ namedParams')),
  224             ("body", body')
  225         ]
  226     Application _ f args ->
  227         let dumpArg (optname, e) = case optname of
  228               Nothing -> dumpExpr e
  229               Just (_, name) -> Dump.subelement ("named " <> name) $ dumpExpr e
  230         in
  231         Dump.fields "Application" [
  232             ("function", dumpExpr f),
  233             ("args", Dump.list $ map dumpArg args)
  234         ]
  235     Let _ dg e ->
  236         Dump.fields "Let" [
  237             ("decls", dumpDeclGroup dg),
  238             ("body", dumpExpr e)
  239         ]
  240     TSig _ e ty ->
  241         Dump.fields "TSig" [
  242             ("expr", dumpExpr e),
  243             ("type", dumpType ty)
  244         ]
  245     IfThenElse _ c t f ->
  246         Dump.fields "IfThenElse" [
  247             ("cond", dumpExpr c),
  248             ("true", dumpExpr t),
  249             ("false", dumpExpr f)
  250         ]
  251 
  252 dumpPattern :: Pattern -> Dump
  253 dumpPattern pat = case pat of
  254     PImplicit _ mty ->
  255         Dump.subelement "PImplicit" $ Dump.maybe dumpType mty
  256     PWild _ mty ->
  257         Dump.subelement "PWild" $ Dump.maybe dumpType mty
  258     PVar _pos _npos name mty ->
  259         Dump.subelement ("PVar " <> name) $ Dump.maybe dumpType mty
  260     PTuple _pos pats ->
  261         Dump.subelements "PTuple" $ map dumpPattern pats
  262 
  263 dumpStmt :: Stmt -> Dump
  264 dumpStmt s0 = case s0 of
  265     StmtBind _pos pat e ->
  266         Dump.fields "StmtBind" [
  267             ("pat", dumpPattern pat),
  268             ("expr", dumpExpr e)
  269         ]
  270     StmtLet _pos RebindableVar dg ->
  271         Dump.subelement "StmtLet rebindable" $ dumpDeclGroup dg
  272     StmtLet _pos ReadOnlyVar dg ->
  273         Dump.subelement "StmtLet nonrebindable" $ dumpDeclGroup dg
  274     StmtCode _pos _tpos txt ->
  275         Dump.subelement "StmtCode" $ Dump.surroundText' "{{" txt "}}"
  276     StmtImport _pos imp ->
  277         let module' = case iModule imp of
  278               Left fp -> Dump.qstring (Text.pack fp)
  279               Right modname -> dumpModName modname
  280         in
  281         Dump.fields "StmtImport" [
  282             ("iIsSubmodule", Dump.bool $ iIsSubmodule imp),
  283             ("iModule", module'),
  284             ("iIsBacktick", Dump.bool $ iIsBacktick imp),
  285             ("iAs", Dump.maybe dumpModName $ iAs imp),
  286             ("iSpec", Dump.maybe dumpImportSpec $ iSpec imp)
  287         ]
  288     StmtInclude _pos name once ->
  289         let once' = if once then "once" else "always" in
  290         Dump.subelement ("Include " <> once') $ Dump.qstring name
  291     StmtTypedef _pos _npos name ty ->
  292         Dump.subelement ("Typedef " <> name) $ dumpType ty
  293     StmtPushdir _ path ->
  294         Dump.subelement "Pushdir" $ Dump.text (Text.pack path)
  295     StmtPopdir _ ->
  296         Dump.singleton "Popdir"
  297 
  298 dumpDecl :: Decl -> Dump
  299 dumpDecl d =
  300     Dump.fields "Decl" [
  301         ("dPos", Dump.text $ ppPosition $ dPos d),
  302         ("dPat", dumpPattern $ dPat d),
  303         ("dType", Dump.maybe dumpSchema $ dType d),
  304         ("dDef", dumpExpr $ dDef d)
  305     ]
  306 
  307 dumpDeclGroup :: DeclGroup -> Dump
  308 dumpDeclGroup dg = case dg of
  309     Recursive ds ->
  310         Dump.subelements "Recursive decls:" $ map dumpDecl ds
  311     NonRecursive d ->
  312         Dump.prepend "NonRecursive" $ dumpDecl d
