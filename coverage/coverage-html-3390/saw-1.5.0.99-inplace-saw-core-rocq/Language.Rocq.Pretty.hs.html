   29 -- | Mark the last entry in a list.
   30 markLast :: [a] -> [(Bool, a)]
   31 markLast xs =
   32     let once x results = (null results, x) : results in
   33     foldr once [] xs

   35 -- | Replace all occurrences of the double quote character @"@ with
   36 --   the string @""@, i.e., two copies of it, as this is how Rocq
   37 --   escapes double quote characters.
   38 escapeStringLit :: Text -> Text
   39 escapeStringLit str =
   40     let oneChar c = if c == '"' then "\"\"" else Text.singleton c in
   41     Text.concatMap oneChar str

   51 -- | Common code to print some names with a type
   52 prettyNamesAndType :: [Ident] -> Type -> PP.Doc ann
   53 prettyNamesAndType names ty =
   54     let names' = PP.hsep $ map prettyIdent names
   55         ty' = prettyTerm PrecNone ty
   56         long = PP.nest 3 $ names' <> ":" <+> ty'
   57         short = PP.group $ names' <> ":" <+> ty'
   58     in
   59     PP.flatAlt long short

   61 -- | Insert parens based on the first argument
   62 parensIf :: Bool -> PP.Doc ann -> PP.Doc ann
   63 parensIf p d = if p then PP.parens d else d

   69 -- | Check if two terms are the \"same\". This is used to fold together
   70 --   forall entries (like @forall (x y: nat)@) and so can be
   71 --   conservative. It does only as much work as needed to make real
   72 --   occurrences work, and calls everything else false. In particular
   73 --   we make no attempt to handle alpha-equivalence.
   74 sameTerm :: Term -> Term -> Bool
   75 sameTerm e1 e2 =
   76     case (e1, e2) of
   77         -- Cases we do not handle
   78         (Lambda{}, _) -> False
   79         (_, Lambda{}) -> False
   80         (Fix{}, _) -> False
   81         (_, Fix{}) -> False
   82         (Pi{}, _) -> False
   83         (_, Pi{}) -> False
   84         (Let{}, _) -> False
   85         (_, Let{}) -> False
   86         (Ltac{}, _) -> False
   87         (_, Ltac{}) -> False
   88 
   89         -- Matching cases we do handle
   90         (If c1 t1 f1, If c2 t2 f2) ->
   91             sameTerm c1 c2 && sameTerm t1 t2 && sameTerm f1 f2
   92         (App f1 args1, App f2 args2) ->
   93             sameTerm f1 f2 && length args1 == length args2 &&
   94                 and (zipWith sameTerm args1 args2)
   95         (Sort s1, Sort s2) ->
   96             s1 == s2
   97         (Var x1, Var x2) ->
   98             x1 == x2
   99         (ExplVar x1, ExplVar x2) ->
  100             x1 == x2
  101         (Ascription e1' t1, Ascription e2' t2) ->
  102             sameTerm e1' e2' && sameTerm t1 t2
  103         (NatLit n1, NatLit n2) ->
  104             n1 == n2
  105         (ZLit n1, ZLit n2) ->
  106             n1 == n2
  107         (List xs1, List xs2) ->
  108             length xs1 == length xs2 && and (zipWith sameTerm xs1 xs2)
  109         (StringLit s1, StringLit s2) ->
  110             s1 == s2
  111         (Scope e1' s1, Scope e2' s2) ->
  112             s1 == s2 && sameTerm e1' e2'
  113 
  114         -- Fill out the cases so if a new constructor appears the
  115         -- compiler reminds us to handle it.
  116         (If{}, _) -> False
  117         (App{}, _) -> False
  118         (Sort{}, _) -> False
  119         (Var{}, _) -> False
  120         (ExplVar{}, _) -> False
  121         (Ascription{}, _) -> False
  122         (NatLit{}, _) -> False
  123         (ZLit{}, _) -> False
  124         (List{}, _) -> False
  125         (StringLit{}, _) -> False
  126         (Scope{}, _) -> False
  127 
  128 -- | Maybe wrapper for sameTerm
  129 sameMaybeTerm :: Maybe Term -> Maybe Term -> Bool
  130 sameMaybeTerm me1 me2 = case (me1, me2) of
  131     (Nothing, Nothing) -> True
  132     (Nothing, Just _) -> False
  133     (Just _, Nothing) -> False
  134     (Just e1, Just e2) -> sameTerm e1 e2

  140 -- | Contracted regular binder.
  141 --
  142 --   Explicit parameters with no type are printed loose: "x"
  143 --   Explicit parameters with a type are grouped by type: "(x y: a)"
  144 --   Implicit parameters are grouped by maybe-type: "{x y}", "{x y: a}".
  145 --
  146 data Binders'
  147     = ExplicitUntypedBinder' Ident
  148     | ExplicitTypedBinders' [Ident] Type
  149     | ImplicitBinders' [Ident] (Maybe Type)
  150 
  151 -- | Contracted pi-binder.
  152 --
  153 --   Anonymous pi "binders" are just type names and are printed loose,
  154 --   with or without braces.
  155 --
  156 --   Named pi binders are grouped first by type, and then all together
  157 --   with a single "forall" keyword: "forall (x y: a) {z: b}, ..."
  158 --
  159 data PiBinders'
  160     = AnonPiBinder' BinderImplicity Type
  161     | NamedPiBinders' [(BinderImplicity, [Ident], Type)]
  162 
  163 -- | Fold together adjacent binders as allowed by the concrete syntax.
  164 contractBinders :: [Binder] -> [Binders']
  165 contractBinders bs0 =
  166     -- state frobs for when we're looking at implicit binders
  167     -- (Just $ Left s in the state)
  168     let newImplicit x mty = ([x], mty)
  169         addImplicit x mty (prevnames, prevmty) =
  170             if sameMaybeTerm mty prevmty then
  171                 Just (x : prevnames, prevmty)
  172             else
  173                 Nothing
  174         popImplicit (names, mty) =
  175             ImplicitBinders' (reverse names) mty
  176     in
  177     -- state frobs for when we're looking at explicit binders
  178     -- with types (Just $ right s in the state)
  179     let newExplicit x ty = ([x], ty)
  180         addExplicit x ty (prevnames, prevty) =
  181             if sameTerm ty prevty then
  182                 Just (x : prevnames, prevty)
  183             else
  184                 Nothing
  185         popExplicit (names, ty) =
  186             ExplicitTypedBinders' (reverse names) ty
  187     in
  188 
  189     -- fold function to do the contraction
  190     let once (results, state) b = case state of
  191           Nothing -> case b of
  192               Binder Implicit x mty ->
  193                   (results, Just $ Left $ newImplicit x mty)
  194               Binder Explicit x Nothing ->
  195                   let here = ExplicitUntypedBinder' x in
  196                   (here : results, Nothing)
  197               Binder Explicit x (Just ty) ->
  198                   (results, Just $ Right $ newExplicit x ty)
  199           Just (Left s) -> case b of
  200               Binder Implicit x mty ->
  201                   case addImplicit x mty s of
  202                       Nothing ->
  203                           let prev = popImplicit s in
  204                           (prev : results, Just $ Left $ newImplicit x mty)
  205                       Just s' ->
  206                           (results, Just $ Left s')
  207               Binder Explicit x Nothing ->
  208                   let prev = popImplicit s
  209                       here = ExplicitUntypedBinder' x
  210                   in
  211                   (here : prev : results, Nothing)
  212               Binder Explicit x (Just ty) ->
  213                   let prev = popImplicit s in
  214                   (prev : results, Just $ Right $ newExplicit x ty)
  215           Just (Right s) -> case b of
  216               Binder Implicit x mty ->
  217                   let prev = popExplicit s in
  218                   (prev : results, Just $ Left $ newImplicit x mty)
  219               Binder Explicit x Nothing ->
  220                   let prev = popExplicit s
  221                       here = ExplicitUntypedBinder' x
  222                   in
  223                   (here : prev : results, Nothing)
  224               Binder Explicit x (Just ty) ->
  225                   case addExplicit x ty s of
  226                       Nothing ->
  227                           let prev = popExplicit s in
  228                           (prev : results, Just $ Right $ newExplicit x ty)
  229                       Just s' ->
  230                           (results, Just $ Right s')
  231     in
  232     let (results, state) = foldl once ([], Nothing) bs0
  233         results' = case state of
  234             Nothing -> results
  235             Just (Left s) -> popImplicit s : results
  236             Just (Right s) -> popExplicit s : results
  237     in
  238     reverse results'
  239 
  240 -- | Type belonging to the internals of `contractPiBinders`.
  241 --
  242 --   (is there no way to put this inside the function?)
  243 type State = ([(BinderImplicity, [Ident], Type)], BinderImplicity, [Ident], Type)
  244 
  245 -- | Fold together adjacent pi-binders as allowed by the concrete
  246 --   syntax.
  247 --
  248 --   "Binders" without names can't be folded together, but names that
  249 --   share the same type can be, and a chain of named binders requires
  250 --   only one "forall" token.
  251 contractPiBinders :: [PiBinder] -> [PiBinders']
  252 contractPiBinders bs0 =
  253     -- State frobs.
  254     let newstate :: BinderImplicity -> Ident -> Type -> State
  255         newstate imp x ty = ([], imp, [x], ty)
  256 
  257         addstate :: BinderImplicity -> Ident -> Type -> State -> State
  258         addstate imp x ty (others, previmp, prevnames, prevty) =
  259             if imp == previmp && sameTerm ty prevty then
  260                 (others, previmp, x : prevnames, prevty)
  261             else
  262                 ((previmp, reverse prevnames, prevty) : others, imp, [x], ty)
  263 
  264         popstate :: State -> PiBinders'
  265         popstate (others, imp, names, ty) =
  266             let others' = (imp, reverse names, ty) : others in
  267             NamedPiBinders' $ reverse others'
  268     in
  269 
  270     -- fold function to do the contraction
  271     let once :: ([PiBinders'], Maybe State) -> PiBinder -> ([PiBinders'], Maybe State)
  272         once (results, state) b = case state of
  273           Nothing -> case b of
  274               PiBinder imp Nothing ty ->
  275                   let here = AnonPiBinder' imp ty in
  276                   (here : results, Nothing)
  277               PiBinder imp (Just x) ty ->
  278                   (results, Just $ newstate imp x ty)
  279           Just s -> case b of
  280               PiBinder imp Nothing ty ->
  281                   let prev = popstate s
  282                       here = AnonPiBinder' imp ty
  283                   in
  284                   (here : prev : results, Nothing)
  285               PiBinder imp (Just x) ty ->
  286                   let s' = addstate imp x ty s in
  287                   (results, Just s')
  288     in
  289     let (results, state) = foldl once ([], Nothing) bs0
  290         results' :: [PiBinders']
  291         results' = case state of
  292             Nothing -> results
  293             Just s -> popstate s : results
  294     in
  295     reverse results'

  301 -- | Type to hold the current expression precedence while printing
  302 data Prec
  303     = PrecNone
  304     | PrecLambda
  305     | PrecApp
  306     | PrecAtom
  307     deriving (Eq, Ord)
  308 
  309 -- | Print an `Ident`
  310 prettyIdent :: Ident -> PP.Doc ann
  311 prettyIdent (Ident s) = text s
  312 
  313 -- | Print a `sort`
  314 prettySort :: Sort -> PP.Doc ann
  315 prettySort s = case s of
  316     Prop -> "Prop"
  317     Set -> "Set"
  318     Type -> "Type"
  319 
  320 -- | Print an ordinary (lambda/let) binder (group)
  321 prettyBinder' :: Binders' -> PP.Doc ann
  322 prettyBinder' b = case b of
  323     ExplicitUntypedBinder' x ->
  324         prettyIdent x
  325     ExplicitTypedBinders' names ty ->
  326         PP.group $ PP.parens $ prettyNamesAndType names ty
  327     ImplicitBinders' names Nothing ->
  328         let names' = PP.hsep $ map prettyIdent names
  329             long = PP.nest 3 names'
  330             short = PP.group names'
  331             entry = PP.flatAlt long short
  332         in
  333         PP.group $ PP.parens entry
  334     ImplicitBinders' names (Just ty) ->
  335         PP.group $ PP.braces $ prettyNamesAndType names ty
  336 
  337 -- | Print a pi (forall) binder
  338 --   (we don't seem to have a representation for @exists@)
  339 prettyPiBinder' :: PiBinders' -> [PP.Doc ann]
  340 prettyPiBinder' b = case b of
  341     AnonPiBinder' Explicit ty ->
  342         [PP.group $ prettyTerm PrecApp ty <+> "->"]
  343     AnonPiBinder' Implicit ty ->
  344         [PP.group $ PP.braces (prettyTerm PrecApp ty) <+> "->"]
  345     NamedPiBinders' entries ->
  346         let prettyEntry (isLast, (imp, names, ty)) =
  347               let entry = prettyNamesAndType names ty
  348                   entry' = case imp of
  349                       Explicit -> PP.parens entry
  350                       Implicit -> PP.braces entry
  351                   entry'' = if isLast then entry' <> "," else entry'
  352               in
  353               PP.group entry''
  354         in
  355         -- FUTURE: it won't matter for the current downstream uses but
  356         -- we might want to graft the "forall" keyword to the first
  357         -- entry instead of letting it float loose.
  358         let entries' = map prettyEntry $ markLast entries in
  359         "forall" : entries'
  360 
  361 -- | Print a chain of bindings. Contract the chain where syntactically
  362 --   possible, because it's a lot more readable that way.
  363 --
  364 --   The return value is one doc for each contracted group so they can
  365 --   be laid out as desired.
  366 --
  367 prettyBinders :: [Binder] -> [PP.Doc ann]
  368 prettyBinders binders =
  369     map prettyBinder' $ contractBinders binders
  370 
  371 -- | Print a chain of foralls. Contract the chain where syntactically
  372 --   possible, because it's a lot more readable that way.
  373 --
  374 --   The return value is one doc for each contracted group, meaning
  375 --   that each entry under a "forall" keyword is sent back separately,
  376 --   so they can be laid out as desired.
  377 --
  378 prettyPiBinders :: [PiBinder] -> [PP.Doc ann]
  379 prettyPiBinders binders =
  380     concatMap prettyPiBinder' $ contractPiBinders binders

  382 -- | Common code for printing things shaped like function headers
  383 --
  384 prettyFnHeader ::
  385     PP.Doc ann -> [PP.Doc ann] -> Maybe (PP.Doc ann) -> PP.Doc ann ->
  386     PP.Doc ann
  387 prettyFnHeader intro bindings mbRet sep =
  388     -- Long form
  389     --    fun x y z ...
  390     --         a b c ...
  391     --         : ty
  392     --       =>
  393     -- Short form:
  394     --    fun x : ty =>
  395     --
  396     -- (the above where intro is "fun", sep is "=>")
  397     --
  398     let firstpieces = intro : bindings
  399         shortHeader = case mbRet of
  400             Nothing -> PP.group $ PP.hsep firstpieces <+> sep
  401             Just ret -> PP.group $ PP.hsep firstpieces <+> ":" <+> ret <+> sep
  402         longBindings = case mbRet of
  403             Nothing -> PP.fillSep firstpieces
  404             Just ret -> PP.fillSep firstpieces <> PP.line <> ":" <+> ret
  405         longHeader = PP.nest 5 longBindings <> PP.line <> PP.indent 3 sep
  406     in
  407     PP.flatAlt longHeader shortHeader
  408 
  409 -- | Common code for printing things shaped like functions
  410 --   (@header@ will normally be the result of `prettyFnHeader`)
  411 prettyFunction :: PP.Doc ann -> PP.Doc ann -> PP.Doc ann
  412 prettyFunction header body =
  413     -- Long form with long argument list and long body:
  414     --    fun x y z ...
  415     --         a b c ...
  416     --       =>
  417     --       ...
  418     --       x + 1
  419     -- Long form with short argument list and long body:
  420     --    fun x =>
  421     --       ...
  422     --       x + 1
  423     -- Long form with long argument list and short body:
  424     --    fun x y z ...
  425     --         a b c ...
  426     --       => x + 1
  427     -- Short form:
  428     --    fun x => x + 1
  429     --
  430     let longbody = PP.line <> PP.indent 3 body <> PP.line
  431         shortbody = PP.group body
  432         body' = PP.flatAlt longbody shortbody
  433         long = header <> body'
  434         short = PP.group $ header <+> body
  435     in
  436     PP.flatAlt long short

  438 -- | Print a term
  439 prettyTerm :: Prec -> Term -> PP.Doc ann
  440 prettyTerm p e0 =
  441   case e0 of
  442     Lambda binders e1 ->
  443         let binders' = prettyBinders binders
  444             e1' = prettyTerm PrecLambda e1
  445             header = prettyFnHeader "fun" binders' Nothing "=>"
  446         in
  447         parensIf (p > PrecLambda) $ prettyFunction header e1'
  448     Fix ident binders returnType body ->
  449         let ident' = prettyIdent ident
  450             binders' = prettyBinders binders
  451             returnType' = Just $ prettyTerm PrecNone returnType
  452             body' = prettyTerm PrecLambda body
  453             intro = "fix" <+> ident'
  454             header = prettyFnHeader intro binders' returnType' ":="
  455         in
  456         parensIf (p > PrecLambda) $ prettyFunction header body'
  457     Pi binders t ->
  458         let binders' = prettyPiBinders binders
  459             t' = prettyTerm PrecLambda t
  460             longbs = PP.nest 5 $ PP.fillSep binders'
  461             shortbs = PP.group $ PP.hsep binders'
  462             finalbs = PP.flatAlt longbs shortbs
  463             long = finalbs <> PP.line <> PP.indent 5 t'
  464             short = PP.group $ finalbs <+> t'
  465         in
  466         parensIf (p > PrecLambda) $ PP.flatAlt long short
  467     Let x binders mty t body ->
  468         let x' = prettyIdent x
  469             binders' = prettyBinders binders
  470             mty' = prettyTerm PrecNone <$> mty
  471             t' = prettyTerm PrecNone t
  472             body' = prettyTerm PrecLambda body
  473             intro = "let" <+> x'
  474             header = prettyFnHeader intro binders' mty' ":="
  475             longest = PP.vsep [header, PP.indent 3 t', "in", body']
  476             second = PP.vsep [PP.group (header <+> t' <+> "in"), body']
  477             shortest = PP.group (header <+> t' <+> "in" <+> body')
  478             shorter = PP.flatAlt second shortest
  479         in
  480         parensIf (p > PrecLambda) $ PP.flatAlt longest shorter
  481     If c t f ->
  482         let c' = prettyTerm PrecNone c
  483             t' = prettyTerm PrecNone t
  484             f' = prettyTerm PrecLambda f
  485             first = "if" <+> c' <+> "then"
  486             long = PP.vsep [first, PP.indent 3 t', "else", PP.indent 3 f']
  487             short = PP.group (first <+> t' <+> "else" <+> f')
  488         in
  489         parensIf (p > PrecLambda) $ PP.flatAlt long short
  490     App f [] ->
  491         prettyTerm p f
  492     App f args ->
  493         --
  494         -- The short form of this is obviously "f arg arg arg".
  495         --
  496         -- For longer forms, I'd like it to pick either
  497         --    blahblah
  498         --    blah f arg arg
  499         --           arg arg
  500         -- or
  501         --    blahblah
  502         --    blahblah blub blub blub
  503         --       f arg arg arg
  504         --         arg
  505         --
  506         -- for short f, or
  507         --    blahblah
  508         --    blah fffff arg arg
  509         --            arg arg
  510         -- or
  511         --    blahblah
  512         --    blahblah blub blub blub
  513         --       fffff arg arg arg
  514         --          arg
  515         --
  516         -- depending on how far off to the right it is. This is
  517         -- complicated by the fact that PP.width doesn't let you
  518         -- examine the width of a subdocument without emitting it,
  519         -- which is annoying and silly of it. (That is, the null
  520         -- usage of PP.width is "PP.width doc $ \_ -> PP.empty",
  521         -- not "PP.width doc $ \_ -> doc". The latter prints doc
  522         -- twice.)
  523         --
  524         -- The logic below does the following:
  525         --    - either indents by the width of f' (and a space) or by 3
  526         --      (this is in long_a)
  527         --    - inserts PP.line before f' if we'll be indenting the
  528         --      args by more than 6 spaces
  529         --      (this is in long)
  530         --
  531         let f' = prettyTerm PrecApp f
  532             args' = map (\a -> PP.group $ prettyTerm PrecAtom a) args
  533 
  534             short_a = PP.group $ PP.hsep (f' : args')
  535             short = parensIf (p > PrecApp) short_a
  536 
  537             long_a = PP.width f' $ \w ->
  538                 let w' = if w <= 5 then w + 1 else 3 in
  539                 " " <> PP.nest w' (PP.fillSep args')
  540             -- This must go here or the line break comes after the
  541             -- parens, and that's one thing for C but very weird here.
  542             long_b = parensIf (p > PrecApp) long_a
  543             long = PP.nesting $ \nest -> PP.column $ \col ->
  544                 if col < nest + 6 then long_b
  545                 else PP.line <> long_b
  546         in
  547         PP.flatAlt long short
  548     Sort s ->
  549         prettySort s
  550     Var x ->
  551         prettyIdent x
  552     ExplVar x ->
  553         let x' = prettyIdent x in
  554         parensIf (p > PrecApp) $ "@" <> x'
  555     Ascription tm tp ->
  556         let tm' = prettyTerm PrecApp tm
  557             tp' = prettyTerm PrecApp tp
  558             long = PP.nest 5 $ PP.fillSep [tm' <> ":", tp']
  559             short = PP.group $ tm' <> ":" <+> tp'
  560         in
  561         parensIf (p > PrecLambda) $ PP.flatAlt long short
  562     NatLit i ->
  563         if i > 1000 then
  564             -- Explicitly convert from Z if an integer is too big
  565             parensIf (p > PrecLambda) ("Z.to_nat" <+> integer i <> "%Z")
  566         else
  567             integer i
  568     ZLit i ->
  569         -- we use hex unless our integer is a positive or negative digit
  570         -- XXX: this cannot possibly work as intended for negative values
  571         if abs i > 9 then
  572             let ui = toInteger (fromInteger i :: Word64)
  573                 ui' = showHex ui []
  574             in
  575             text ("0x" <> Text.pack ui' <> "%Z")
  576         else if i < 0 then
  577             text ("(" <> Text.pack (show i) <> ")%Z")
  578         else
  579             text (Text.pack (show i) <> "%Z")
  580     List ts ->
  581         let ts' = PP.punctuate ";" $ map (prettyTerm PrecNone) ts
  582             long = PP.brackets (PP.line <> PP.indent 3 (PP.vsep ts') <> PP.line)
  583             short = PP.group $ PP.brackets $ PP.hsep ts'
  584         in
  585         PP.flatAlt long short
  586     StringLit s ->
  587         PP.dquotes (text $ escapeStringLit s)
  588     Scope term scope ->
  589         let term' = prettyTerm PrecAtom term
  590             scope' = text scope
  591         in
  592         term' <> "%" <> scope'
  593     Ltac s ->
  594         "ltac:" <> PP.parens (text s)
  595 
  596 -- | Print a single constructor
  597 prettyConstructor :: Constructor -> PP.Doc ann
  598 prettyConstructor (Constructor {..}) =
  599     let name' = prettyIdent constructorName
  600         ty' = prettyTerm PrecNone constructorType
  601     in
  602     PP.nest 5 $ "|" <+> PP.group (name' <> ":" <+> ty')

  604 -- | Print an inductive type declaration
  605 prettyInductive :: Inductive -> PP.Doc ann
  606 prettyInductive (Inductive {..}) =
  607     let name' = prettyIdent inductiveName
  608         params' = prettyBinders inductiveParameters
  609         indices' = prettyPiBinders inductiveIndices
  610         sort' = prettySort inductiveSort
  611         ctors' = map prettyConstructor inductiveConstructors
  612         intro' = "Inductive" <+> name'
  613         lhs' = case params' of
  614           [] -> intro' <> ":"
  615           _ ->
  616               let short = PP.group $ intro' <+> PP.hsep params' <+> ":"
  617                   longparams = PP.indent 5 (PP.vsep params')
  618                   long = PP.vsep [intro', longparams, PP.indent 3 ":"]
  619               in
  620               PP.flatAlt long short
  621         rhs' = case indices' of
  622           [] -> sort' <+> ":="
  623           _ ->
  624               let short = PP.group $ PP.hsep indices' <+> sort' <+> ":="
  625                   long = PP.nest 5 $ PP.fillSep $ indices' ++ [sort' <+> ":="]
  626               in
  627               PP.flatAlt long short
  628         header = PP.group (lhs' <+> rhs')
  629     in
  630     PP.vsep ([header] ++ ctors' ++ ["."])

  632 -- | Common code for the simple declarations
  633 prettyBasicDecl :: PP.Doc ann -> Ident -> Type -> PP.Doc ann
  634 prettyBasicDecl what nm ty =
  635     let nm' = prettyIdent nm
  636         ty' = prettyTerm PrecNone ty
  637         lhs' = PP.group $ what <+> nm' <> ":"
  638         long = PP.nest 3 $ lhs' <+> ty' <> "."
  639         short = PP.group $ lhs' <+> ty' <> "."
  640     in
  641     PP.flatAlt long short
  642 
  643 -- | Print a Definition
  644 prettyDefinition :: Ident -> [Binder] -> Maybe Type -> Term -> PP.Doc ann
  645 prettyDefinition nm params mty body =
  646     let nm' = prettyIdent nm
  647         params' = prettyBinders params
  648         mty' = prettyTerm PrecNone <$> mty
  649         body' = prettyTerm PrecNone body
  650         intro' = "Definition" <+> nm'
  651         lhs' = case params' of
  652           [] -> intro' <> ":"
  653           _ ->
  654               let short = PP.group $ intro' <+> PP.hsep params' <+> ":"
  655                   longparams = PP.indent 5 (PP.vsep params')
  656                   long = PP.vsep [intro', longparams, PP.indent 3 ":"]
  657               in
  658               PP.flatAlt long short
  659         rhs' = case mty' of
  660           Nothing -> ":="
  661           Just ty' -> ty' <+> ":="
  662         header = PP.group (lhs' <+> rhs')
  663     in
  664     let short = PP.group $ header <+> body' <> "."
  665         long = PP.nest 3 $ PP.vsep [header, body' <> "."]
  666     in
  667     PP.flatAlt long short
  668 
  669 -- | Print a top-level declaration
  670 prettyDecl :: Decl -> PP.Doc ann
  671 prettyDecl decl = case decl of
  672     Axiom nm ty ->
  673         prettyBasicDecl "Axiom" nm ty <> PP.hardline
  674     Parameter nm ty ->
  675         prettyBasicDecl "Parameter" nm ty <> PP.hardline
  676     Variable nm ty ->
  677         prettyBasicDecl "Variable" nm ty <> PP.hardline
  678     Comment s ->
  679         -- None of the comments we generate are multiline. If that ever
  680         -- changes, we'll need more logic here to print them properly.
  681         "(*" <+> text s <+> "*)" <> PP.hardline
  682     Definition nm binders mty body ->
  683         prettyDefinition nm binders mty body <> PP.hardline
  684     InductiveDecl ind ->
  685         prettyInductive ind <> PP.hardline

  687         let nm' = prettyIdent nm
